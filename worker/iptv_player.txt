        function sanitizeUrlForLog(urlStr) {
            if (!urlStr || typeof urlStr !== 'string') return '';
            if (!urlStr.startsWith('http://') && !urlStr.startsWith('https://')) {
                return urlStr.replace(/(token|authtoken|auth|key|jwt|sig|signature|pass|password|user|uid|ticket)=[^&\s"']+/gi, '$1=***');
            }
            try {
                const u = new URL(urlStr);
                const sensitiveParams = ['token', 'authtoken', 'auth', 'key', 'jwt', 'sig', 'signature', 'pass', 'password', 'user', 'uid', 'ticket'];
                for (const k of Array.from(u.searchParams.keys())) {
                    if (sensitiveParams.some(sp => k.toLowerCase().includes(sp))) {
                        u.searchParams.set(k, '***');
                    }
                }
                return u.toString();
            } catch(e) {
                return urlStr.replace(/(token|authtoken|auth|key|jwt|sig|signature|pass|password|user|uid|ticket)=[^&\s"']+/gi, '$1=***');
            }
        }

        function getLogTypeColor(type) {

        function isLogMatchingFilter(entry, filter) {

            if (filter === 'CHUNKS') return ['CHUNK', 'KEY'].includes(entry.type);

        // Bounded LRU Translation Cache (max 2500 entries, 7-day TTL)
        class TranslationLRUCache {
            constructor(maxEntries = 2500, ttlMs = 7 * 24 * 3600 * 1000) {
                this.max = maxEntries;
                this.ttl = ttlMs;
                this.cache = new Map();
            }
            get(key) {
                if (!this.cache.has(key)) return null;
                const item = this.cache.get(key);
                if (Date.now() - item.ts > this.ttl) {
                    this.cache.delete(key);
                    return null;
                }
                // Refresh LRU order
                this.cache.delete(key);
                this.cache.set(key, item);
                return item.val;
            }
            set(key, val) {
                if (this.cache.has(key)) {
                    this.cache.delete(key);
                } else if (this.cache.size >= this.max) {
                    // Evict oldest (first key in Map iterator)
                    const oldest = this.cache.keys().next().value;
                    this.cache.delete(oldest);
                }
                this.cache.set(key, { val: val, ts: Date.now() });
            }
            has(key) {
                return this.get(key) !== null;
            }
            clear() {
                this.cache.clear();
            }
        }

        function cleanStreamUrlKey(url) {
            try {
                const u = new URL(url);
                return (u.hostname + u.pathname).toLowerCase();
            } catch(e) {
                return String(url).split('?')[0].toLowerCase();
            }
        }

        function getSmartStreamRoute(url) {
            try {
                const routes = JSON.parse(localStorage.getItem('iptv_smart_routes') || '{}');
                const key = cleanStreamUrlKey(url);
                const item = routes[key];
                if (item && item.route && (Date.now() - item.ts < 7 * 24 * 3600 * 1000)) {
                    return item.route; // 'direct' or 'proxy'
                }
            } catch(e) {}
            return null;
        }

        function saveSmartStreamRoute(url, route) {
            try {
                const routes = JSON.parse(localStorage.getItem('iptv_smart_routes') || '{}');
                const key = cleanStreamUrlKey(url);
                routes[key] = { route: route, ts: Date.now() };
                const keys = Object.keys(routes);
                if (keys.length > 500) {
                    delete routes[keys[0]];
                }
                localStorage.setItem('iptv_smart_routes', JSON.stringify(routes));
            } catch(e) {}
        }

        function playStream(url, channelName, forceMode = null) {
            trackChannelView(channelName);
            if (activeChannel && activeChannel !== channelName && currentStreamUrl) {
                lastPlayedChannel = { name: activeChannel, url: currentStreamUrl };
            }
            currentStreamUrl = url;
            activeChannel = channelName;
            hideErrorOverlay();

            let mode = forceMode || document.getElementById('playbackMode').value;
            const isHttpsSite = window.location.protocol === 'https:';
            const isHttpStream = url.startsWith('http://');

            let upgradedDueToMixedContent = false;
            // Prevent Mixed-Content blockage on HTTPS by upgrading HTTP streams to proxy instantly (0s wait)
            if (isHttpsSite && isHttpStream && (mode === 'turbo-auto' || mode === 'direct-only' || mode === 'direct-proxy')) {
                mode = 'proxy-only';
                upgradedDueToMixedContent = true;
            }

            // Known providers that block direct browser requests via CORS or IP restriction
            // Auto-convert BBC Persian DASH (.mpd) to official working Akamai HLS (.m3u8)
            if (url.includes('bbc_persian_tv') && url.includes('.mpd')) {
                url = 'https://vs-hls-pushb-ww-live.akamaized.net/x=4/i=urn:bbc:pips:service:bbc_persian_tv/pc_hd_abr_v2.m3u8';
                showActionToast('📡 تبدیل خودکار لینک بی‌بی‌سی به HLS پرسرعت');
            }

            // Providers known to block direct browser requests via CORS or IP restrictions
            const isKnownProxyStream = url.includes('jmp2.uk') || url.includes('pluto.tv') || url.includes('rtve.es') || url.includes('presstv.ir') || url.includes('wns.live') || url.includes('persiana.live') || url.includes('wurl.com');

            // Check Smart Memory for previously learned successful route
            const learnedRoute = getSmartStreamRoute(url);

            let targetUrl = url;
            let useProxy = false;
            let fallbackToProxy = null;

            if ((mode === 'turbo-auto' || mode === 'direct-proxy') && (isKnownProxyStream || learnedRoute === 'proxy')) {
                // Known or previously learned proxy stream: Instant 0s proxy load!
                targetUrl = '/proxy-stream?url=' + encodeURIComponent(url);
                useProxy = true;
                fallbackToProxy = false;
            } else if (mode === 'turbo-auto' || mode === 'direct-proxy') {
                // Test direct with adaptive 3.5s turbo failover to proxy!
                targetUrl = url;
                fallbackToProxy = true;
            } else if (mode === 'proxy-only' || mode === 'proxy-direct') {
                targetUrl = '/proxy-stream?url=' + encodeURIComponent(url);
                useProxy = true;
                fallbackToProxy = false;
            } else if (mode === 'direct-only') {
                targetUrl = url;
                fallbackToProxy = null;
            }

            // Update UI Status with clear, live badge
            updateNowPlayingBadge(channelName, useProxy, upgradedDueToMixedContent);
            document.getElementById('streamUrlInput').value = url;
            document.getElementById('streamInfoPanel').style.display = 'flex';
            document.getElementById('vpnNotice').style.display = (useProxy || mode === 'proxy-only') ? 'block' : 'none';

            // Add to Recently Played
            addToRecent(channelName, url);
            if (!window.testedChannelsHistory) window.testedChannelsHistory = [];
            window.testedChannelsHistory.push({
                time: new Date().toLocaleTimeString('en-GB'),
                name: channelName,
                url: sanitizeUrlForLog(url),
                mode: mode,
                isProxy: useProxy
            });

            executeHlsLoad(url, targetUrl, channelName, useProxy, fallbackToProxy);
        }

        function executeHlsLoad(originalUrl, targetUrl, name, isProxy, fallbackOption) {

                    maxFragLookUpTolerance: 0.5,

                    abrBandWidthUpFactor: 0.70,         // پلکانی و نرم بالا رفتن کیفیت هنگام افزایش سرعت
                    abrEwmaDefaultEstimate: 1500000,    // تخمین پایه اولیه ۱.۵ مگابیت بر ثانیه

                    updateActiveQualityLabel();
                    syncActiveChannelLiveCapabilities();
                    addAppLog('HLS', 'مانیفست دریافت شد | کیفیت‌ها: ' + ((hls.levels && hls.levels.length) || 1) + ' | صدا: ' + ((hls.audioTracks && hls.audioTracks.length) || 1) + ' | زیرنویس: ' + ((hls.subtitleTracks && hls.subtitleTracks.length) || 0));
                });

                // Track parsed subtitle cues from HLS fragments & store in buffer
                hls.on(Hls.Events.CUES_PARSED, (event, data) => {
                    if (data && data.cues && data.cues.length > 0) {
                        currentSubCues = data.cues;
                        let newCuesCount = 0;
                        data.cues.forEach(c => {
                            if (c && c.text) {
                                const exists = activeSubCueBuffer.some(existing => existing.startTime === c.startTime && existing.text === c.text);
                                if (!exists) {
                                    activeSubCueBuffer.push(c);
                                    newCuesCount++;
                                }
                            }
                        });
                        const now = video.currentTime;
                        if (activeSubCueBuffer.length > 250) {
                            activeSubCueBuffer = activeSubCueBuffer.filter(c => c.endTime >= now - 120);
                        }
                        if (newCuesCount > 0) {
                            addAppLog('SUB', 'قطعات زیرنویس دریافت شد: ' + newCuesCount + ' دیالوگ جدید');
                        }
                        if (subTranslateEnabled) {
                            prefetchCuesList(data.cues);
                        }
                        renderActiveSubtitleCues();
                    }
                });
                hls.on(Hls.Events.SUBTITLE_TRACK_LOADED, (event, data) => {
                    if (video && video.textTracks) {
                        for (let i = 0; i < video.textTracks.length; i++) {
                            const tt = video.textTracks[i];
                            if (activeSubtitleTrack !== -1) {
                                tt.mode = 'hidden';
                                tt.oncuechange = renderActiveSubtitleCues;
                            }
                        }
                    }
                    if (hls && hls.subtitleTracks && activeSubtitleTrack >= 0 && hls.subtitleTracks[activeSubtitleTrack]) {
                        const st = hls.subtitleTracks[activeSubtitleTrack];
                        if (st.textTrack) {
                            st.textTrack.mode = 'hidden';
                            st.textTrack.oncuechange = renderActiveSubtitleCues;
                        }
                    }
                    addAppLog('SUB', 'تراک زیرنویس بارگذاری شد | کل تراک‌ها: ' + ((video && video.textTracks && video.textTracks.length) || 0));
                    renderActiveSubtitleCues();
                });
                hls.on(Hls.Events.SUBTITLE_FRAG_PROCESSED, () => renderActiveSubtitleCues());

                // Real-time resolution indicator in Auto mode & level switches
                hls.on(Hls.Events.LEVEL_SWITCHED, (event, data) => {
                    updateActiveQualityLabel();
                    addAppLog('HLS', 'سوئیچ سطح کیفیت به شماره ' + data.level);
                });
                hls.on(Hls.Events.FRAG_LOADED, () => {
                    clearTimeout(failoverWatchdogTimer);
                    refreshQualityItemLabels();
                });
                hls.on(Hls.Events.LEVEL_LOADED, () => {
                    clearTimeout(failoverWatchdogTimer);
                    refreshQualityItemLabels();
                });

                // Stall Detector & Gentle Recovery State Machine
                clearTimeout(stallCheckTimer);
                video.onwaiting = () => {
                    if (capturedGen !== currentGeneration) return;
                    clearTimeout(stallCheckTimer);

                    // waiting alone is NOT a failure; monitor before taking action
                    stallCheckTimer = setTimeout(() => {
                        if (capturedGen !== currentGeneration) return;
                        if (!video.paused && video.readyState < 3 && hls && currentPlaybackSession) {
                            const now = Date.now();
                            if (now - currentPlaybackSession.lastRecoveryTime < CONFIG.recovery.cooldownMs) {
                                return;
                            }
                            if (currentPlaybackSession.recoveryCount < CONFIG.recovery.maxActionsPerSession) {
                                currentPlaybackSession.recoveryCount++;
                                currentPlaybackSession.lastRecoveryTime = now;
                                console.warn("Buffer waiting detected (Action " + currentPlaybackSession.recoveryCount + "/" + CONFIG.recovery.maxActionsPerSession + "): Resuming load...");
                                addAppLog('STALL', 'هشدار فریز یا توقف بافر (اقدام ' + currentPlaybackSession.recoveryCount + '/' + CONFIG.recovery.maxActionsPerSession + ')');
                                try { hls.startLoad(); } catch(e){}
                            } else if (currentPlaybackSession.fullRestarts < CONFIG.recovery.fullRestartMaxPerSession) {
                                currentPlaybackSession.fullRestarts++;
                                currentPlaybackSession.lastRecoveryTime = now;
                                console.warn("Buffer persistent delay: Performing safe HLS media recovery...");
                                addAppLog('STALL', 'توقف ادامه‌دار: بازیابی پایپ‌لاین مدیا HLS...');
                                try {
                                    hls.recoverMediaError();
                                    hls.startLoad();
                                } catch(e){}
                            }
                        }
                    }, CONFIG.recovery.monitorBeforeRecoveryMs);
                };

                let streamPlayLogged = false;
                video.onplaying = () => {
                    clearTimeout(stallCheckTimer);
                    clearTimeout(failoverWatchdogTimer);
                    hideErrorOverlay();
                    updateNowPlayingBadge(name, isProxy);
                    updateActiveQualityLabel();
                    saveSmartStreamRoute(originalUrl, isProxy ? 'proxy' : 'direct');
                    if (!streamPlayLogged) {
                        streamPlayLogged = true;
                        addAppLog('SUCCESS', 'پخش تصویر با موفقیت آغاز شد 🟢 (' + (isProxy ? 'پروکسی ورکر' : 'اتصال مستقیم') + ')');
                    }
                };

                hls.on(Hls.Events.AUDIO_TRACKS_UPDATED, () => { detectAudioTracks(); syncActiveChannelLiveCapabilities(); });
                hls.on(Hls.Events.SUBTITLE_TRACKS_UPDATED, () => { detectSubtitleTracks(); syncActiveChannelLiveCapabilities(); });

                hls.on(Hls.Events.ERROR, (event, data) => {
                    if (data.fatal) {
                        clearTimeout(failoverWatchdogTimer);
                        const errCode = (data.response && data.response.code) ? data.response.code : (data.response && data.response.status ? data.response.status : null);

                        // If testing direct connection and an error occurs: INSTANT FAILOVER to proxy!
                        if (fallbackOption !== null) {
                            console.warn("Direct connection failed, turbo failover to proxy immediately...");
                            clearTimeout(failoverWatchdogTimer);
                            document.getElementById('nowPlayingText').innerHTML = '<span>' + name + '</span> <span class="status-badge badge-warning">سوئیچ توربو به پروکسی ورکر 🚀</span>';
                            saveSmartStreamRoute(originalUrl, 'proxy');
                            
                            // Watchdog: If proxy doesn't start playback within 15 seconds, show clear error
                            failoverWatchdogTimer = setTimeout(() => {
                                if (video.readyState < 2 && video.paused) {
                                    console.warn("Proxy failover watchdog reached timeout");
                                    clearTimeout(failoverWatchdogTimer);
                                    showErrorOverlay('سیگنال در دسترس نیست', 'سیگنال این شبکه حتی با پروکسی اختصاصی ورکر نیز دریافت نشد (احتمالاً سرور مبدأ شبکه خاموش یا تغییر آدرس داده است).', 'می‌توانید شبکه دیگری را امتحان کنید یا بعداً مجدداً تلاش نمایید.', false);
                                    document.getElementById('nowPlayingText').innerHTML = '<span>' + name + '</span> <span class="status-badge badge-error">سیگنال در دسترس نیست</span>';
                                    if (hls) { try { hls.stopLoad(); } catch(e){} }
                                }
                            }, 15000);

                            const nextUseProxy = fallbackOption;
                            const nextTarget = nextUseProxy ? ('/proxy-stream?url=' + encodeURIComponent(originalUrl)) : originalUrl;
                            executeHlsLoad(originalUrl, nextTarget, name, nextUseProxy, null);
                            return;
                        }

                        // On Proxy: If manifest failed to load, don't loop endlessly
                        if (data.details === Hls.ErrorDetails.MANIFEST_LOAD_ERROR || data.details === Hls.ErrorDetails.MANIFEST_LOAD_TIMEOUT) {
                            const diag = translateError(errCode, data.type, isProxy);
                            showErrorOverlay(diag.title, diag.msg, diag.suggestion, false);
                            document.getElementById('nowPlayingText').innerHTML = '<span>' + name + '</span> <span class="status-badge badge-error">سیگنال در دسترس نیست</span>';
                            return;
                        }

                        // On Proxy: Stage 1 & 2 recovery for Media Errors (codecs, buffer corruption)
                        if (data.type === Hls.ErrorTypes.MEDIA_ERROR) {
                            if (mediaErrorRecoveryCount === 0) {
                                mediaErrorRecoveryCount++;
                                console.warn("Sensitive Stream Recovery (Stage 1): Re-syncing media buffer...");
                                hls.recoverMediaError();
                                return;
                            } else if (mediaErrorRecoveryCount === 1) {
                                mediaErrorRecoveryCount++;
                                console.warn("Sensitive Stream Recovery (Stage 2): Swapping audio codec & re-syncing...");
                                hls.swapAudioCodec();
                                hls.recoverMediaError();
                                return;
                            }
                        }

                        // On Proxy: Recovery for temporary network drops
                        if (data.type === Hls.ErrorTypes.NETWORK_ERROR && networkErrorRecoveryCount < 4) {
                            networkErrorRecoveryCount++;
                            console.warn("Sensitive Stream Recovery (Attempt " + networkErrorRecoveryCount + "): Resuming network load...");
                            hls.startLoad();
                            return;
                        }

                        const diag = translateError(errCode, data.type, isProxy);
                        showErrorOverlay(diag.title, diag.msg, diag.suggestion, diag.showProxyRetry);
                        document.getElementById('nowPlayingText').innerHTML = '<span>' + name + '</span> <span class="status-badge badge-error">قطع استریم</span>';
                    }
                });

            } else if (video.canPlayType('application/vnd.apple.mpegurl')) {
                // iOS Safari Native HLS
                video.src = targetUrl;
                video.play().catch(() => {});

                if (isTestingDirect) {
                    clearTimeout(failoverWatchdogTimer);
                    failoverWatchdogTimer = setTimeout(() => {
                        if (video.readyState < 2 && video.paused) {
                            console.warn("iOS direct watchdog timeout (3.5s), failover to proxy...");
                            saveSmartStreamRoute(originalUrl, 'proxy');
                            const nextTarget = '/proxy-stream?url=' + encodeURIComponent(originalUrl);
                            executeHlsLoad(originalUrl, nextTarget, name, true, null);
                        }
                    }, 3500);
                }

                video.onerror = () => {
                    clearTimeout(failoverWatchdogTimer);
                    if (fallbackOption !== null) {
                        saveSmartStreamRoute(originalUrl, 'proxy');
                        const nextUseProxy = fallbackOption;
                        const nextTarget = nextUseProxy ? ('/proxy-stream?url=' + encodeURIComponent(originalUrl)) : originalUrl;
                        executeHlsLoad(originalUrl, nextTarget, name, nextUseProxy, null);
                    } else {
                        const diag = translateError(null, 'IOS_ERROR', isProxy);
                        showErrorOverlay(diag.title, diag.msg, diag.suggestion, diag.showProxyRetry);
                    }
                };
                video.onloadedmetadata = () => {
                    clearTimeout(failoverWatchdogTimer);
                    updateNowPlayingBadge(name, isProxy);
                    saveSmartStreamRoute(originalUrl, isProxy ? 'proxy' : 'direct');
                };
            }
        }


        // Dedicated Quality Selector in Stream Toolbar & Fullscreen HUD
        function getLevelBitrate(level) {
            return (level && level.bitrate && level.bitrate > 0) ? Math.round(level.bitrate / 1000) : 0;
        }

        function getLevelResolution(level) {
            if (level && level.width && level.height) return level.width + 'x' + level.height;
            return '';
        }

        function getLevelHeight(level) {
            return (level && level.height && level.height > 0) ? level.height : 0;
        }

        function hasReliableLevelMetadata(level) {
            return Boolean(getLevelHeight(level) > 0 || getLevelBitrate(level) > 0);
        }

        function formatQualityLabel(level, idx) {
            const h = getLevelHeight(level);
            const br = getLevelBitrate(level);
            if (h > 0 && br > 0) return '\u200E' + h + 'p · ' + br + ' kbps\u200E';
            if (h > 0) return '\u200E' + h + 'p\u200E';
            if (br > 0) return '\u200E' + br + ' kbps\u200E (کیفیت ' + (idx + 1) + ')';
            return 'کیفیت ' + (idx + 1);
        }

        function refreshQualityItemLabels() {
            if (!hls || !hls.levels || hls.levels.length === 0) return;
            hls.levels.forEach((level, idx) => {
                const newLabel = formatQualityLabel(level, idx);
                document.querySelectorAll('#qualityPopup .popup-item, #hudQualityPopup .popup-item').forEach(el => {
                    if (el.dataset.idx === String(idx)) {
                        el.innerText = newLabel;
                    }
                });
            });
            updateActiveQualityLabel();
        }

        function updateActiveQualityLabel() {
            const labels = [document.getElementById('qualityLabel'), document.getElementById('hudQualityLabel')].filter(Boolean);
            if (!hls) return;

            // If user manually chose a quality, lock it and do NOT revert to auto
            if (userSelectedQualityIndex !== -1) {
                let manualLabel = 'کیفیت ' + (userSelectedQualityIndex + 1);
                if (hls.levels && hls.levels[userSelectedQualityIndex]) {
                    manualLabel = formatQualityLabel(hls.levels[userSelectedQualityIndex], userSelectedQualityIndex);
                }
                labels.forEach(l => l.innerText = manualLabel);
                return;
            }

            // Auto Mode:
            let curIdx = hls.currentLevel;
            if (curIdx >= 0 && hls.levels && hls.levels[curIdx]) {
                const lvl = hls.levels[curIdx];
                const res = (lvl.height && lvl.height > 0) ? (lvl.height + 'p') : (lvl.bitrate ? ('\u200E' + Math.round(lvl.bitrate / 1000) + 'k\u200E') : ('کیفیت ' + (curIdx + 1)));
                labels.forEach(l => l.innerText = 'خودکار (' + res + ')');
            } else {
                labels.forEach(l => l.innerText = 'کیفیت (خودکار)');
            }
        }

        function setupQualityLevels() {
            const popups = [document.getElementById('qualityPopup'), document.getElementById('hudQualityPopup')].filter(Boolean);
            if (popups.length === 0) return;

            popups.forEach(popup => {
                popup.innerHTML = '';

                // Auto option
                const autoItem = document.createElement('div');
                const isAutoSelected = (userSelectedQualityIndex === -1 && (!hls || hls.autoLevelEnabled));
                autoItem.className = 'popup-item' + (isAutoSelected ? ' selected' : '');
                autoItem.dataset.idx = '-1';
                autoItem.innerText = 'خودکار (Auto)';
                autoItem.onclick = (e) => {
                    e.stopPropagation();
                    setQualityLevel(-1, 'خودکار (Auto)');
                };
                popup.appendChild(autoItem);

                if (!hls || !hls.levels || hls.levels.length <= 1) return;

                hls.levels.forEach((level, idx) => {
                    const item = document.createElement('div');
                    const resLabel = formatQualityLabel(level, idx);
                    const isSelected = (userSelectedQualityIndex === idx || (userSelectedQualityIndex === -1 && !hls.autoLevelEnabled && hls.currentLevel === idx));
                    item.className = 'popup-item' + (isSelected ? ' selected' : '');
                    item.dataset.idx = String(idx);
                    item.innerText = resLabel;
                    item.onclick = (e) => {
                        e.stopPropagation();
                        setQualityLevel(idx, resLabel);
                    };
                    popup.appendChild(item);
                });
            });

            updateActiveQualityLabel();
        }

        function setQualityLevel(levelIndex, labelText) {
            if (!hls) return;
            userSelectedQualityIndex = levelIndex;

            if (levelIndex === -1) {
                hls.currentLevel = -1; // Auto
                hls.loadLevel = -1;
                updateActiveQualityLabel();
                addAppLog('HLS', 'کیفیت تصویر روی حالت خودکار (Auto) تنظیم شد');
            } else {
                // Explicitly lock manual quality and disable automatic ABR switching:
                hls.currentLevel = levelIndex;
                hls.loadLevel = levelIndex;
                if (currentPlaybackSession) {
                    currentPlaybackSession.selectedLevel = levelIndex;
                    currentPlaybackSession.qualityMode = 'MANUAL';
                }
                [document.getElementById('qualityLabel'), document.getElementById('hudQualityLabel')].filter(Boolean).forEach(l => l.innerText = labelText);
                addAppLog('HLS', 'کیفیت تصویر روی سطح ' + levelIndex + ' (' + labelText + ') قفل شد');
            }
            document.querySelectorAll('#qualityPopup .popup-item, #hudQualityPopup .popup-item').forEach(el => {
                el.classList.toggle('selected', el.dataset.idx === String(levelIndex));
            });
            document.querySelectorAll('#qualityPopup, #hudQualityPopup').forEach(p => p.classList.remove('open'));
            showActionToast('📊 کیفیت: ' + (levelIndex === -1 ? 'خودکار' : labelText));
        }

        function cycleQualityLevel() {
            if (!hls || !hls.levels || hls.levels.length <= 1) {
                showActionToast('📊 تنها یک کیفیت برای این استریم موجود است');
                return;
            }
            let nextIdx;
            if (userSelectedQualityIndex === -1) {
                nextIdx = 0;
            } else {
                nextIdx = userSelectedQualityIndex + 1;
                if (nextIdx >= hls.levels.length) nextIdx = -1;
            }
            let label = 'خودکار (Auto)';
            if (nextIdx >= 0) {
                const lvl = hls.levels[nextIdx];
                label = formatQualityLabel(lvl, nextIdx);
            }
            setQualityLevel(nextIdx, label);
        }

        let audioCtx = null;
        let audioSourceNode = null;
        let monoSplitter = null;
        let monoMerger = null;
        let isMonoAudio = (localStorage.getItem('iptv_mono_audio') === 'true');

        function toggleMonoAudio(e) {
            if (e) e.stopPropagation();
            const video = document.getElementById('player');
            if (!video) return;

            isMonoAudio = !isMonoAudio;
            try { localStorage.setItem('iptv_mono_audio', String(isMonoAudio)); } catch(err) {}

            applyMonoAudioState();
            detectAudioTracks();
            showActionToast(isMonoAudio ? '🎧 حالت صدای مونو (پخش در هر دو گوش): فعال' : '🎧 حالت صدای استریو عادی: فعال');
        }

            detectAudioTracks();
        }

        function applyMonoAudioState() {
            const video = document.getElementById('player');
            if (!video) return;

            if (isMonoAudio) {
                try {
                    const AudioContextClass = window.AudioContext || window.webkitAudioContext;
                    if (!AudioContextClass) return;
                    if (!audioCtx) audioCtx = new AudioContextClass();
                    if (audioCtx.state === 'suspended') audioCtx.resume();
                    if (!audioSourceNode) audioSourceNode = audioCtx.createMediaElementSource(video);

                    try { audioSourceNode.disconnect(); } catch(e){}
                    if (!monoSplitter) monoSplitter = audioCtx.createChannelSplitter(2);
                    if (!monoMerger) monoMerger = audioCtx.createChannelMerger(2);

                    const mixGain = audioCtx.createGain();
                    mixGain.gain.value = 0.5;

                    audioSourceNode.connect(monoSplitter);
                    monoSplitter.connect(mixGain, 0);
                    monoSplitter.connect(mixGain, 1);
                    mixGain.connect(monoMerger, 0, 0);
                    mixGain.connect(monoMerger, 0, 1);
                    monoMerger.connect(audioCtx.destination);
                } catch(e) {
                    console.warn('Web Audio Mono setup:', e);
                }
            } else {
                if (audioSourceNode && audioCtx) {
                    try {
                        audioSourceNode.disconnect();
                        if (monoSplitter) monoSplitter.disconnect();
                        if (monoMerger) monoMerger.disconnect();
                        audioSourceNode.connect(audioCtx.destination);
                    } catch(e) {}
                }
            }
        }

        // Online Subtitle Switcher (زیرنویس) with Direct Overlay Engine, Size, Theme, Background & Position

        function setSubtitleTrack(idx, name) {
            activeSubtitleTrack = idx;
            const video = document.getElementById('player');
            const overlay = document.getElementById('customSubtitleOverlay');
            const labels = [document.getElementById('subTrackLabel'), document.getElementById('hudSubLabel')].filter(Boolean);

            if (idx === -1) {
                if (hls) {
                    hls.subtitleTrack = -1;
                    hls.subtitleDisplay = false;
                }
                if (video && video.textTracks) {
                    for (let i = 0; i < video.textTracks.length; i++) {
                        video.textTracks[i].mode = 'disabled';
                    }
                }
                if (overlay) {
                    overlay.innerText = '';
                    overlay.classList.remove('active');
                }
                labels.forEach(l => l.innerText = 'زیرنویس');
                addAppLog('SUB', 'زیرنویس خاموش شد');
            } else {
                if (hls) {
                    hls.subtitleDisplay = true;
                    hls.subtitleTrack = idx;
                }
                if (video && video.textTracks) {
                    for (let i = 0; i < video.textTracks.length; i++) {
                        video.textTracks[i].mode = 'hidden';
                        video.textTracks[i].oncuechange = renderActiveSubtitleCues;
                    }
                }
                labels.forEach(l => l.innerText = 'زیرنویس (' + name + ')');
                renderActiveSubtitleCues();
                addAppLog('SUB', 'زیرنویس فعال شد: شماره ' + idx + ' (' + name + ')');
            }

            document.querySelectorAll('#subPopup .popup-item, #hudSubPopup .popup-item').forEach(el => {
                el.classList.toggle('selected', el.dataset.idx === String(idx));
            });
            document.querySelectorAll('#subPopup, #hudSubPopup').forEach(p => p.classList.remove('open'));
        }

        function toggleSubtitleTrack() {
            if (!hls || !hls.subtitleTracks || hls.subtitleTracks.length === 0) {
                showActionToast('💬 هیچ زیرنویسی برای این استریم یافت نشد');
                return;
            }
            let nextIdx = activeSubtitleTrack + 1;
            if (nextIdx >= hls.subtitleTracks.length) nextIdx = -1;
            const title = nextIdx === -1 ? 'خاموش' : (hls.subtitleTracks[nextIdx].name || hls.subtitleTracks[nextIdx].lang || ('زیرنویس ' + (nextIdx + 1)));
            setSubtitleTrack(nextIdx, title);
            showActionToast('💬 زیرنویس: ' + title);
        }

        function isRtlText(text) {
            const rtlRegex = /[\u0591-\u07FF\uFB1D-\uFDFD\uFE70-\uFEFC]/;
            return rtlRegex.test(text);
        }


        function detectLanguageFromText(text) {
            if (!text || typeof text !== 'string') return 'auto';
            const s = text.trim();
            if (!s) return 'auto';

            // 1. Script / Non-Latin Unicode block detection
            if (/[\uAC00-\uD7AF\u1100-\u11FF]/.test(s)) return 'ko'; // Korean
            if (/[\u3040-\u309F\u30A0-\u30FF]/.test(s)) return 'ja'; // Japanese
            if (/[\u4E00-\u9FFF]/.test(s)) return 'zh';               // Chinese
            if (/[\u0900-\u097F]/.test(s)) return 'hi';               // Hindi
            if (/[\u0400-\u04FF]/.test(s)) return 'ru';               // Russian / Cyrillic
            if (/[\u0370-\u03FF]/.test(s)) return 'el';               // Greek

            // Arabic / Persian / Kurdish script
            if (/[\u0600-\u06FF\u0750-\u077F\uFB50-\uFDFF\uFE70-\uFEFF]/.test(s)) {
                if (/[ڕڵێۆڤ]/.test(s)) return 'ku'; // Kurdish Sorani
                if (/[گچپژ]/.test(s)) return 'fa';  // Persian
                if (/[يىةءؤإأ]/.test(s)) return 'ar'; // Arabic
                return 'ar';
            }

            // 2. Latin script language scoring
            let scoreEs = (s.match(/[áéíóúñ¿¡]/gi) || []).length * 3;
            let scoreTr = (s.match(/[ğışĞİŞ]/g) || []).length * 4 + (s.match(/[çöüÇÖÜ]/g) || []).length * 2;
            let scoreDe = (s.match(/[äöüßÄÖÜ]/g) || []).length * 3;
            let scoreFr = (s.match(/[àâçéèêëîïôûùüœæ]/gi) || []).length * 2;
            let scoreIt = (s.match(/[àèéìòù]/gi) || []).length * 2;
            let scoreKu = (s.match(/[êîû]/gi) || []).length * 3;
            let scoreEn = 0;

            const words = s.toLowerCase().replace(/[^\p{L}\s]/gu, ' ').split(/\s+/).filter(Boolean);

            const dictEs = new Set(['el','la','los','las','del','al','para','con','por','pero','como','mas','muy','tambien','donde','cuando','que','quien','fue','son','este','esta','esto','estos','estas','hay','haber','tiene','tienen','noticias','gobierno','receptivo','atento','cada','uno','misa','esperanza','policia','investiga','incidente','cerca','militar','encontrado','explosivos','detenidos']);
            const dictTr = new Set(['bir','bu','icin','cok','daha','gibi','kadar','var','yok','ben','sen','biz','siz','onlar','evet','hayir','ama','fakat','nasil','guzel','olan','olarak','edildi','oldu','bugun','aciklama','bulundu','kararlari','acikladi']);
            const dictDe = new Set(['der','die','das','den','dem','des','ein','eine','einer','einem','einen','nicht','und','von','mit','auf','fur','ist','sind','war','wurde','werden','aber','oder','wenn','dass','bundesregierung','massnahmen','beschlossen','heute']);
            const dictFr = new Set(['le','la','les','des','du','dans','pour','avec','sur','pas','plus','par','ce','cette','ces','qui','que','dont','est','sont','mais','donc','nous','vous','ils','elles','president','republique','devant','presse']);
            const dictIt = new Set(['il','lo','gli','dei','degli','delle','tra','fra','sono','siamo','siete','questo','questa','questi','queste','quello','quella','non','piu','anche','molto','grazie','ciao','buongiorno','presidente','annunciato','nuove','misure','sicurezza']);
            const dictEn = new Set(['the','is','are','was','were','been','have','has','had','do','does','did','will','would','shall','should','may','might','can','could','not','but','because','about','into','through','after','over','between','under','minister','delivered','speech','national','security','today']);
            const dictKu = new Set(['rojbas','heval','hun','cawa','delal','ere','ez','tu','em','ji','bo','li','di']);

            words.forEach(w => {
                if (dictEs.has(w)) scoreEs += 2;
                if (dictTr.has(w)) scoreTr += 2;
                if (dictDe.has(w)) scoreDe += 2;
                if (dictFr.has(w)) scoreFr += 2;
                if (dictIt.has(w)) scoreIt += 2;
                if (dictEn.has(w)) scoreEn += 2;
                if (dictKu.has(w)) scoreKu += 3;
            });

            const scores = [
                { lang: 'es', score: scoreEs },
                { lang: 'tr', score: scoreTr },
                { lang: 'de', score: scoreDe },
                { lang: 'fr', score: scoreFr },
                { lang: 'it', score: scoreIt },
                { lang: 'en', score: scoreEn },
                { lang: 'ku', score: scoreKu }
            ];

            scores.sort((a, b) => b.score - a.score);
            if (scores[0].score > 0) return scores[0].lang;
            return 'en';
        }

        function normalizeSubtitleText(raw) {
            if (!raw) return '';
            let text = raw.replace(/<[^>]+>/g, '').trim();
            if (text.includes('- ') || text.includes('— ') || text.includes(String.fromCharCode(10) + '-') || text.includes(String.fromCharCode(10) + '—')) {
                return text;
            }
            const lines = text.split(String.fromCharCode(10)).map(function(s) {
                return s.replace(String.fromCharCode(13), '').trim();
            }).filter(Boolean);
            return lines.join(' ');
        }

        function prefetchCuesList(cues) {
            if (!subTranslateEnabled || activeSubtitleTrack === -1 || !cues || !cues.length) return;
            const activeTrackObj = (hls && hls.subtitleTracks) ? hls.subtitleTracks[activeSubtitleTrack] : null;
            let sourceLang = detectSourceLangFromTrack(activeTrackObj);
            const targetLang = subTranslateTargetLang || 'fa';
            const curTime = (video && video.currentTime) || 0;

            // 3-Tier Strategy:
            // Tier 2 (Near Future): curTime - 1 to curTime + 12s -> Top queue priority
            // Tier 3 (Far Future Pre-buffer): curTime + 12s to curTime + 60s -> Background queue priority
            for (let i = 0; i < cues.length; i++) {
                const c = cues[i];
                if (!c || !c.text) continue;
                const clean = normalizeSubtitleText(c.text);
                if (!clean) continue;

                const itemLang = (sourceLang === 'auto') ? detectLanguageFromText(clean) : sourceLang;
                const cacheKey = itemLang + '_' + targetLang + '_' + clean;
                if (subTranslationMemory.has(cacheKey) || activeTranslationRequests.has(cacheKey)) continue;

                if (c.startTime >= (curTime - 1) && c.startTime <= (curTime + 12)) {
                    queueSubtitleForTranslation(clean, itemLang, targetLang, c.startTime || 0);
                } else if (c.startTime > (curTime + 12) && c.startTime <= (curTime + 60)) {
                    queueSubtitleForTranslation(clean, itemLang, targetLang, (c.startTime || 0) + 100);
                }
            }
        }

        function renderActiveSubtitleCues() {
            const overlay = document.getElementById('customSubtitleOverlay');
            if (!overlay) return;

            if (activeSubtitleTrack === -1) {
                overlay.innerText = '';
                overlay.classList.remove('active');
                return;
            }

            const video = document.getElementById('player');
            if (!video) return;

            let cueTexts = [];

            // Source 1: Check matching HLS TextTrack or any active textTracks in browser
            let targetTrack = null;
            if (hls && hls.subtitleTracks && hls.subtitleTracks[activeSubtitleTrack]) {
                const subMeta = hls.subtitleTracks[activeSubtitleTrack];
                if (subMeta.textTrack) targetTrack = subMeta.textTrack;
                if (!targetTrack && video.textTracks) {
                    for (let i = 0; i < video.textTracks.length; i++) {
                        const tt = video.textTracks[i];
                        if ((subMeta.name && tt.label === subMeta.name) || (subMeta.lang && tt.language === subMeta.lang)) {
                            targetTrack = tt;
                            break;
                        }
                    }
                }
            }
            if (!targetTrack && video.textTracks && activeSubtitleTrack >= 0 && activeSubtitleTrack < video.textTracks.length) {
                targetTrack = video.textTracks[activeSubtitleTrack];
            }

            if (targetTrack && targetTrack.activeCues && targetTrack.activeCues.length > 0) {
                for (let i = 0; i < targetTrack.activeCues.length; i++) {
                    const c = targetTrack.activeCues[i];
                    if (c && c.text) cueTexts.push(c.text);
                }
            }

            // Fallback: check any active textTrack on video
            if (cueTexts.length === 0 && video.textTracks) {
                for (let i = 0; i < video.textTracks.length; i++) {
                    const tt = video.textTracks[i];
                    if (tt.activeCues && tt.activeCues.length > 0) {
                        for (let j = 0; j < tt.activeCues.length; j++) {
                            const c = tt.activeCues[j];
                            if (c && c.text) cueTexts.push(c.text);
                        }
                        if (cueTexts.length > 0) break;
                    }
                }
            }

            // Source 2: Check activeSubCueBuffer or currentSubCues
            if (cueTexts.length === 0) {
                const curTime = video.currentTime;
                const pool = (activeSubCueBuffer && activeSubCueBuffer.length > 0) ? activeSubCueBuffer : currentSubCues;
                if (pool && pool.length > 0) {
                    for (let i = 0; i < pool.length; i++) {
                        const c = pool[i];
                        if (c && (c.startTime - 0.3) <= curTime && curTime <= (c.endTime + 0.3) && c.text) {
                            cueTexts.push(c.text);
                        }
                    }
                }
            }

            if (cueTexts.length > 0) {
                const cleanText = normalizeSubtitleText(cueTexts.join(String.fromCharCode(10)));
                if (cleanText) {
                    if (subTranslateEnabled && activeSubtitleTrack !== -1) {
                        const activeTrackObj = (hls && hls.subtitleTracks) ? hls.subtitleTracks[activeSubtitleTrack] : null;
                        let sourceLang = detectSourceLangFromTrack(activeTrackObj);
                        if (sourceLang === 'auto') {
                            const detected = detectLanguageFromText(cleanText);
                            if (detected !== 'auto') sourceLang = detected;
                        }
                        const targetLang = subTranslateTargetLang || 'fa';
                        const cacheKey = sourceLang + '_' + targetLang + '_' + cleanText;

                        let translated = subTranslationMemory.get(cacheKey);

                        // Compound check: if multiple cues are active, check if each was individually cached
                        if (!translated && cueTexts.length > 1) {
                            const parts = cueTexts.map(t => normalizeSubtitleText(t)).filter(Boolean);
                            const transParts = parts.map(p => subTranslationMemory.get(sourceLang + '_' + targetLang + '_' + p));
                            if (transParts.every(tp => tp && tp.length > 0)) {
                                translated = transParts.join(' ');
                                subTranslationMemory.set(cacheKey, translated);
                            }
                        }

                        if (translated) {
                            if (subTranslateMode === 'bilingual' && translated !== cleanText) {
                                overlay.textContent = '';
                                const transEl = document.createElement('div');
                                transEl.className = 'sub-line-translated';
                                transEl.dir = isRtlText(translated) ? 'rtl' : 'ltr';
                                transEl.textContent = translated;

                                const origEl = document.createElement('div');
                                origEl.className = 'sub-line-original';
                                origEl.dir = isRtlText(cleanText) ? 'rtl' : 'ltr';
                                origEl.textContent = cleanText;

                                overlay.appendChild(transEl);
                                overlay.appendChild(origEl);
                            } else {
                                overlay.innerText = translated;
                                overlay.dir = isRtlText(translated) ? 'rtl' : 'ltr';
                            }
                        } else {
                            // URGENT: Fetch active on-screen cue immediately via single /translate endpoint for 0ms delay!
                            fetchSubtitleTranslation(cleanText, sourceLang, targetLang);
                            if (cueTexts.length > 1) {
                                cueTexts.forEach(ct => {
                                    const cClean = normalizeSubtitleText(ct);
                                    if (cClean) fetchSubtitleTranslation(cClean, sourceLang, targetLang);
                                });
                            }
                            
                            // User preference: Fallback to original if delayed, or strictly keep hidden
                            if (subTranslateFallbackOriginal) {
                                overlay.innerText = cleanText;
                                overlay.dir = isRtlText(cleanText) ? 'rtl' : 'ltr';
                            } else {
                                overlay.innerText = '';
                                overlay.classList.remove('active');
                                return;
                            }
                        }
                    } else {
                        overlay.innerText = cleanText;
                        overlay.dir = isRtlText(cleanText) ? 'rtl' : 'ltr';
                    }
                    overlay.classList.add('active');
                    prefetchUpcomingSubtitles(video);
                } else {
                    overlay.innerText = '';
                    overlay.classList.remove('active');
                }
            } else {
                overlay.innerText = '';
                overlay.classList.remove('active');
            }
            prefetchUpcomingSubtitles(video);
        }

        async function fetchBatchTranslations(batchItems, sourceLang, targetLang) {
            if (!batchItems || !batchItems.length) return;
            const capturedGen = currentGeneration;
            const t0 = Date.now();

            const payloadTexts = batchItems.map(item => item.text);
            addAppLog('TRANS', 'ارسال ' + payloadTexts.length + ' دیالوگ جهت ترجمه به ' + targetLang.toUpperCase());

            try {
                const res = await fetch('/translate-batch', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json; charset=utf-8' },
                    body: JSON.stringify({
                        texts: payloadTexts,
                        sl: sourceLang,
                        tl: targetLang
                    })
                });

                if (res.ok) {
                    const data = await res.json();
                    if (data && Array.isArray(data.results)) {
                        let translatedCount = 0;
                        data.results.forEach((r, idx) => {
                            const origItem = batchItems[idx];
                            if (origItem && r && r.translated) {
                                const trans = r.translated.trim();
                                const orig = origItem.text.trim();
                                const cacheKey = sourceLang + '_' + targetLang + '_' + origItem.text;
                                if (trans) {
                                    subTranslationMemory.set(cacheKey, trans);
                                    translatedCount++;
                                }
                            }
                        });
                        const duration = Date.now() - t0;
                        addAppLog('TRANS', 'ترجمه ' + translatedCount + ' از ' + batchItems.length + ' دیالوگ دریافت شد (' + duration + 'ms)');
                        if (capturedGen === currentGeneration) {
                            renderActiveSubtitleCues();
                        }
                    } else {
                        addAppLog('TRANS', 'پاسخ ترجمه نامعتبر بود');
                    }
                } else {
                    addAppLog('TRANS', 'خطای سرور در ترجمه: وضعیت ' + res.status);
                }
            } catch(e) {
                console.warn('Batch translation fetch failed:', e);
                addAppLog('TRANS', 'خطای ارتباط در ترجمه: ' + (e.message || e));
            } finally {
                batchItems.forEach(item => {
                    const cacheKey = sourceLang + '_' + targetLang + '_' + item.text;
                    activeTranslationRequests.delete(cacheKey);
                });
            }
        }

        function queueSubtitleForTranslation(cleanText, sourceLang, targetLang, priorityTime = 0) {
            if (!cleanText || !cleanText.trim()) return;
            const text = cleanText.trim();
            const cacheKey = sourceLang + '_' + targetLang + '_' + text;
            if (subTranslationMemory.has(cacheKey) || activeTranslationRequests.has(cacheKey)) return;

            // Check if already in queue
            const exists = translationQueue.some(item => item.text === text && item.sl === sourceLang && item.tl === targetLang);
            if (!exists) {
                if (translationQueue.length >= CONFIG.translation.maxQueuedCues) {
                    // Evict lowest priority
                    translationQueue.pop();
                }
                translationQueue.push({
                    text: text,
                    sl: sourceLang,
                    tl: targetLang,
                    priorityTime: priorityTime,
                    ts: Date.now()
                });
            }

            if (!batchFlushTimer) {
                batchFlushTimer = setTimeout(flushTranslationQueue, CONFIG.translation.batchFlushMs);
            }
        }

        function flushTranslationQueue() {
            batchFlushTimer = null;
            if (!translationQueue.length || !subTranslateEnabled || activeSubtitleTrack === -1) return;

            // Sort queue by priorityTime ascending (current cue first)
            translationQueue.sort((a, b) => a.priorityTime - b.priorityTime);

            const firstItem = translationQueue[0];
            const targetSl = firstItem.sl;
            const targetTl = firstItem.tl;

            const batch = [];
            const remaining = [];
            for (let i = 0; i < translationQueue.length; i++) {
                const item = translationQueue[i];
                if (item.sl === targetSl && item.tl === targetTl && batch.length < CONFIG.translation.maxBatchCues) {
                    batch.push(item);
                } else {
                    remaining.push(item);
                }
            }
            translationQueue.length = 0;
            translationQueue.push(...remaining);

            if (!batch.length) return;

            batch.forEach(item => {
                const cacheKey = item.sl + '_' + item.tl + '_' + item.text;
                activeTranslationRequests.add(cacheKey);
            });

            fetchBatchTranslations(batch, targetSl, targetTl);

            if (translationQueue.length > 0) {
                batchFlushTimer = setTimeout(flushTranslationQueue, CONFIG.translation.batchFlushMs);
            }
        }

        async function fetchSubtitleTranslation(text, sourceLang, targetLang) {
            if (!text || !text.trim()) return;
            const cleanText = text.trim();
            const cacheKey = sourceLang + '_' + targetLang + '_' + cleanText;

            if (subTranslationMemory.has(cacheKey) || activeTranslationRequests.has(cacheKey)) {
                return;
            }

            activeTranslationRequests.add(cacheKey);

            try {
                const queryParams = new URLSearchParams({
                    q: cleanText,
                    sl: sourceLang,
                    tl: targetLang
                });
                const res = await fetch('/translate?' + queryParams.toString());
                if (res.ok) {
                    const data = await res.json();
                    if (data && data.translated) {
                        subTranslationMemory.set(cacheKey, data.translated);
                        renderActiveSubtitleCues();
                    }
                }
            } catch(e) {
                console.warn('Subtitle translation fetch failed:', e);
            } finally {
                activeTranslationRequests.delete(cacheKey);
            }
        }

        let lastPrefetchTimestamp = 0;
        function prefetchUpcomingSubtitles(video, force = false) {
            if (!subTranslateEnabled || activeSubtitleTrack === -1 || !video) return;
            const now = Date.now();
            if (!force && (now - lastPrefetchTimestamp < 1000)) return;
            lastPrefetchTimestamp = now;

            const activeTrackObj = (hls && hls.subtitleTracks) ? hls.subtitleTracks[activeSubtitleTrack] : null;
            let sourceLang = detectSourceLangFromTrack(activeTrackObj);
            const targetLang = subTranslateTargetLang || 'fa';
            const curTime = video.currentTime || 0;

            function processCue(c) {
                if (!c || !c.text) return;
                const clean = normalizeSubtitleText(c.text);
                if (!clean) return;

                const itemLang = (sourceLang === 'auto') ? detectLanguageFromText(clean) : sourceLang;
                const cacheKey = itemLang + '_' + targetLang + '_' + clean;
                if (subTranslationMemory.has(cacheKey) || activeTranslationRequests.has(cacheKey)) return;

                // 3-Tier Strategy:
                // Tier 2: 0 to 12s ahead (Top priority in batch)
                // Tier 3: 12 to 60s ahead (Pre-buffered ahead of time)
                if (c.startTime >= (curTime - 1) && c.startTime <= (curTime + 12)) {
                    queueSubtitleForTranslation(clean, itemLang, targetLang, c.startTime || 0);
                } else if (c.startTime > (curTime + 12) && c.startTime <= (curTime + 60)) {
                    queueSubtitleForTranslation(clean, itemLang, targetLang, (c.startTime || 0) + 100);
                }
            }

            // 1. Primary Source: activeSubCueBuffer (Hls.Events.CUES_PARSED)
            if (activeSubCueBuffer && activeSubCueBuffer.length > 0) {
                for (let i = 0; i < activeSubCueBuffer.length; i++) {
                    processCue(activeSubCueBuffer[i]);
                }
            }

            // 2. Video native textTracks fallback
            if (video.textTracks && activeSubtitleTrack >= 0 && activeSubtitleTrack < video.textTracks.length) {
                const track = video.textTracks[activeSubtitleTrack];
                if (track && track.cues) {
                    for (let i = 0; i < track.cues.length; i++) {
                        processCue(track.cues[i]);
                    }
                }
            }
        }

        // Action Toast HUD & Volume Wheel Controller
        let volumeToastTimer = null;
        function setVolumeDelta(delta) {
            const video = document.getElementById('player');
            if (!video) return;
            let newVol = Math.max(0, Math.min(1, video.volume + delta));
            newVol = Math.round(newVol * 100) / 100;
            video.volume = newVol;
            if (window.player) {
                try { window.player.volume = newVol; } catch(e) {}
            }
            if (newVol > 0 && video.muted) {
                video.muted = false;
                if (window.player) {
                    try { window.player.muted = false; } catch(e) {}
                }
            }
            showVolumeToast(newVol, video.muted);
        }

        function showVolumeToast(vol, isMuted) {
            const toast = document.getElementById('volumeScrollToast');
            if (!toast) return;
            const pct = Math.round(vol * 100);
            const icon = (isMuted || vol === 0) ? '🔇' : (vol < 0.5 ? '🔉' : '🔊');
            toast.innerText = icon + ' ' + pct + '٪';
            toast.classList.add('show');

                toast.classList.remove('show');

        // Multi-tier Wheel Volume Scroll (عملکرد قطعی در حالت عادی و تمام‌صفحه)
        function setupVolumeScroll() {
            const wrapper = document.getElementById('videoWrapper');
            const onWheel = (e) => {
                const isOverPlayer = e.target && (
                    e.target.closest('.plyr') ||
                    e.target.closest('#videoWrapper') ||
                    document.fullscreenElement
                );
                if (!isOverPlayer) return;
                e.preventDefault();
                const delta = e.deltaY < 0 ? 0.05 : -0.05;
                setVolumeDelta(delta);
            };

            if (wrapper) wrapper.addEventListener('wheel', onWheel, { passive: false });

            window.addEventListener('wheel', (e) => {
                if (document.fullscreenElement) {
                    const fsEl = document.fullscreenElement;
                    if (fsEl.contains(e.target) || e.target === fsEl) {
                        e.preventDefault();
                        const delta = e.deltaY < 0 ? 0.05 : -0.05;
                        setVolumeDelta(delta);
                    }
                }
            }, { passive: false });
        }

        function playNextChannel() {
            const list = (activeRenderList && activeRenderList.length > 0) ? activeRenderList : fullChannelList;
            if (!list || list.length === 0) return;
            let currentIndex = list.findIndex(c => c.name === activeChannel);
            let nextIndex = (currentIndex + 1) % list.length;
            const nextCh = list[nextIndex];
            if (nextCh) {
                highlightChannelInList(nextIndex);
                playStream(nextCh.url, nextCh.name);
            }
        }

        function playPrevChannel() {
            const list = (activeRenderList && activeRenderList.length > 0) ? activeRenderList : fullChannelList;
            if (!list || list.length === 0) return;
            let currentIndex = list.findIndex(c => c.name === activeChannel);
            let prevIndex = currentIndex <= 0 ? (list.length - 1) : (currentIndex - 1);
            const prevCh = list[prevIndex];
            if (prevCh) {
                highlightChannelInList(prevIndex);
                playStream(prevCh.url, prevCh.name);
            }
        }

            }
        }

        // Category & Playlist Loading
        async function loadCategory(catUrl) {
            const listEl = document.getElementById('channelList');
            listEl.innerHTML = '<div style="text-align:center;padding:40px;color:var(--text-muted);"><span class="material-icons animate-spin">refresh</span> در حال دریافت شبکه‌ها...</div>';
            
            if (catUrl === 'favorites') {
                renderChannels(myFavorites);
                return;
            }
            if (catUrl === 'recent') {
                renderChannels(myRecentChannels);
                return;
            }
            if (catUrl.startsWith('custom:')) {
                const listId = catUrl.split('custom:')[1];
                const found = myCustomLists.find(l => l.id === listId);
                if (found) {
                    if (found.type === 'single') {
                        renderChannels([{ name: found.name, url: found.content, logo: '', group: 'شخصی' }]);
                    } else if (found.type === 'file') {
                        parseAndDisplayM3U(found.content);
                    } else if (found.type === 'url') {
                        fetchM3U(found.content);
                    }
                }
                return;
            }

            fetchM3U(catUrl);
        }

        function parseAndDisplayM3U(text) {
            if (!text) return;
            const lines = text.split(String.fromCharCode(10));
            fullChannelList = [];
            let current = null;

            for (let i = 0; i < lines.length; i++) {
                const line = lines[i].replace(String.fromCharCode(13), '').trim();
                if (!line) continue;
                if (line.startsWith('#EXTINF:')) {
                    // Prompt.txt Section 55: Split on FIRST comma as delimiter to preserve commas in channel names
                    const commaIdx = line.indexOf(',');
                    const metaPart = commaIdx !== -1 ? line.substring(0, commaIdx) : line;
                    const name = commaIdx !== -1 ? line.substring(commaIdx + 1).trim() : 'کانال نامشخص';

                    let logo = '';
                    const logoMatch = metaPart.match(/tvg-logo=["'](.*?)["']/i);
                    if (logoMatch) logo = logoMatch[1];

                    let group = 'عمومی';
                    const groupMatch = metaPart.match(/group-title=["'](.*?)["']/i);
                    if (groupMatch) group = groupMatch[1].trim();

                    let tvgId = '';
                    const idMatch = metaPart.match(/tvg-id=["'](.*?)["']/i);
                    if (idMatch) tvgId = idMatch[1].trim();

                    current = { name, logo, group, tvgId, url: '' };
                } else if ((line.startsWith('http://') || line.startsWith('https://')) && current) {
                    current.url = line;
                    fullChannelList.push(current);
                    current = null;
                }
            }

            buildSubCategoryFilter(fullChannelList);
            applyCombinedFilters();
        }

        // Smart Channel Capabilities Manager (Multi-Audio & Subtitles Detection)
        // 4-Tier Zero-Waste: Heuristics -> 24h Cache -> Viewport IntersectionObserver -> /probe-stream
        // ============================================================================
        const CAP_CACHE_KEY = 'iptv_channel_capabilities_v1';
        const CAP_TTL_MS = 24 * 60 * 60 * 1000; // 24 Hours

        function getCachedCapability(url) {
            if (!url) return null;
            try {
                const store = JSON.parse(localStorage.getItem(CAP_CACHE_KEY) || '{}');
                const item = store[url];
                if (item && (Date.now() - item.ts < CAP_TTL_MS)) {
                    return item;
                }
            } catch(e) {}
            return null;
        }

        function setCachedCapability(url, data) {
            if (!url || !data) return;
            try {
                const store = JSON.parse(localStorage.getItem(CAP_CACHE_KEY) || '{}');
                store[url] = {
                    multiAudio: !!data.multiAudio,
                    audioCount: data.audioCount || (data.multiAudio ? 2 : 1),
                    subtitles: !!data.subtitles,
                    subCount: data.subCount || (data.subtitles ? 1 : 0),
                    ts: Date.now()
                };
                const keys = Object.keys(store);
                if (keys.length > 600) {
                    delete store[keys[0]];
                }
                localStorage.setItem(CAP_CACHE_KEY, JSON.stringify(store));
            } catch(e) {}
        }

        let channelCapObserver = null;

        let activeProbes = 0;
        const MAX_CONCURRENT_PROBES = 2;
        const pendingProbeTimeouts = new Map();

        function initChannelCapObserver() {
            // پروب خودکار کانال‌ها در پس‌زمینه متوقف شد تا لیست کانال‌ها با نهایت سرعت و بدون هنگ اسکرول شود

        function queueChannelProbe(el, url) {
            if (el.dataset.probed === 'true') return;
            const cached = getCachedCapability(url);
            if (cached) {
                el.dataset.probed = 'true';
                updateChannelCardBadges(el, cached);
                if (channelCapObserver) channelCapObserver.unobserve(el);
                return;
            }

            if (!probeQueue.some(item => item.url === url)) {
                probeQueue.push({ el, url });
                processProbeQueue();
            }
        }

        function processProbeQueue() {
            if (activeProbes >= MAX_CONCURRENT_PROBES || probeQueue.length === 0) return;

            const item = probeQueue.shift();
            if (!item || !item.el || !item.url) return;

            if (typeof document !== 'undefined' && document.contains && !document.contains(item.el)) {
                processProbeQueue();
                return;
            }

            activeProbes++;
            item.el.dataset.probed = 'true';
            if (channelCapObserver) channelCapObserver.unobserve(item.el);

            fetch('/probe-stream?url=' + encodeURIComponent(item.url))
                .then(r => r.json())
                .then(data => {
                    if (data && data.success) {
                        setCachedCapability(item.url, data);
                        updateChannelCardBadges(item.el, data);
                    }
                })
                .catch(() => {})
                .finally(() => {
                    activeProbes--;
                    setTimeout(processProbeQueue, 80);
                });
        }

        function applySorting(channels) {
            if (!channels || channels.length <= 1 || currentSortMode === 'default') {
                return channels;
            }
            const list = [...channels];
            if (currentSortMode === 'asc') {
                list.sort((a, b) => (a.name || '').localeCompare(b.name || '', 'fa', { numeric: true, sensitivity: 'base' }));
            } else if (currentSortMode === 'desc') {
                list.sort((a, b) => (b.name || '').localeCompare(a.name || '', 'fa', { numeric: true, sensitivity: 'base' }));
            } else if (currentSortMode === 'views') {
                try {
                    const views = JSON.parse(localStorage.getItem('iptv_channel_views_v2') || '{}');
                    list.sort((a, b) => {
                        const countA = views[a.name] || 0;
                        const countB = views[b.name] || 0;
                        if (countB !== countA) return countB - countA;
                        return (a.name || '').localeCompare(b.name || '', 'fa', { numeric: true, sensitivity: 'base' });
                    });
                } catch(e){}
            
            }
            return list;
        }

        function setupSubCategoryDragScroll(bar) {
            if (isSubCatDragSetup || !bar) return;
            isSubCatDragSetup = true;

            let isDown = false;
            let startX = 0;
            let scrollLeft = 0;

            bar.addEventListener('mousedown', (e) => {
                isDown = true;
                bar.classList.add('grabbing');
                startX = e.pageX - bar.offsetLeft;
                scrollLeft = bar.scrollLeft;
            });
            bar.addEventListener('mouseleave', () => {
                isDown = false;
                bar.classList.remove('grabbing');
            });
            bar.addEventListener('mouseup', () => {
                isDown = false;
                bar.classList.remove('grabbing');
            });
            bar.addEventListener('mousemove', (e) => {
                if (!isDown) return;
                e.preventDefault();
                const x = e.pageX - bar.offsetLeft;
                const walk = (x - startX) * 1.2;
                bar.scrollLeft = scrollLeft - walk;
            });
        }

        function buildSubCategoryFilter(channels) {
            const bar = document.getElementById('subCategoryBar');
            if (!bar) return;

            setupSubCategoryDragScroll(bar);

            activeSubCategory = 'ALL';
            if (!channels || channels.length === 0) {
                bar.style.display = 'none';
                return;
            }

            const groupCounts = {};
            channels.forEach(ch => {
                const g = (ch.group || 'عمومی').trim();
                groupCounts[g] = (groupCounts[g] || 0) + 1;
            });

            const groups = Object.keys(groupCounts).filter(g => g && g !== 'undefined' && g !== 'شخصی');
            if (groups.length <= 1) {
                bar.style.display = 'none';
                return;
            }

            bar.innerHTML = '';
            bar.style.display = 'flex';

            const allChip = document.createElement('div');
            allChip.className = 'sub-cat-chip active';
            allChip.innerHTML = '<span>همه</span> <span style="opacity:0.7;font-size:0.68rem;">(' + channels.length + ')</span>';
            allChip.onclick = () => filterBySubCategory('ALL');
            bar.appendChild(allChip);

            groups.sort((a, b) => groupCounts[b] - groupCounts[a]).forEach(grp => {
                const chip = document.createElement('div');
                chip.className = 'sub-cat-chip';
                chip.innerHTML = '<span>' + grp + '</span> <span style="opacity:0.7;font-size:0.68rem;">(' + groupCounts[grp] + ')</span>';
                chip.onclick = () => filterBySubCategory(grp);
                bar.appendChild(chip);
            });
        }

            if (sBox && !sBox.value) {
                sBox.placeholder = '🔍 جستجو در ' + sortedChannels.length + ' شبکه...';
            }


            if (channelCapObserver) { try { channelCapObserver.disconnect(); } catch(e){} }
            probeQueue.length = 0;

                listEl.innerHTML = '<div style="text-align:center;padding:40px;color:var(--text-muted);">شبکه‌ای یافت نشد.</div>';
                return;

            renderedBatchCount = 0;
            appendNextChannelBatch();

                listEl.appendChild(sentinel);

                if (listObserver) listObserver.disconnect();
                listObserver = new IntersectionObserver((entries) => {
                    if (entries[0].isIntersecting) {
                        appendNextChannelBatch();
                    }
                }, { root: listEl, threshold: 0.1 });
                listObserver.observe(sentinel);
            }
        }

        // Robust HTML5 Drag & Drop Handlers (URL-matched reordering)
        function handleDragStart(e) {
            const li = e.target.closest('.channel-item');
            if (!li) return;
            draggedIndex = parseInt(li.dataset.idx);
            li.classList.add('dragging');
            if (e.dataTransfer) {
                e.dataTransfer.effectAllowed = 'move';
                e.dataTransfer.setData('text/plain', String(draggedIndex));

        function handleDragOver(e) {
            e.preventDefault();
            if (e.dataTransfer) e.dataTransfer.dropEffect = 'move';
            const li = e.target.closest('.channel-item');
            if (li) li.classList.add('drag-over');
        }
        function handleDragLeave(e) {
            const li = e.target.closest('.channel-item');
            if (li) li.classList.remove('drag-over');
        }
        function handleDrop(e) {
            e.preventDefault();
            const targetLi = e.target.closest('.channel-item');
            if (!targetLi) return;
            targetLi.classList.remove('drag-over');
            const targetIndex = parseInt(targetLi.dataset.idx);

            if (draggedIndex !== null && !isNaN(draggedIndex) && !isNaN(targetIndex) && draggedIndex !== targetIndex) {
                const sourceChannel = activeRenderList[draggedIndex];
                const targetChannel = activeRenderList[targetIndex];
                if (!sourceChannel || !targetChannel) return;

                const isFavView = document.getElementById('categorySelect')?.value === 'favorites';
                if (isFavView) {
                    const sourceIdx = myFavorites.findIndex(f => f.url === sourceChannel.url);
                    const targetIdx = myFavorites.findIndex(f => f.url === targetChannel.url);
                    if (sourceIdx > -1 && targetIdx > -1) {
                        const itemToMove = myFavorites.splice(sourceIdx, 1)[0];
                        myFavorites.splice(targetIdx, 0, itemToMove);
                        localStorage.setItem('iptv_favorites_v7', JSON.stringify(myFavorites));
                        currentSortMode = 'default';
                        const sSelect = document.getElementById('channelSortSelect');
                        if (sSelect) sSelect.value = 'default';
                        renderChannels(myFavorites);
                        showActionToast('✅ ترتیب کانال‌های مورد علاقه به‌روزرسانی شد');
                    }
                } else {
                    const sourceIdxRaw = rawActiveChannels.findIndex(c => c.url === sourceChannel.url);
                    const targetIdxRaw = rawActiveChannels.findIndex(c => c.url === targetChannel.url);
                    if (sourceIdxRaw > -1 && targetIdxRaw > -1) {
                        const itemToMove = rawActiveChannels.splice(sourceIdxRaw, 1)[0];
                        rawActiveChannels.splice(targetIdxRaw, 0, itemToMove);
                    }

                    if (window.allChannels && Array.isArray(window.allChannels)) {
                        const sIdxAll = window.allChannels.findIndex(c => c.url === sourceChannel.url);
                        const tIdxAll = window.allChannels.findIndex(c => c.url === targetChannel.url);
                        if (sIdxAll > -1 && tIdxAll > -1) {
                            const itemToMove = window.allChannels.splice(sIdxAll, 1)[0];
                            window.allChannels.splice(tIdxAll, 0, itemToMove);
                        }
                    }

                    currentSortMode = 'default';
                    const sSelect = document.getElementById('channelSortSelect');
                    if (sSelect) sSelect.value = 'default';
                    renderChannels(rawActiveChannels);
                    showActionToast('✅ ترتیب کانال با موفقیت جابجا شد');
                }
            }
            draggedIndex = null;
        }
        function handleDragEnd(e) {

                el.classList.remove('drag-over');

        // Dual-Tier Dead Link Cleaner (تست مستقیم کاربر + تست پروکسی ورکر)
        async function startDualTierCleaning() {
            const btn = document.getElementById('startCleanBtn');
            btn.disabled = true;
            btn.innerText = 'در حال اسکن دو مرحله‌ای...';

            const sourceMode = document.getElementById('cleanerSourceSelect').value;
            const listToClean = sourceMode === 'favorites' ? [...myFavorites] : [...fullChannelList];
            
            const total = listToClean.length;
            let alive = 0, dead = 0, done = 0;
            const cleanedList = [];

            document.getElementById('cleanTotal').innerText = total;

            for (let i = 0; i < total; i++) {
                const ch = listToClean[i];
                let isAlive = false;

                // مرحله اول: تست مستقیم از مرورگر کاربر (برای شبکه‌های داخلی ایران)
                try {
                    const directController = new AbortController();
                    const directTimeout = setTimeout(() => directController.abort(), 2000);
                    const res = await fetch(ch.url, { method: 'HEAD', signal: directController.signal, mode: 'no-cors' });
                    clearTimeout(directTimeout);
                    isAlive = true;
                } catch(e) {
                    // مرحله دوم: تست کمکی از طریق ورکر برای عبور از فیلترینگ
                    try {
                        const fd = new FormData();
                        fd.append('url', ch.url);
                        const proxyRes = await fetch('/test-stream', { method: 'POST', body: fd });
                        if (proxyRes.ok) isAlive = true;
                    } catch(err) {
                        isAlive = false;
                    }
                }

                if (isAlive) {
                    alive++;
                    cleanedList.push(ch);
                } else {
                    dead++;
                }

                done++;
                document.getElementById('cleanAlive').innerText = alive;
                document.getElementById('cleanDead').innerText = dead;
                document.getElementById('cleanProgress').style.width = Math.round((done / total) * 100) + '%';
                document.getElementById('cleanStatusText').innerText = \`بررسی شده: \${done} از \${total}\`;
            }

            btn.style.display = 'none';
            document.getElementById('downloadCleanBtn').style.display = 'block';
            window.cleanedM3UData = cleanedList;
        }

        function downloadCleanedM3U() {
            if (!window.cleanedM3UData) return;

            a.download = 'cleaned_playlist.m3u';

        // Keyboard Shortcuts & Smart TV Remote Control Handler
        function setupKeyboardShortcuts() {
            window.addEventListener('keydown', (e) => {
                const activeTag = (document.activeElement?.tagName || '').toLowerCase();
                if (['input', 'textarea', 'select'].includes(activeTag)) return;
                const v = document.getElementById('player');
                if (!v) return;

                // Smart TV Remote Dedicated Keys
                if (e.key === 'ChannelUp' || e.code === 'ChannelUp') {
                    e.preventDefault();
                    playNextChannel();
                    return;
                }
                if (e.key === 'ChannelDown' || e.code === 'ChannelDown') {
                    e.preventDefault();
                    playPrevChannel();
                    return;
                }
                if (e.key === 'MediaPlayPause' || e.code === 'MediaPlayPause') {
                    e.preventDefault();
                    v.paused ? v.play().catch(()=>{}) : v.pause();
                    return;
                }
                if (e.key === 'MediaPlay' || e.code === 'MediaPlay') {
                    e.preventDefault();
                    v.play().catch(()=>{});
                    return;
                }
                if (e.key === 'MediaPause' || e.code === 'MediaPause') {
                    e.preventDefault();
                    v.pause();
                    return;
                }
                // Tizen / webOS TV Back Key
                if (e.keyCode === 10009 || e.keyCode === 461) {
                    e.preventDefault();
                    closeCleanerModal();
                    closeHotkeysModal();
                    closeMobileToolsSheet();
                    closeLoggerModal();
                    if (document.fullscreenElement) document.exitFullscreen().catch(()=>{});
                    return;
                }

                switch(e.code) {
                    case 'Space':
                    case 'KeyK':
                        e.preventDefault();
                        v.paused ? v.play().catch(() => {}) : v.pause();
                        break;

                    case 'ArrowUp':
                        e.preventDefault();
                        setVolumeDelta(0.05);
                        break;

                    case 'ArrowDown':
                        e.preventDefault();
                        setVolumeDelta(-0.05);
                        break;

                    case 'KeyM':
                        e.preventDefault();
                        v.muted = !v.muted;
                        if (window.player) {
                            try { window.player.muted = v.muted; } catch(err) {}
                        }
                        showVolumeToast(v.volume, v.muted);
                        break;

                    case 'KeyF':
                        e.preventDefault();
                        if (window.player && window.player.fullscreen) {
                            window.player.fullscreen.toggle();
                        } else if (document.fullscreenElement) {
                            document.exitFullscreen().catch(() => {});
                        } else {
                            document.getElementById('videoWrapper')?.requestFullscreen().catch(() => {});
                        }
                        break;

                    case 'KeyN':
                    case 'BracketRight':
                    case 'PageDown':
                        e.preventDefault();
                        playNextChannel();
                        break;

                    case 'KeyP':
                    case 'BracketLeft':
                    case 'PageUp':
                        e.preventDefault();
                        playPrevChannel();
                        break;

                    case 'KeyR':
                    case 'Backspace':
                        e.preventDefault();
                        playLastChannel();
                        break;

                    case 'KeyI':
                        e.preventDefault();
                        togglePiP();
                        break;

                    case 'KeyA':
                        e.preventDefault();
                        cycleAudioTrack();
                        break;

                    case 'KeyC':
                    case 'KeyV':
                        e.preventDefault();
                        toggleSubtitleTrack();
                        break;

                    case 'KeyT':
                        e.preventDefault();
                        toggleSubtitleTranslation();
                        break;

                    case 'KeyS':
                        e.preventDefault();
                        if (document.fullscreenElement) {
                            toggleDropdown('hudSubPopup');
                        } else {
                            toggleDropdown('subPopup');
                        }
                        break;

                    case 'KeyQ':
                        e.preventDefault();
                        if (e.shiftKey) {
                            cycleQualityLevel();
                        } else {
                            if (document.fullscreenElement) {
                                toggleDropdown('hudQualityPopup');
                            } else {
                                toggleDropdown('qualityPopup');
                            }
                        }
                        break;

                    case 'KeyL':
                        e.preventDefault();
                        openLoggerModal();
                        break;

                    case 'KeyH':
                    case 'Slash':
                        if (e.key === '?' || e.code === 'KeyH') {
                            e.preventDefault();
                            openHotkeysModal();
                        }
                        break;

                    case 'Home':
                        e.preventDefault();
                        if (v.duration && isFinite(v.duration)) {
                            v.currentTime = v.duration;
                            showActionToast('⚡ همگام با پخش زنده');
                        }
                        break;

                    case 'Escape':
                        closeHotkeysModal();
                        closeCleanerModal();
                        closeLoggerModal();
                        document.querySelectorAll('.dropdown-popup').forEach(p => p.classList.remove('open'));
                        break;
                }
            });
        }

function isPrivateOrLocalHost(hostname) {
    if (!hostname) return true;
    const host = hostname.toLowerCase().trim();
    if (host === 'localhost' || host.endsWith('.localhost') || host.endsWith('.local') || host.endsWith('.internal')) return true;
    if (host === '0.0.0.0' || host === '127.0.0.1' || host === '::1') return true;

    // IPv4 private/reserved ranges:
    if (/^127\.\d{1,3}\.\d{1,3}\.\d{1,3}$/.test(host)) return true;
    if (/^10\.\d{1,3}\.\d{1,3}\.\d{1,3}$/.test(host)) return true;
    if (/^172\.(1[6-9]|2[0-9]|3[0-1])\.\d{1,3}\.\d{1,3}$/.test(host)) return true;
    if (/^192\.168\.\d{1,3}\.\d{1,3}$/.test(host)) return true;
    if (/^169\.254\.\d{1,3}\.\d{1,3}$/.test(host)) return true;
    if (/^100\.(6[4-9]|[7-9][0-9]|1[0-1][0-9]|12[0-7])\.\d{1,3}\.\d{1,3}$/.test(host)) return true;
    if (/^198\.1[89]\.\d{1,3}\.\d{1,3}$/.test(host)) return true;
    if (/^224\.\d{1,3}\.\d{1,3}\.\d{1,3}$/.test(host)) return true;

    // IPv6 private/unique local
    if (/^(fc|fd|fe[89ab])/i.test(host)) return true;
    return false;
}

function validateTargetUrl(urlStr) {
    if (!urlStr || typeof urlStr !== 'string' || urlStr.length > 8192) return null;
    try {
        const u = new URL(urlStr);
        if (u.protocol !== 'http:' && u.protocol !== 'https:') return null;
        if (u.username || u.password) return null;
        if (isPrivateOrLocalHost(u.hostname)) return null;
        return u;
    } catch(e) {
        return null;
    }
}

async function safeFetchWithRedirects(initialUrlStr, fetchOptions = {}, maxRedirects = 3) {
    let currentUrl = initialUrlStr;
    let redirects = 0;
    while (redirects <= maxRedirects) {
        const validated = validateTargetUrl(currentUrl);
        if (!validated) {
            throw new Error("Invalid or blocked target URL (SSRF Guard): " + currentUrl);
        }
        const opts = { ...fetchOptions, redirect: 'manual' };
        const res = await fetch(currentUrl, opts);
        if ([301, 302, 303, 307, 308].includes(res.status)) {
            const loc = res.headers.get("location");
            if (!loc) return res;
            currentUrl = new URL(loc, currentUrl).toString();
            redirects++;
            continue;
        }
        return res;
    }
    throw new Error("Exceeded max redirects");
}

                    }
                };

                // Inject appropriate Referer & Origin (only for domains that require it like Telewebion)
                const safeHeaders = new Headers();
                safeHeaders.set("User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/124.0.0.0 Safari/537.36");
                safeHeaders.set("Accept", "*/*");
                safeHeaders.set("Accept-Language", "en-US,en;q=0.9,fa;q=0.8");
                safeHeaders.set("Sec-Fetch-Mode", "cors");
                safeHeaders.set("Sec-Fetch-Dest", "empty");
                
                if (targetStreamUrl.hostname.includes("telewebion")) {
                    safeHeaders.set("Referer", "https://telewebion.ir/");
                    safeHeaders.set("Origin", "https://telewebion.ir");
                }
                // Note: Generic video CDNs (e.g. Pluto TV / Akamai) block requests with Origin headers with 403 Forbidden!

                const clientRange = request.headers.get("Range");
                if (clientRange) safeHeaders.set("Range", clientRange);

                const isM3u = (targetStreamUrl.pathname.endsWith(".m3u8") || targetStreamUrl.pathname.endsWith(".m3u"));
                let currentUrl = targetStreamUrl.toString();
                let upstreamRes = null;
                finalUpstreamUrl = currentUrl;

                // Follow up to 5 HTTP redirects manually to reliably track the true CDN destination URL
                for (let r = 0; r < 5; r++) {
                    const fetchRes = await fetch(currentUrl, {
                        headers: safeHeaders,
                        redirect: "manual",
                        cf: {
                            cacheTtl: isM3u ? 0 : (hasParentQuery ? 0 : 60),
                            cacheEverything: !isM3u && !hasParentQuery
                        }
                    });

                    if ([301, 302, 303, 307, 308].includes(fetchRes.status)) {
                        const loc = fetchRes.headers.get("Location");
                        if (loc) {
                            currentUrl = new URL(loc, currentUrl).toString();
                            finalUpstreamUrl = currentUrl;
                            continue;
                        }
                    }
                    upstreamRes = fetchRes;
                    finalUpstreamUrl = fetchRes.url || currentUrl;
                    break;
                }

                if (!upstreamRes) {
                    upstreamRes = await fetch(currentUrl, { headers: safeHeaders });
                    finalUpstreamUrl = currentUrl;
                }

                const proxyStatus = upstreamRes.status;
                if (!upstreamRes.ok && proxyStatus !== 206) {
                    return new Response(`Upstream Error: ${proxyStatus}`, { status: proxyStatus, headers: corsHeaders });
                }

                let contentType = upstreamRes.headers.get("content-type") || "application/octet-stream";
                let responseBody = upstreamRes.body;

                if (targetStreamUrl.pathname.endsWith(".vtt") || contentType.includes("text/vtt")) {
                    contentType = "text/vtt; charset=utf-8";
                }

                const responseHeaders = new Headers(corsHeaders);
                responseHeaders.set("Content-Type", contentType);

                // Rewrite M3U8 Master and Chunk Playlists (Prompt.txt #72-#78)
                if (contentType.includes("mpegurl") || isM3u) {
                    let text = await upstreamRes.text();
                    finalUpstreamUrl = finalUpstreamUrl || upstreamRes.url || targetStreamUrl.toString();
                    const finalBase = finalUpstreamUrl.substring(0, finalUpstreamUrl.lastIndexOf("/") + 1);

                    // Fix malformed lines (e.g. Telewebion's missing '#' before EXT-X-)
                    text = text.replace(/^EXT-X-/gm, "#EXT-X-");

                    const finalUrlObj = new URL(finalUpstreamUrl);
                    // 1. Rewrite media segment URLs and propagate security tokens
                    text = text.replace(/^([^#][^\r\n]*)/gm, (line) => {
                        const trimmed = line.trim();
                        if (!trimmed) return line;
                        const absoluteResolved = mergeChildQuery(new URL(trimmed, finalBase).toString(), finalUrlObj);
                        return `${requestUrl.origin}/proxy-stream?url=${encodeURIComponent(absoluteResolved)}`;
                    });

                    // 2. Rewrite all URI="..." attributes (EXT-X-KEY, EXT-X-MAP, EXT-X-MEDIA, EXT-X-PART, etc.)
                    text = text.replace(/URI="([^"]+)"/g, (match, p1) => {
                        if (p1.startsWith("data:")) return match;
                        const absoluteResolved = mergeChildQuery(new URL(p1, finalBase).toString(), finalUrlObj);
                        return `URI="${requestUrl.origin}/proxy-stream?url=${encodeURIComponent(absoluteResolved)}"`;
                    });

                    responseBody = text;
                    responseHeaders.set("Content-Type", "application/vnd.apple.mpegurl");
                    responseHeaders.set("Cache-Control", hasParentQuery ? "no-store" : "no-cache, no-store, must-revalidate");
                } else {
                    // Video Segments (.ts, .m4s, .mp4) - Cacheable & Range support (Prompt.txt #77-#78)
                    if (hasParentQuery) {
                        responseHeaders.set("Cache-Control", "no-store");
                    } else {
                        responseHeaders.set("Cache-Control", "public, max-age=60, s-maxage=60");
                    }
                    if (upstreamRes.headers.has("Content-Range")) {
                        responseHeaders.set("Content-Range", upstreamRes.headers.get("Content-Range"));
                    }
                    if (upstreamRes.headers.has("Accept-Ranges")) {
                        responseHeaders.set("Accept-Ranges", upstreamRes.headers.get("Accept-Ranges"));
                    }
                    if (upstreamRes.headers.has("Content-Length")) {
                        responseHeaders.set("Content-Length", upstreamRes.headers.get("Content-Length"));
                    }
                }

                return new Response(responseBody, {
                    status: proxyStatus,
                    headers: responseHeaders
                });
            }
            return new Response("Not Found", { status: 404, headers: corsHeaders });

        } catch(criticalErr) {
            return new Response(`Worker Error: ${criticalErr.message}`, { status: 500, headers: corsHeaders });
        }
    }
};
