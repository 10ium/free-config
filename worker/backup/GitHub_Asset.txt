export default {
  async fetch(request, env) {
    const url = new URL(request.url);
    if (url.pathname.startsWith('/api/releases/')) {
      return this.handleApiRequest(url, env);
    }
    if (url.pathname.startsWith('/download/')) {
      return this.handleDownloadProxy(url);
    }
    if (url.pathname === '/') {
      return this.serveHtmlUi();
    }
    return new Response('Not Found', { status: 404 });
  },

  async handleDownloadProxy(url) {
    const targetUrl = url.pathname.substring('/download/'.length);
    const fullTargetUrl = `${targetUrl}${url.search}`;
    try {
      new URL(fullTargetUrl);
    } catch (e) {
      return new Response('Invalid URL for download proxy.', { status: 400 });
    }
    const response = await fetch(fullTargetUrl);
    const newHeaders = new Headers(response.headers);
    newHeaders.set('Access-Control-Allow-Origin', '*');
    return new Response(response.body, {
      status: response.status,
      statusText: response.statusText,
      headers: newHeaders
    });
  },
  
  async handleApiRequest(url, env) {
    const parts = url.pathname.split('/');
    const contentType = { headers: { 'Content-Type': 'application/json' } };
    if (parts.length < 5) { return new Response(JSON.stringify({ error: 'Invalid API request' }), { ...contentType, status: 400 }); }
    const user = parts[3];
    const repo = parts[4];
    const includePrerelease = url.searchParams.get('beta') === 'true';
    const apiUrl = includePrerelease ? `https://api.github.com/repos/${user}/${repo}/releases?per_page=1` : `https://api.github.com/repos/${user}/${repo}/releases/latest`;
    const headers = { 'User-Agent': 'Cloudflare-Worker-GitHub-Tool', 'Accept': 'application/vnd.github.v3+json' };
    if (env.GITHUB_TOKEN) { headers.Authorization = `token ${env.GITHUB_TOKEN}`; }
    try {
      const response = await fetch(apiUrl, { headers });
      if (!response.ok) { return new Response(JSON.stringify({ error: `Releases not found. Status: ${response.status}` }), { ...contentType, status: response.status }); }
      let data = await response.json();
      if (Array.isArray(data)) { data = data[0]; }
      if (!data) { return new Response(JSON.stringify({ error: 'No releases found.' }), { ...contentType, status: 404 }); }
      return new Response(JSON.stringify(data), contentType);
    } catch (error) { return new Response(JSON.stringify({ error: error.message }), { ...contentType, status: 500 }); }
  },

            const checksumMap = new Map(checksumAssets.map(asset => [asset.name.replace('.sha256', ''), asset]));

            const bestAsset = findBestAsset(primaryAssets, clientInfo);
            const sortedAssets = sortAssets(primaryAssets, clientInfo, bestAsset);
            
            if (bestAsset && repoInfo) {
                localStorage.setItem(\`pref-\${repoInfo.user}-\${repoInfo.repo}\`, bestAsset.browser_download_url);
            }
            const bodyHtml = releaseData.body ? marked.parse(releaseData.body) : '<p>بدون توضیحات.</p>';
            let assetsHtml = '';
            let lastOsGroup = null;
            const osDisplayNames = { android: 'اندروید', windows: 'ویندوز', macos: 'مک', linux: 'لینوکس', ios: 'iOS', bsd: 'BSD', other: 'سایر فایل‌ها' };
            
            sortedAssets.forEach(asset => {
                const currentOsGroup = getAssetOs(asset);
                if (currentOsGroup !== lastOsGroup) {
                    assetsHtml += \`<li class="os-separator">\${osDisplayNames[currentOsGroup] || 'سایر'}</li>\`;
                    lastOsGroup = currentOsGroup;
                }
                const isPref = bestAsset && asset.id === bestAsset.id;
                const size = (asset.size / 1048576).toFixed(2) + ' MB';
                const checksumAsset = checksumMap.get(asset.name);
                checksumMap.delete(asset.name); // Mark as used

                assetsHtml += \`<li class="asset-item \${isPref ? 'preferred' : ''}">
                    <div class="asset-info">
                        <div><span class="pref-badge">پیشنهادی</span><span class="asset-name">\${asset.name}</span></div>
                        <span class="asset-meta">📅 \${new Date(asset.updated_at).toLocaleDateString('fa-IR')} | 💾 \${size}</span>
                    </div>
                    <div class="asset-actions">
                        <a href="\${asset.browser_download_url}" class="act-btn dl-btn">دانلود</a>
                        \${checksumAsset ? \`<a href="\${checksumAsset.browser_download_url}" class="act-btn checksum-btn" title="دانلود checksum">SHA256</a>\` : ''}
                    </div>
                </li>\`;
            });

            if (checksumMap.size > 0) {
                assetsHtml += \`<li class="os-separator">فایل‌های Checksum باقی‌مانده</li>\`;
                checksumMap.forEach(asset => {
                    const size = (asset.size / 1024).toFixed(2) + ' KB';
                     assetsHtml += \`<li class="asset-item"><div class="asset-info"><span class="asset-name">\${asset.name}</span><span class="asset-meta">💾 \${size}</span></div><div class="asset-actions"><a href="\${asset.browser_download_url}" class="act-btn checksum-btn">دانلود</a></div></li>\`;
                });
            }

            return \`<div class="rel-sec"><h3>📦 \${releaseData.name || releaseData.tag_name}</h3><div class="rel-body">\${bodyHtml}</div><h4>فایل‌ها:</h4><ul class="asset-list">\${assetsHtml}</ul></div>\`;
        }

        function getAssetOs(asset) {
            const name = asset.name.toLowerCase();
            for (const os in OS_Matchers) {
                const matcher = OS_Matchers[os];
                if (matcher.names.some(n => name.includes(n)) || (matcher.exts.length > 0 && matcher.exts.some(e => name.endsWith(e)))) {
                    return os;
                }
            }
            return 'other';
        }

        function sortAssets(assets, clientInfo, bestAsset) {
            const defaultOsPriority = ['android', 'windows', 'macos', 'linux', 'ios', 'bsd', 'other'];
            return [...assets].sort((a, b) => {
                if (bestAsset) {
                    if (a.id === bestAsset.id) return -1;
                    if (b.id === bestAsset.id) return 1;
                }
                const osA = getAssetOs(a);
                const osB = getAssetOs(b);
                const aIsMatch = osA === clientInfo.os;
                const bIsMatch = osB === clientInfo.os;
                if (aIsMatch && !bIsMatch) return -1;
                if (!aIsMatch && bIsMatch) return 1;
                const priorityA = defaultOsPriority.indexOf(osA);
                const priorityB = defaultOsPriority.indexOf(osB);
                if (priorityA !== priorityB) return priorityA - priorityB;
                return a.name.localeCompare(b.name);
            });
        }
        
        function findBestAsset(assets, clientInfo) {
            const scores = assets.map(asset => ({ asset, score: calculateAssetScore(asset, clientInfo) }));
            if (scores.length === 0) {
                return null;
            }
            const best = scores.reduce((a, b) => a.score > b.score ? a : b);
            return best.score > 0 ? best.asset : null;
        }
        
        function calculateAssetScore(asset, clientInfo) {
            const name = asset.name.toLowerCase();
            const unwanted = ['.sha256', '.asc', '.dSYM.zip', '.plist', '.mobileprovision'];
            if (unwanted.some(term => name.endsWith(term))) {
                return -1;
            }
            let score = 0;
            const archMatchers = {
                x64: ['x64', 'amd64', 'x86_64'], arm64: ['arm64', 'aarch64'], arm64e: ['arm64e'],
                'arm64-v8a': ['arm64-v8a'], 'armeabi-v7a': ['armeabi-v7a', 'armv7'],
                x86: ['x86', '386', 'i686'], universal: ['universal', 'universal2'],
                ppc64le: ['ppc64le'], s390x: ['s390x'], riscv64: ['riscv64'], wasm: ['wasm'],
            };
            const assetOs = getAssetOs(asset);
            const osMatches = assetOs === clientInfo.os;
            if (osMatches) {
                score += 10;
            }
            if (osMatches && clientInfo.os === 'android' && name.includes('universal')) {
                score += 50;
            }
            const selectedArchKeywords = archMatchers[clientInfo.arch];
            const archMatches = selectedArchKeywords && selectedArchKeywords.some(keyword => name.includes(keyword));
            if (archMatches) {
                score += 5;
            }
            if (osMatches && archMatches) {
                score += 2;
            }
            return score;
        }

        function createOutputGroup(label, value, showDl) {
            const dlBtn = showDl ? \`<button class="act-btn dl-raw-btn" data-url="\${value}">دانلود</button>\` : '';
            return \`<div style="margin-top:10px;"><label>\${label}</label><div class="output-field"><input type="text" value="\${value}" readonly><button class="act-btn copy-btn">کپی</button><a href="\${value}" target="_blank" class="act-btn open-btn">باز کردن</a>\${dlBtn}</div></div>\`;
        }

        async function downloadRawContent(button, url) {
            const originalText = button.textContent;
            button.textContent = 'در حال دریافت...';
            button.disabled = true;
            try {
                const response = await fetch('/download/' + url);
                if (!response.ok) {
                    throw new Error(\`HTTP error! status: \${response.status}\`);
                }
                const blob = await response.blob();
                const blobUrl = URL.createObjectURL(blob);
                const link = document.createElement('a');
                link.href = blobUrl;
                link.download = url.substring(url.lastIndexOf('/') + 1) || 'download';
                document.body.appendChild(link);
                link.click();
                document.body.removeChild(link);
                URL.revokeObjectURL(blobUrl);
            } catch (error) {
                console.error('Download failed:', error);
                alert('دانلود با خطا مواجه شد. لطفاً از دکمه "باز کردن" استفاده کنید.');
            } finally {
                button.textContent = originalText;
                button.disabled = false;
            }
        }

        function setupActionButtons(card) {
            card.addEventListener('click', function(event) {
                const target = event.target;
                if (target.classList.contains('copy-btn')) {
                    const input = target.parentElement.querySelector('input');
                    if (input) {
                        navigator.clipboard.writeText(input.value);
                        target.textContent = '✔';
                        target.classList.add('copied');
                        setTimeout(() => {
                            target.textContent = 'کپی';
                            target.classList.remove('copied');
                        }, 2000);
                    }
                } else if (target.classList.contains('dl-raw-btn')) {
                    const url = target.dataset.url;
                    if (url) {
                        downloadRawContent(target, url);
                    }
                }
            });
        }
        
        function getClientInfo() {
            const selectedOs = DOM.osSelect.value;
            let selectedArch = DOM.archSelect.value;
            let os = selectedOs;
            let arch = selectedArch;
            if (os === 'auto') {
                const ua = navigator.userAgent.toLowerCase();
                if (/win/.test(ua)) { os = 'windows'; }
                else if (/mac|darwin/.test(ua)) { os = 'macos'; }
                else if (/android/.test(ua)) { os = 'android'; }
                else if (/linux/.test(ua)) { os = 'linux'; }
                else { os = 'unknown'; }
            }
            if (arch === 'auto') {
                const ua = navigator.userAgent.toLowerCase();
                if (/arm64|aarch64/.test(ua)) { arch = 'arm64-v8a'; }
                else if (/x86_64|x64|amd64/.test(ua)) { arch = 'x64'; }
                else if (/x86(?!_64)|i686|i386/.test(ua)) { arch = 'x86'; }
                else { arch = 'unknown'; }
            }
            return { os, arch };
        }
        
        function extractRepoInfo(url) {
            try {
                const match = url.match(/github(?:usercontent)?\\.com\\/([^/]+)\\/([^/]+)/);
                return match ? { user: match[1], repo: match[2].replace('.git', '') } : null;
            } catch (e) {
                return null;
            }
        }

        function convertGithubLink(url) {
            try {
                const u = new URL(url);
                const parts = u.pathname.split('/').filter(Boolean);
                if (u.hostname === 'github.com' && parts[2] === 'blob') {
                    const [user, repo, , branch, ...path] = parts;
                    return { output1: \`https://github.com/\${user}/\${repo}/raw/refs/heads/\${branch}/\${path.join('/')}\`, output2: \`https://github.com/\${user}\` };
                }
                if (u.hostname === 'github.com' && parts[2] === 'raw' && parts[3] === 'refs') {
                    const [user, repo, ,,,, branch, ...path] = parts;
                    return { output1: \`https://github.com/\${user}/\${repo}/blob/\${branch}/\${path.join('/')}\`, output2: \`https://github.com/\${user}\` };
                }
                if (u.hostname === 'raw.githubusercontent.com') {
                    const [user, repo, branch, ...path] = parts;
                    return { output1: \`https://github.com/\${user}/\${repo}/blob/\${branch}/\${path.join('/')}\`, output2: \`https://github.com/\${user}\` };
                }
            } catch (e) {
                /* Ignore */
            }
            return null;
        }
        
        initializeUI();
    </script>
</body>
</html>`;
    return new Response(html, { headers: { 'Content-Type': 'text/html;charset=UTF-8' } });
  }
