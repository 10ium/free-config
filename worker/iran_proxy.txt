// A list of sources to fetch proxies from.
const PROXY_SOURCES = [
  { url: 'https://raw.githubusercontent.com/proxifly/free-proxy-list/refs/heads/main/proxies/countries/IR/data.txt', type: 'text' },
  { url: 'https://raw.githubusercontent.com/iplocate/free-proxy-list/refs/heads/main/countries/IR/proxies.txt', type: 'text' },
  { url: 'https://raw.githubusercontent.com/sunny9577/proxy-scraper/refs/heads/master/generated/http_proxies.json', type: 'json_sunny', protocol: 'http' },
  { url: 'https://raw.githubusercontent.com/sunny9577/proxy-scraper/refs/heads/master/generated/socks4_proxies.json', type: 'json_sunny', protocol: 'socks4' },
  { url: 'https://raw.githubusercontent.com/sunny9577/proxy-scraper/refs/heads/master/generated/socks5_proxies.json', type: 'json_sunny', protocol: 'socks5' },
  { url: 'https://raw.githubusercontent.com/themiralay/Proxy-List-World/refs/heads/master/data-with-geolocation.json', type: 'json_themiralay' },
  { url: 'https://raw.githubusercontent.com/vakhov/fresh-proxy-list/refs/heads/master/proxylist.json', type: 'json_vakhov' },
];

addEventListener('fetch', event => {
  event.respondWith(handleRequest(event.request));
});

async function handleRequest(request) {
  if (request.method !== 'GET') {
    return new Response('متد غیرمجاز. لطفاً فقط از متد GET استفاده کنید.', { status: 405 });
  }

  const { pathname } = new URL(request.url);
  
  try {
    const { uniqueProxies, report } = await fetchAndProcessProxies();

    if (uniqueProxies.length === 0 && pathname !== '/') {
      return new Response('متاسفانه هیچ پروکسی قابل استفاده‌ای از منابع تعریف‌شده یافت نشد.', { status: 503 });
    }

    switch (pathname) {
      case '/xray':
        return new Response(uniqueProxies.join('\n'), {
          headers: { 'Content-Type': 'text/plain; charset=utf-8' },
        });

      case '/clash':
        return new Response(formatForClash(uniqueProxies), {
          headers: { 'Content-Type': 'text/yaml; charset=utf-8' },
        });

      case '/':
        return new Response(createLandingPage(report, uniqueProxies), {
          headers: { 'Content-Type': 'text/html; charset=utf-8' },
        });

      default:
        return new Response('مسیر یافت نشد.', { status: 404 });
    }
  } catch (error) {
    console.error('خطای داخلی در ورکر:', error);
    return new Response('یک خطای داخلی در سرور رخ داد.', { status: 500 });
  }
}

async function fetchAndProcessProxies() {
  const sourceReports = [];
  const promises = PROXY_SOURCES.map(source =>
    fetch(source.url)
      .then(async response => {
        if (!response.ok) throw new Error(`دریافت ناموفق با کد وضعیت: ${response.status}`);
        if (source.type === 'json_vakhov') {
          const text = await response.text();
          if (!text || text.trim() === '') return [];
          const correctedJsonText = '[' + text.replace(/}\s*{/g, '},{') + ']';
          try { return JSON.parse(correctedJsonText); } 
          catch (e) { throw new Error('قالب JSON اصلاح شده نامعتبر است.'); }
        }
        return source.type.startsWith('json') ? response.json() : response.text();
      })
      .then(data => {
        const { proxies, stats } = parseProxies(data, source);
        sourceReports.push({ منبع: source.url.split('/').slice(-3).join('/'), وضعیت: 'موفق', تعداد_دریافتی: proxies.length, تفکیک_پروتکل: stats.protocols, خطا: null });
        return proxies;
      })
      .catch(error => {
        console.warn(`[لاگ حرفه‌ای] خطا در پردازش منبع ${source.url}: ${error.message}`);
        sourceReports.push({ منبع: source.url.split('/').slice(-3).join('/'), وضعیت: 'ناموفق', تعداد_دریافتی: 0, تفکیک_پروتکل: {}, خطا: error.message });
        return [];
      })
  );

  const proxyGroups = await Promise.all(promises);
  const allProxies = proxyGroups.flat();
  const totalFetched = allProxies.length;
  const uniqueProxies = [...new Set(allProxies)];
  const duplicatesRemoved = totalFetched - uniqueProxies.length;
  const report = {
    خلاصه_گزارش: { 'تعداد کل دریافتی': totalFetched, 'تعداد تکراری حذف شده': duplicatesRemoved, 'تعداد نهایی یکتا': uniqueProxies.length },
    جزئیات_منابع: sourceReports,
  };
  return { uniqueProxies, report };
}

function parseProxies(data, source) {
  let proxies = [];
  const protocolStats = {};
  const countProtocol = (proxyStr) => {
    const protocol = (proxyStr.split('://')[0] || 'unknown').toLowerCase();
    protocolStats[protocol] = (protocolStats[protocol] || 0) + 1;
  };

  if (source.type.startsWith('json') && !Array.isArray(data)) { return { proxies: [], stats: { protocols: {} } }; }

  switch (source.type) {
    case 'text':
      proxies = data.split('\n').map(p => p.trim()).filter(Boolean);
      break;
    case 'json_sunny':
      proxies = data.filter(p => p.country && p.country.startsWith('Iran')).map(p => `${source.protocol}://${p.ip}:${p.port}`);
      break;
    case 'json_themiralay':
      proxies = data.filter(p => p.geolocation && p.geolocation.countryCode === 'IR').map(p => `http://${p.ip}:${p.port}`);
      break;
    case 'json_vakhov':
      proxies = data.filter(p => p.country_code === 'IR').map(p => {
        if (p.socks5 === '1') return `socks5://${p.ip}:${p.port}`;
        if (p.socks4 === '1') return `socks4://${p.ip}:${p.port}`;
        if (p.ssl === '1') return `https://${p.ip}:${p.port}`;
        if (p.http === '1') return `http://${p.ip}:${p.port}`;
        return null;
      }).filter(Boolean);
      break;
  }
  proxies.forEach(countProtocol);
  return { proxies, stats: { protocols: protocolStats } };
}

function formatForClash(proxies) {
  let clashProxies = 'proxies:\n';
  proxies.forEach((proxyStr) => {
    const match = proxyStr.match(/^(socks5|socks4|https|http):\/\/([^:]+):(\d+)$/);
    if (!match) return;
    
    let [, protocol, server, port] = match;
    let type = protocol;
    let additionalFields = '';

    if (protocol === 'https') {
      type = 'http';
      additionalFields += '    tls: true\n';
      additionalFields += '    skip-cert-verify: true\n';
    } else if (protocol === 'socks5') {
      additionalFields += '    udp: true\n';
    }

    // --- FIX: Exclude 'socks4' as it is not a valid type in Clash configuration. ---
    const supportedTypes = ['http', 'socks5'];
    if (!supportedTypes.includes(type)) return;
    // --- END OF FIX ---
    
    const name = `IR - ${protocol.toUpperCase()} - ${server}:${port}`;
    clashProxies += `  - name: "${name}"\n`;
    clashProxies += `    type: ${type}\n`;
    clashProxies += `    server: ${server}\n`;
    clashProxies += `    port: ${port}\n`;
    clashProxies += additionalFields;
  });
  return clashProxies;
}

function createLandingPage(report, proxies) {

    const protocols = Object.entries(item.تفکیک_پروتکل).map(([p, c]) => `${p}: ${c}`).join(', ') || 'N/A';
    return `<tr><td>${item.منبع}</td><td class="${statusClass}">${item.وضعیت}</td><td>${item.تعداد_دریافتی}</td><td>${protocols}</td><td>${item.خطا || '—'}</td></tr>`;
  }).join('');
  
  const proxyListText = proxies.join('\n');

  return `

  <head>

      <title>گزارش و لیست پروکسی‌های ایران</title>
      <style>
          body { font-family: -apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, "Helvetica Neue", Arial, sans-serif; line-height: 1.6; padding: 20px; max-width: 900px; margin: auto; background-color: #f4f4f9; color: #333; }

          table { width: 100%; border-collapse: collapse; margin-top: 20px; table-layout: fixed; }
          th, td { padding: 12px; border: 1px solid #ddd; text-align: right; word-wrap: break-word; }
          th { background-color: #007bff; color: white; }
          .status-success { color: #28a745; font-weight: bold; }
          .status-failure { color: #dc3545; font-weight: bold; }
          .proxy-list-container { position: relative; }
          #proxy-list { width: 100%; min-height: 200px; font-family: "Courier New", Courier, monospace; direction: ltr; text-align: left; padding: 10px; border-radius: 5px; border: 1px solid #ced4da; background-color: #e9ecef; box-sizing: border-box; }
