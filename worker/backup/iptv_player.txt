    <script src="https://cdn.plyr.io/3.7.8/plyr.polyfilled.js"></script>
    <script>
        let fullChannelList = [];
        let filteredChannelList = [];
        let renderedCount = 0;
        const CHUNK_SIZE = 50; 
        let player, hls, itemObserver, infiniteScrollObserver;
        let myFavorites = [];
        let myCustomLists = []; 
        let draggedItemIndex = null;

        document.addEventListener('DOMContentLoaded', () => {
            try {
                const savedFavs = localStorage.getItem('iptv_favorites_v7');
                if (savedFavs) myFavorites = JSON.parse(savedFavs);
                
                const savedLists = localStorage.getItem('iptv_custom_lists_v2');
                if (savedLists) myCustomLists = JSON.parse(savedLists);
            } catch(e) {}

            renderCustomListOptions();

            const video = document.getElementById('player');
            
            // 🔥 ADDED SETTINGS GEAR ICON: Allows playback speed adjustment & quality options if available 🔥
            player = new Plyr(video, {
                controls: ['play-large', 'play', 'progress', 'current-time', 'mute', 'volume', 'settings', 'pip', 'airplay', 'fullscreen'],
                settings: ['quality', 'speed'],
                i18n: { speed: 'سرعت پخش', quality: 'کیفیت', normal: 'عادی' }
            });

            itemObserver = new IntersectionObserver((entries, observer) => {
                entries.forEach(entry => {
                    if (entry.isIntersecting) {
                        const li = entry.target;
                        const img = li.querySelector('img');
                        if (img && img.dataset.src) {
                            img.src = img.dataset.src;
                            img.removeAttribute('data-src');
                            img.onerror = () => { img.style.opacity = '0'; };
                        }
                        observer.unobserve(li);
                    }
                });
            }, { root: document.getElementById('channelList'), rootMargin: '100px' });

            infiniteScrollObserver = new IntersectionObserver((entries) => {
                if (entries[0].isIntersecting) { loadMoreChannels(); }
            }, { root: document.getElementById('channelList'), rootMargin: '300px' });

            loadCategory("https://iptv-org.github.io/iptv/languages/fas.m3u");
        });

        function copyStreamUrl() {
            const input = document.getElementById('streamUrlInput');
            input.select();
            input.setSelectionRange(0, 99999); 
            try {
                navigator.clipboard.writeText(input.value).then(() => {
                    const btn = document.getElementById('copyBtn');
                    btn.innerText = "✅ کپی شد";
                    btn.style.background = "#10b981";
                    setTimeout(() => {
                        btn.innerText = "📋 کپی لینک مستقیم";
                        btn.style.background = "var(--primary)";
                    }, 2000);
                });
            } catch(err) {
                document.execCommand("copy"); 
            }
        }

        function resetInfoPanel() {
            document.getElementById('streamInfoPanel').style.display = 'none';
            document.getElementById('vpnNotice').style.display = 'none';
            document.getElementById('mediaFeaturesBox').style.display = 'none';
            document.getElementById('audioFeatures').style.display = 'none';
            document.getElementById('subtitleFeatures').style.display = 'none';
            document.getElementById('audioFeatures').innerHTML = '';
            document.getElementById('subtitleFeatures').innerHTML = '';
        }

        function toggleFavorite(event, url, name, logo, group) {
            event.stopPropagation(); 
            const existingIndex = myFavorites.findIndex(f => f.url === url);
            if (existingIndex > -1) {
                myFavorites.splice(existingIndex, 1);
                event.target.classList.remove('active');
            } else {
                myFavorites.push({ url, name, logo, group });
                event.target.classList.add('active');
            }
            localStorage.setItem('iptv_favorites_v7', JSON.stringify(myFavorites));
            
            if (document.getElementById('categorySelect').value === 'favorites') {
                loadCategory('favorites');
            }
        }

        function generateId() { return Math.random().toString(36).substring(2, 9); }

        function renderCustomListOptions() {
            const optGroup = document.getElementById('customListsGroup');
            optGroup.innerHTML = '';
            
            if (myCustomLists.length > 0) {
                optGroup.style.display = 'block';
                myCustomLists.forEach((list) => {
                    const opt = document.createElement('option');
                    opt.value = "custom:" + list.id;
                    const icon = list.type === 'single' ? '▶️' : '📁';
                    opt.text = \`\${icon} \${list.name}\`;
                    optGroup.appendChild(opt);
                });
            } else {
                optGroup.style.display = 'none';
            }
        }

        function handleCategoryChange() {
            const val = document.getElementById('categorySelect').value;
            const delBtn = document.getElementById('deleteListBtn');
            
            // Delete button ONLY appears for custom lists
            if (val.startsWith('custom:')) {
                delBtn.style.display = 'block';
                loadCategory(val);
            } else {
                delBtn.style.display = 'none';
                loadCategory(val);
            }
        }

        function deleteCustomList() {
            const val = document.getElementById('categorySelect').value;
            if (val.startsWith('custom:')) {
                const listId = val.split('custom:')[1];
                if(confirm("آیا از حذف این لیست شخصی مطمئن هستید؟")) {
                    myCustomLists = myCustomLists.filter(list => list.id !== listId);
                    localStorage.setItem('iptv_custom_lists_v2', JSON.stringify(myCustomLists));
                    renderCustomListOptions();
                    document.getElementById('deleteListBtn').style.display = 'none';
                    document.getElementById('categorySelect').value = "favorites";
                    loadCategory('favorites');
                }
            }
        }

        function handleCustomInputUrl() {
            const inputUrl = document.getElementById('customUrlInput').value.trim();
            if (!inputUrl) return;

            let isSingleChannel = inputUrl.includes('.m3u8') && !inputUrl.includes('.m3u?');
            let listName = prompt(
                isSingleChannel ? "لطفاً نامی برای این کانال شخصی وارد کنید:" : "لطفاً نامی برای این لیست پخش وارد کنید:", 
                isSingleChannel ? "کانال من" : "لیست شخصی من"
            );

            if (!listName) listName = isSingleChannel ? "کانال بدون نام" : "لیست بدون نام";

            const newId = generateId();
            myCustomLists.push({ 
                id: newId, 
                name: listName, 
                type: isSingleChannel ? 'single' : 'url', 
                content: inputUrl 
            });
            
            localStorage.setItem('iptv_custom_lists_v2', JSON.stringify(myCustomLists));
            renderCustomListOptions();

            document.getElementById('customUrlInput').value = "";
            document.getElementById('categorySelect').value = "custom:" + newId;
            handleCategoryChange();
        }

        function handleFileUpload(event) {
            const file = event.target.files[0];
            if (!file) return;

            let listName = prompt("لطفاً نامی برای فایل لیست پخش وارد کنید:", file.name);
            if (!listName) listName = file.name;

            const reader = new FileReader();
            reader.onload = function(e) {
                const textContent = e.target.result;
                const newId = generateId();
                
                myCustomLists.push({ 
                    id: newId, 
                    name: listName, 
                    type: 'file', 
                    content: textContent 
                });
                
                localStorage.setItem('iptv_custom_lists_v2', JSON.stringify(myCustomLists));
                renderCustomListOptions();
                
                document.getElementById('categorySelect').value = "custom:" + newId;
                handleCategoryChange();
            };
            reader.readAsText(file);
            event.target.value = ""; 
        }

        async function loadCategory(val) {
            const listEl = document.getElementById('channelList');
            listEl.innerHTML = \`<div class="loading-text">در حال بارگیری...<br><small>لطفاً صبر کنید</small></div>\`;
            document.getElementById('channelCount').innerText = "...";

            if (val === 'favorites') {
                fullChannelList = [...myFavorites];
                filteredChannelList = [...fullChannelList];
                initRender();
                return;
            }
            
            if (val.startsWith('custom:')) {
                const listId = val.split('custom:')[1];
                const customItem = myCustomLists.find(l => l.id === listId);
                
                if (customItem) {
                    if (customItem.type === 'single') {
                        fullChannelList = [{ name: customItem.name, url: customItem.content, logo: '', group: 'شخصی' }];
                        filteredChannelList = [...fullChannelList];
                        initRender();
                        setTimeout(()=> {
                            const firstEl = document.querySelector('.channel-item');
                            if(firstEl) startPlaybackLogic(customItem.content, customItem.name, firstEl);
                        }, 500);
                        return;
                    } 
                    else if (customItem.type === 'file') {
                        fullChannelList = parseM3U(customItem.content);
                        filteredChannelList = [...fullChannelList];
                        initRender();
                        return;
                    }
                    else if (customItem.type === 'url') {
                        val = customItem.content; 
                    }
                }
            }

            try {
                const response = await fetch('/proxy-m3u?url=' + encodeURIComponent(val));
                if (!response.ok) throw new Error("Network Error");
                const text = await response.text();
                fullChannelList = parseM3U(text);
                
                const term = document.getElementById('searchInput').value.trim().toLowerCase();
                if (term.length > 0) {
                    filteredChannelList = fullChannelList.filter(c => c.name.toLowerCase().includes(term) || c.group.toLowerCase().includes(term));
                } else {
                    filteredChannelList = [...fullChannelList];
                }
                
                initRender();
            } catch (error) {
                listEl.innerHTML = \`<div class="loading-text" style="color:#ef4444">خطا در دریافت لیست. لطفاً از فعال بودن لینک مطمئن شوید.</div>\`;
            }
        }

        function parseM3U(m3uText) {
            const lines = m3uText.split(/\\r?\\n/);
            const parsedChannels = [];
            let currentChannel = null;

            if (!m3uText.includes('#EXTM3U') && !m3uText.includes('#EXTINF')) return [];

            for (let i = 0; i < lines.length; i++) {
                const line = lines[i].trim();
                if (line.startsWith('#EXTINF:')) {
                    const logoMatch = line.match(/tvg-logo="([^"]+)"/);
                    const groupMatch = line.match(/group-title="([^"]+)"/) || line.match(/#EXTGRP:(.*)/); 
                    
                    let groupName = 'عمومی';
                    if (groupMatch) groupName = groupMatch[1];
                    
                    if(i + 1 < lines.length && lines[i+1].startsWith('#EXTGRP:')) {
                        groupName = lines[i+1].replace('#EXTGRP:', '').trim();
                    }

                    const namePart = line.split(',').pop();
                    
                    let logoUrl = logoMatch && logoMatch[1] ? logoMatch[1] : '';
                    if(logoUrl) logoUrl = \`https://wsrv.nl/?url=\${encodeURIComponent(logoUrl)}&w=64&h=64&fit=contain\`;

                    currentChannel = {
                        name: namePart ? namePart.trim() : 'شبکه ناشناس',
                        logo: logoUrl,
                        group: groupName,
                        url: ''
                    };
                } else if (line.startsWith('http') && currentChannel) {
                    currentChannel.url = line;
                    parsedChannels.push(currentChannel);
                    currentChannel = null;
                }
            }
            return parsedChannels;
        }

        function handleDragStart(e) {
            if (document.getElementById('searchInput').value.trim() !== '') {
                e.preventDefault();
                return;
            }
            draggedItemIndex = parseInt(e.currentTarget.dataset.absIndex);
            e.currentTarget.classList.add('dragging');
            e.dataTransfer.effectAllowed = 'move';
        }

        function handleDragOver(e) {
            e.preventDefault();
            e.currentTarget.classList.add('drag-over');
        }

        function handleDragLeave(e) {
            e.currentTarget.classList.remove('drag-over');
        }

        function handleDrop(e) {
            e.preventDefault();
            const targetElement = e.currentTarget;
            targetElement.classList.remove('drag-over');
            
            const targetIndex = parseInt(targetElement.dataset.absIndex);
            if (draggedItemIndex === null || draggedItemIndex === targetIndex) return;

            const itemToMove = filteredChannelList.splice(draggedItemIndex, 1)[0];
            filteredChannelList.splice(targetIndex, 0, itemToMove);

            if (document.getElementById('categorySelect').value === 'favorites') {
                myFavorites = [...filteredChannelList];
                localStorage.setItem('iptv_favorites_v7', JSON.stringify(myFavorites));
            }

            const ul = document.getElementById('channelList');
            const scrollPos = ul.scrollTop;
            initRender();
            setTimeout(() => { ul.scrollTop = scrollPos; }, 50);
        }

        function handleDragEnd(e) {
            e.currentTarget.classList.remove('dragging');
            document.querySelectorAll('.channel-item').forEach(el => el.classList.remove('drag-over'));
            draggedItemIndex = null;
        }

        function initRender() {
            const ul = document.getElementById('channelList');
            if (infiniteScrollObserver) infiniteScrollObserver.disconnect();
            
            ul.innerHTML = '';
            renderedCount = 0;

            const sentinel = document.createElement('div');
            sentinel.id = 'scroll-sentinel';
            sentinel.className = 'loading-more';
            ul.appendChild(sentinel);
            infiniteScrollObserver.observe(sentinel);

            if (filteredChannelList.length === 0) {
                sentinel.innerText = "لیست خالی است یا موردی یافت نشد.";
            } else {
                loadMoreChannels();
            }
        }

        function loadMoreChannels() {
            if (renderedCount >= filteredChannelList.length) {
                const sentinel = document.getElementById('scroll-sentinel');
                if(sentinel) sentinel.innerText = "پایان لیست.";
                return;
            }

            const ul = document.getElementById('channelList');
            const sentinel = document.getElementById('scroll-sentinel');
            const nextBatch = filteredChannelList.slice(renderedCount, renderedCount + CHUNK_SIZE);
            
            nextBatch.forEach((channel, i) => {
                const absoluteIndex = renderedCount + i;
                const isFav = myFavorites.some(f => f.url === channel.url);
                
                const li = document.createElement('li');
                li.className = 'channel-item';
                li.dataset.streamUrl = channel.url; 
                li.dataset.absIndex = absoluteIndex;
                li.draggable = true;
                
                li.innerHTML = \`
                    <span class="drag-handle" title="جابجایی (Drag & Drop)">↕</span>
                    <img data-src="\${channel.logo}" class="channel-logo">
                    <div class="channel-info">
                        <div class="channel-name-row">
                            <div class="channel-name">\${channel.name}</div>
                        </div>
                        <div class="channel-category">\${channel.group}</div>
                    </div>
                    <span class="favorite-btn \${isFav ? 'active' : ''}" title="نشان کردن" 
                          onclick="toggleFavorite(event, '\${channel.url}', '\${channel.name}', '\${channel.logo}', '\${channel.group}')">★</span>
                \`;
                
                li.addEventListener('dragstart', handleDragStart);
                li.addEventListener('dragover', handleDragOver);
                li.addEventListener('dragleave', handleDragLeave);
                li.addEventListener('drop', handleDrop);
                li.addEventListener('dragend', handleDragEnd);

                li.onclick = (e) => {
                    if(e.target.classList.contains('drag-handle') || e.target.classList.contains('favorite-btn')) return;
                    startPlaybackLogic(channel.url, channel.name, li);
                };
                
                ul.insertBefore(li, sentinel);
                itemObserver.observe(li);
            });

            renderedCount += nextBatch.length;
            document.getElementById('channelCount').innerText = \`نمایش: \${renderedCount} از \${filteredChannelList.length}\`;
        }

        document.getElementById('searchInput').addEventListener('input', (e) => {
            const term = e.target.value.toLowerCase();
            filteredChannelList = term.length === 0 
                ? [...fullChannelList] 
                : fullChannelList.filter(c => c.name.toLowerCase().includes(term) || c.group.toLowerCase().includes(term));
            initRender();
        });

        function translateError(responseCode) {
            if (!responseCode) return "خطای شبکه / عدم دسترسی مرورگر";
            if (responseCode === 403) return "مسدود شده ⛔ (تحریم/انقضای لینک)";
            if (responseCode === 404) return "حذف شده ❌ (لینک از کار افتاده)";
            if (responseCode >= 500 && responseCode <= 504) return "سرور مبدا قطع است ⚠️";
            if (responseCode > 504) return "خطای کلودفلر/ارتباطی مبدا ⚠️";
            return \`خطای \${responseCode}\`;
        }

        function startPlaybackLogic(originalUrl, name, element) {
            if(element) {
                document.querySelectorAll('.channel-item').forEach(el => el.classList.remove('active'));
                element.classList.add('active');
            }
            
            resetInfoPanel();
            
            let mode = document.getElementById('playbackMode').value;
            const isHttpsSite = window.location.protocol === 'https:';
            const isHttpStream = originalUrl.startsWith('http://');

            if (isHttpsSite && isHttpStream && (mode === 'direct-proxy' || mode === 'direct-only')) {
                mode = (mode === 'direct-only') ? 'proxy-only' : 'proxy-direct';
                document.getElementById('nowPlayingText').innerHTML = \`هشدار امنیتی: <span class="status-badge badge-warning">HTTP مسدود شد</span> تغییر اجباری به پروکسی...\`;
            }

            let attempt1Proxy = false, attempt2Proxy = null;
            if (mode === 'direct-proxy') { attempt1Proxy = false; attempt2Proxy = true; }
            else if (mode === 'proxy-direct') { attempt1Proxy = true; attempt2Proxy = false; }
            else if (mode === 'proxy-only') { attempt1Proxy = true; attempt2Proxy = null; }
            else if (mode === 'direct-only') { attempt1Proxy = false; attempt2Proxy = null; }

            executeHlsLoad(originalUrl, name, attempt1Proxy, attempt2Proxy);
        }

        function executeHlsLoad(originalUrl, name, useProxy, fallbackToProxyState) {
            const statusLabel = document.getElementById('nowPlayingText');
            const video = document.getElementById('player');
            
            const targetUrl = useProxy ? ('/proxy-stream?url=' + encodeURIComponent(originalUrl)) : originalUrl;
            
            document.getElementById('streamUrlInput').value = originalUrl;
            document.getElementById('streamInfoPanel').style.display = 'flex';
            
            if (useProxy) {
                document.getElementById('vpnNotice').style.display = 'block';
            } else {
                document.getElementById('vpnNotice').style.display = 'none';
            }

            statusLabel.innerHTML = \`در حال اتصال: \${name} <span class="status-badge \${useProxy ? 'badge-proxy' : 'badge-direct'}">\${useProxy ? 'پروکسی' : 'مستقیم'}</span>\`;

            const handleStreamError = (errorReason) => {
                if (fallbackToProxyState !== null) {
                    statusLabel.innerHTML = \`خطا (\${errorReason}). <span class="status-badge badge-warning">تلاش مسیر دوم...</span>\`;
                    executeHlsLoad(originalUrl, name, fallbackToProxyState, null);
                } else {
                    statusLabel.innerHTML = \`خطا در پخش: \${name} <br><span class="status-badge badge-error" style="margin-top:5px; font-size:0.85rem">\${errorReason}</span>\`;
                }
            };

            if (Hls.isSupported()) {
                if (hls) hls.destroy();
                hls = new Hls({ maxBufferLength: 30 });
                
                hls.loadSource(targetUrl);
                hls.attachMedia(video);
                
                hls.on(Hls.Events.MANIFEST_PARSED, () => { 
                    video.play().catch(e => console.log("Autoplay blocked")); 
                    statusLabel.innerHTML = \`در حال پخش: \${name} <span class="status-badge \${useProxy ? 'badge-proxy' : 'badge-direct'}">\${useProxy ? 'پروکسی' : 'مستقیم'}</span>\`;
                    
                    // Populate Quality Levels dynamically
                    if (hls.levels.length > 1) {
                        const availableQualities = hls.levels.map((l) => l.height);
                        availableQualities.unshift(0); 
                        player.config.quality = {
                            default: 0,
                            options: availableQualities,
                            forced: true,
                            onChange: (e) => {
                                if (e === 0) {
                                    hls.currentLevel = -1; // Auto
                                } else {
                                    hls.levels.forEach((level, levelIndex) => {
                                        if (level.height === e) hls.currentLevel = levelIndex;
                                    });
                                }
                            }
                        };
                    }
                });

                hls.on(Hls.Events.AUDIO_TRACKS_UPDATED, function (event, data) {
                    if (data.audioTracks && data.audioTracks.length > 1) {
                        const languages = data.audioTracks.map((t, i) => t.name || t.lang || \`زبان \${i+1}\`).join('، ');
                        const audioBox = document.getElementById('audioFeatures');
                        audioBox.innerHTML = \`🎧 <b>صداهای یافت شده:</b> \${languages}\`;
                        audioBox.style.display = 'block';
                        document.getElementById('mediaFeaturesBox').style.display = 'flex';
                    }
                });

                hls.on(Hls.Events.SUBTITLE_TRACKS_UPDATED, function (event, data) {
                    if (data.subtitleTracks && data.subtitleTracks.length > 0) {
                        const languages = data.subtitleTracks.map((t, i) => t.name || t.lang || \`زیرنویس \${i+1}\`).join('، ');
                        const subBox = document.getElementById('subtitleFeatures');
                        subBox.innerHTML = \`💬 <b>زیرنویس‌های یافت شده:</b> \${languages}\`;
                        subBox.style.display = 'block';
                        document.getElementById('mediaFeaturesBox').style.display = 'flex';
                    }
                });

                hls.on(Hls.Events.ERROR, function (event, data) {
                    if (data.fatal) {
                        let reason = "قطعی شبکه";
                        if (data.response && data.response.code) {
                            reason = translateError(data.response.code);
                        } else if (data.type === Hls.ErrorTypes.NETWORK_ERROR) {
                            reason = "محدودیت مرورگر (CORS) یا قطعی نت";
                        }
                        handleStreamError(reason);
                    }
                });

            } else if (video.canPlayType('application/vnd.apple.mpegurl')) {
                video.src = targetUrl;
                video.addEventListener('error', () => handleStreamError("خطای پخش‌کننده داخلی گوشی"));
                video.addEventListener('loadedmetadata', () => { 
                    video.play().catch(e=>e); 
                    statusLabel.innerHTML = \`در حال پخش: \${name} <span class="status-badge \${useProxy ? 'badge-proxy' : 'badge-direct'}">\${useProxy ? 'پروکسی' : 'مستقیم'}</span>\`;
                });
            }
        }

    </script>
</body>
</html>
`;

export default {
  async fetch(request) {
    const corsHeaders = {
        "Access-Control-Allow-Origin": "*",
        "Access-Control-Allow-Methods": "GET, HEAD, POST, OPTIONS",
        "Access-Control-Allow-Headers": "Content-Type, Range, User-Agent, X-Requested-With, Origin, Accept",
        "Access-Control-Expose-Headers": "Content-Length, Content-Range, Content-Type"
    };

    if (request.method === "OPTIONS") return new Response(null, { headers: corsHeaders });

    try {
      const requestUrl = new URL(request.url);
      const path = requestUrl.pathname;

      if (path === "/" || path === "") {
        return new Response(HTML_UI, { 
            headers: { 
                "Content-Type": "text/html; charset=utf-8",
                "Cache-Control": "no-store, no-cache, must-revalidate, proxy-revalidate",
                "Pragma": "no-cache",
                "Expires": "0"
            } 
        });
      }

      if (path === "/proxy-m3u") {
        const targetUrl = requestUrl.searchParams.get("url");
        if (!targetUrl || !targetUrl.startsWith("http")) return new Response("Invalid URL", { status: 400, headers: corsHeaders });
        const m3uRes = await fetch(targetUrl, { headers: { "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36" } });
        return new Response(m3uRes.body, { headers: { ...corsHeaders, "Content-Type": "text/plain", "Cache-Control": "public, max-age=3600" }});
      }

      if (path === "/proxy-stream") {
        const targetUrlString = requestUrl.searchParams.get("url");
        if (!targetUrlString) return new Response("Missing url", { status: 400, headers: corsHeaders });

        const targetUrl = new URL(targetUrlString);
        const queryString = targetUrl.search || "";

        const safeHeaders = new Headers();
        safeHeaders.set("User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/114.0.0.0 Safari/537.36");
        safeHeaders.set("Accept", "*/*");
        safeHeaders.set("Referer", targetUrl.origin + "/");
        safeHeaders.set("Origin", targetUrl.origin);
        
        const clientRange = request.headers.get("Range");
        if (clientRange) safeHeaders.set("Range", clientRange);

        const upstreamRes = await fetch(targetUrl.toString(), {
          headers: safeHeaders,
          redirect: "follow"
        });

        const proxyStatus = upstreamRes.status;

        if (!upstreamRes.ok && proxyStatus !== 206) {
            return new Response(`Upstream Error: ${proxyStatus}`, { status: proxyStatus, headers: corsHeaders });
        }

        let contentType = upstreamRes.headers.get("content-type") || "application/octet-stream";
        let body = upstreamRes.body;

        if (targetUrl.pathname.endsWith('.vtt') || contentType.includes('text/vtt')) {
            contentType = "text/vtt; charset=utf-8";
        }

        if (contentType.includes("mpegurl") || targetUrl.pathname.endsWith(".m3u8") || targetUrl.pathname.endsWith(".m3u")) {
            let text = await upstreamRes.text();
            const finalUrl = upstreamRes.url; 
            const finalBase = finalUrl.substring(0, finalUrl.lastIndexOf("/") + 1);

            text = text.replace(/^([^#][^\r\n]*)/gm, (line) => {
                if (!line.trim()) return line;
                const absoluteUpstreamUrl = new URL(line, finalBase);
                if (queryString && !absoluteUpstreamUrl.search) {
                    absoluteUpstreamUrl.search = queryString;
                }
                return `${requestUrl.origin}/proxy-stream?url=${encodeURIComponent(absoluteUpstreamUrl.toString())}`;
            });
            
            text = text.replace(/URI="([^"]+)"/g, (match, p1) => {
                if(p1.startsWith('data:')) return match;
                const absoluteUrl = new URL(p1, finalBase);
                if (queryString && !absoluteUrl.search && !p1.includes('key')) {
                    absoluteUrl.search = queryString;
                }
                return `URI="${requestUrl.origin}/proxy-stream?url=${encodeURIComponent(absoluteUrl.toString())}"`;
            });
            
            body = text;
        }

        return new Response(body, {
          status: proxyStatus,
          headers: { ...corsHeaders, "Content-Type": contentType }
        });
      }

      return new Response("Not found", { status: 404, headers: corsHeaders });
    } catch (err) {
      return new Response("Worker Error", { status: 500 });
    }
  }
}
