export default {
  async fetch(request, env, ctx) {
    // 0. هندل کردن پیش‌درخواست‌های CORS (OPTIONS Preflight)
    if (request.method === 'OPTIONS') {
      return new Response(null, {
        status: 204,
        headers: {
          'Access-Control-Allow-Origin': '*',
          'Access-Control-Allow-Methods': 'GET, POST, HEAD, OPTIONS',
          'Access-Control-Allow-Headers': request.headers.get('Access-Control-Request-Headers') || '*',
          'Access-Control-Max-Age': '86400'
        }
      });
    }

    const url = new URL(request.url);
    const path = url.pathname;

    // 1. فاوآیکون اختصاصی
    if (path === '/favicon.ico') {
      return this.serveFavicon();
    }

    // 2. صفحه اصلی و رابط کاربری
    if (path === '/' || path === '') {
      return this.serveHtmlUi(request);
    }

    // 3. API دریافت اطلاعات سرور و موقعیت دیتاسنتر کلودفلر
    if (path === '/api/info') {
      return this.handleInfoApi(request);
    }

    // 4. API ریلیزهای گیت‌هاب
    if (path.startsWith('/api/releases/')) {
      return this.handleApiRequest(url, request, env);
    }

    // 5. API بررسی مشخصات فایل (متد HEAD)
    if (path === '/api/inspect') {
      return this.handleInspection(url);
    }

    // 6. پروکسی دانلود استاندارد
    if (path.startsWith('/download/')) {
      return this.handleDownloadProxy(request, ctx);
    }

    // 7. روتینگ مستقیم، تمیز و گیت (پشتیبانی از git clone، گیست، گیت‌لب، هاگینگ‌فیس و دامنه‌ها)
    const cleanPath = path.replace(/^\/+/, '');
    if (cleanPath.length > 0) {
      return this.handleDirectMirror(request, ctx, cleanPath);
    }

    return new Response('Not Found', { status: 404 });
  },

  handleInfoApi(request) {
    const cf = request.cf || {};
    const info = {
      colo: cf.colo || 'EDGE',
      country: cf.country || 'Global',
      city: cf.city || 'Global',
      ip: request.headers.get('CF-Connecting-IP') || ''
    };
    return new Response(JSON.stringify(info), {
      headers: {
        'Content-Type': 'application/json',
        'Access-Control-Allow-Origin': '*'
      }
    });
  },

  async handleDownloadProxy(request, ctx) {
    const url = new URL(request.url);
    let targetUrlStr = url.pathname.substring('/download/'.length);

    if (!targetUrlStr) return new Response('Invalid URL', { status: 400 });

    if (url.search) targetUrlStr += url.search;
    if (!targetUrlStr.startsWith('http')) targetUrlStr = 'https://' + targetUrlStr;

    const isClashMode = url.searchParams.get('clash') === '1' || url.searchParams.get('process') === 'clash';
    return this.proxyFetch(targetUrlStr, request, ctx, isClashMode, url.origin);
  },

  async handleDirectMirror(request, ctx, cleanPath) {
    const url = new URL(request.url);
    const isClashMode = url.searchParams.get('clash') === '1' || url.searchParams.get('process') === 'clash';
    const parts = cleanPath.split('/');

    // حالت ۱: پشتیبانی از دستورات git clone با آدرس کامل (مانند https://... یا http://...)
    if (cleanPath.startsWith('http://') || cleanPath.startsWith('https://')) {
      let target = cleanPath;
      if (url.search) target += url.search;
      return this.proxyFetch(target, request, ctx, isClashMode, url.origin);
    }

    // حالت ۲: دامنه‌های مستقل (گیت‌هاب، گیت‌لب، هاگینگ‌فیس، گیست و غیره)
    if (parts[0].includes('.')) {
      let target = 'https://' + cleanPath;
      if (parts[0] === 'gitlab.com' && target.includes('/-/blob/')) {
        target = target.replace('/-/blob/', '/-/raw/');
      }
      if (parts[0] === 'huggingface.co' && target.includes('/blob/')) {
        target = target.replace('/blob/', '/resolve/');
      }
      if (url.search) target += url.search;
      return this.proxyFetch(target, request, ctx, isClashMode, url.origin);
    }

    // حالت ۳: مسیرهای مستقیم کوتاه گیت‌هاب (مانند owner/repo/branch/file.yaml)
    if (parts.length >= 3) {
      const [owner, repo, ...rest] = parts;
      let restOfPath = rest.join('/');
      restOfPath = restOfPath.replace(/^refs\/heads\//, '');
      let target = 'https://raw.githubusercontent.com/' + owner + '/' + repo + '/' + restOfPath;
      if (url.search) target += url.search;
      return this.proxyFetch(target, request, ctx, isClashMode, url.origin, {
        isGitHubRaw: true,
        owner,
        repo,
        branch: rest[0],
        filePath: rest.slice(1).join('/')
      });
    }

    return new Response('Not Found', { status: 404 });
  },

  async proxyFetch(targetUrlStr, request, ctx, isClashMode, workerOrigin, githubMeta = null) {
    try {
      const targetUrlObj = new URL(decodeURIComponent(targetUrlStr));
      targetUrlObj.searchParams.delete('clash');
      targetUrlObj.searchParams.delete('process');
      const normalizedTarget = targetUrlObj.toString();

      const cache = caches.default;
      const cacheKey = new Request(normalizedTarget, request);

      // کش بومی کلودفلر (برای متد GET معمولی غیر کلش)
      if (!isClashMode && request.method === 'GET') {
        let cachedRes = await cache.match(cacheKey);
        if (cachedRes) {
          let newHeaders = new Headers(cachedRes.headers);
          newHeaders.set('X-Worker-Cache', 'HIT');
          return new Response(cachedRes.body, { status: cachedRes.status, headers: newHeaders });
        }
      }

      // تنظیم هدرهای درخواست بالادستی
      const upstreamHeaders = new Headers(request.headers);
      upstreamHeaders.set('User-Agent', 'Mozilla/5.0 (MirrorPRO; Enterprise Gateway)');
      upstreamHeaders.set('Host', targetUrlObj.host);
      upstreamHeaders.delete('CF-Connecting-IP');
      upstreamHeaders.delete('CF-Ray');
      upstreamHeaders.delete('CF-IPCountry');

      const reqInit = {
        method: request.method,
        headers: upstreamHeaders,
        redirect: 'follow'
      };

      // پشتیبانی از بادی در متدهای POST (مانند git-upload-pack در git clone)
      if (request.method !== 'GET' && request.method !== 'HEAD') {
        reqInit.body = request.body;
        reqInit.duplex = 'half';
      }

      let upstreamRes = await fetch(normalizedTarget, reqInit);

      // سیستم هوشمند جایگزین jsDelivr (Fallback خودکار)
      // اگر فایل خام گیت‌هاب در دسترس نبود یا خطای شبکه داد، خودکار از jsDelivr واکشی می‌شود
      if (!upstreamRes.ok && githubMeta && githubMeta.isGitHubRaw && githubMeta.filePath) {
        try {
          const jsdelivrUrl = `https://cdn.jsdelivr.net/gh/${githubMeta.owner}/${githubMeta.repo}@${githubMeta.branch}/${githubMeta.filePath}`;
          const fallbackRes = await fetch(jsdelivrUrl, {
            headers: { 'User-Agent': 'Mozilla/5.0 (MirrorPRO; Enterprise Gateway)' }
          });
          if (fallbackRes.ok) {
            upstreamRes = fallbackRes;
          }
        } catch (e) {
          // در صورت بروز خطا از همان پاسخ اصلی استفاده می‌شود
        }
      }

      if (!upstreamRes.ok && upstreamRes.status !== 304 && upstreamRes.status !== 206) {
        return upstreamRes;
      }

      let resHeaders = new Headers(upstreamRes.headers);
      resHeaders.set('Access-Control-Allow-Origin', '*');
      resHeaders.set('Access-Control-Allow-Methods', 'GET, POST, HEAD, OPTIONS');
      resHeaders.set('X-Worker-Cache', 'MISS');

      // پردازش هوشمند فایل‌های کانفیگ کلش (YAML / TXT)
      const isConfigExt = (normalizedTarget.includes('.yaml') || normalizedTarget.includes('.yml') || normalizedTarget.includes('.txt') || normalizedTarget.includes('target=clash'));
      if (isClashMode && isConfigExt && request.method === 'GET') {
        const textContent = await upstreamRes.text();
        const processedText = this.processClashYaml(textContent, workerOrigin);

        resHeaders.set('Content-Type', 'text/plain; charset=utf-8');
        return new Response(processedText, { status: 200, headers: resHeaders });
      }

      // کش یک ساله برای فایل‌های باینری و استاتیک موفق در کلودفلر
      if (upstreamRes.status === 200 && request.method === 'GET' && !isClashMode) {
        resHeaders.set('Cache-Control', 'public, max-age=31536000');
      }

      const response = new Response(upstreamRes.body, {
        status: upstreamRes.status,
        statusText: upstreamRes.statusText,
        headers: resHeaders
      });

      if (upstreamRes.status === 200 && !isClashMode && request.method === 'GET') {
        ctx.waitUntil(cache.put(cacheKey, response.clone()));
      }

      return response;
    } catch (e) {
      return new Response('Error proxying: ' + e.message, { status: 500 });
    }
  },

  processClashYaml(yamlText, workerOrigin) {
    const mirrorableExtensions = ['.yaml', '.yml', '.txt', '.json', '.srs'];
    let inProviderBlock = false;
    let healthCheckIndent = -1;
    let processNextLineAsUrl = false;
    const lines = yamlText.split('\n');

    const shouldSkipUrl = (u) => {
      return (
        u.includes(workerOrigin) ||
        u.includes('gstatic.com') ||
        u.includes('cloudflare.com') ||
        u.includes('v2fly.org') ||
        u.includes('1.1.1.1') ||
        u.includes('google.com/generate_204') ||
        u.includes('cp.cloudflare.com')
      );
    };

    const makeWorkerUrl = (origUrl) => {
      if (shouldSkipUrl(origUrl)) return origUrl;
      return workerOrigin + '/download/' + encodeURIComponent(origUrl);
    };

    const processedLines = lines.map(line => {
      const indentMatch = line.match(/^(\s*)/);
      const indent = indentMatch ? indentMatch[1].length : 0;
      const trimmedLine = line.trim();

      // تشخیص کلیدهای سطح اصلی
      const isRoot = indent === 0 && trimmedLine.length > 0 && !trimmedLine.startsWith('#');
      if (isRoot) {
        inProviderBlock = /^(rule-providers|proxy-providers)\s*:/i.test(line);
        healthCheckIndent = -1;
      }

      // بررسی ورود یا خروج از بلاک health-check
      if (inProviderBlock) {
        if (/^\s+health-check\s*:/i.test(line)) {
          healthCheckIndent = indent;
        } else if (healthCheckIndent !== -1 && indent <= healthCheckIndent && trimmedLine.length > 0 && !trimmedLine.startsWith('#')) {
          healthCheckIndent = -1;
        }
      }

      // پشتیبانی از ساختار چندخطی YAML (url: >- یا url: |-)
      if (processNextLineAsUrl) {
        processNextLineAsUrl = false;
        const urlMatch = trimmedLine.match(/^(https?:[/]{2}[^"'\s]+)/);
        if (urlMatch && !shouldSkipUrl(urlMatch[0])) {
          return line.replace(urlMatch[0], makeWorkerUrl(urlMatch[0]));
        }
        return line;
      }

      if (trimmedLine.match(/^url:\s*(?:>|\|)-?\s*$/)) {
        processNextLineAsUrl = true;
        return line;
      }

      // اگر داخل پروایدر باشیم و در health-check نباشیم
      if (inProviderBlock && healthCheckIndent === -1) {
        return line.replace(/(https?:\/\/[^\s"'\|]+)/g, (match) => {
          return makeWorkerUrl(match);
        });
      }

      // اگر خارج از پروایدر باشد ولی لینک آشکار رول‌ست یا کانفیگ باشد
      const singleLineUrlRegex = /(:\s*["']?)(https?:[/]{2}[^"'\s]+)(["']?)/;
      const singleLineMatch = trimmedLine.match(singleLineUrlRegex);
      if (singleLineMatch) {
        const foundUrl = singleLineMatch[2];
        const isMirrorable = mirrorableExtensions.some(ext => foundUrl.split('?')[0].endsWith(ext)) || foundUrl.includes('target=clash');
        if (isMirrorable && !shouldSkipUrl(foundUrl)) {
          return line.replace(foundUrl, makeWorkerUrl(foundUrl));
        }
      }

      return line;
    });

    return processedLines.join('\n');
  },

  async handleInspection(url) {
    const target = url.searchParams.get('url');
    if (!target) return new Response(JSON.stringify({ success: false }), { status: 400 });
    try {
      const res = await fetch(decodeURIComponent(target), { method: 'HEAD', headers: { 'User-Agent': 'MirrorPRO' } });
      const size = res.headers.get('content-length') || 0;
      return new Response(JSON.stringify({ success: true, size: size }), { headers: { 'Content-Type': 'application/json' } });
    } catch (e) {
      return new Response(JSON.stringify({ success: false, error: e.message }), { status: 500 });
    }
  },

  async handleApiRequest(url, request, env) {
    const parts = url.pathname.split('/');
    const contentType = { headers: { 'Content-Type': 'application/json' } };
    if (parts.length < 5) return new Response(JSON.stringify({ _apiError: 'مسیر درخواست نامعتبر است.' }), contentType);

    const user = parts[3];
    const repo = parts[4];
    const includePrerelease = url.searchParams.get('beta') === 'true';
    const apiUrl = includePrerelease ? 'https://api.github.com/repos/' + user + '/' + repo + '/releases?per_page=1' : 'https://api.github.com/repos/' + user + '/' + repo + '/releases/latest';

    const headers = { 'User-Agent': 'Cloudflare-Worker-GitHub-Tool', 'Accept': 'application/vnd.github.v3+json' };

    const clientToken = request.headers.get('X-GitHub-Token');
    if (clientToken) {
      headers.Authorization = 'token ' + clientToken;
    } else if (env && env.GITHUB_TOKEN) {
      headers.Authorization = 'token ' + env.GITHUB_TOKEN;
    }

    try {
      const response = await fetch(apiUrl, { headers });

      if (response.status === 403 || response.status === 429) {
        return new Response(JSON.stringify({ _apiError: 'محدودیت درخواست گیت‌هاب (Rate Limit) پر شده است. لطفاً توکن خود را در تنظیمات وارد کنید.' }), contentType);
      }
      if (!response.ok) {
        return new Response(JSON.stringify({ _apiError: 'نسخه ریلیز برای این مخزن یافت نشد.' }), contentType);
      }

      let data = await response.json();
      if (Array.isArray(data)) data = data[0];
      if (!data) return new Response(JSON.stringify({ _apiError: 'هیچ فایلی در ریلیز یافت نشد.' }), contentType);

      return new Response(JSON.stringify(data), contentType);
    } catch (error) {
      return new Response(JSON.stringify({ _apiError: 'خطای سرور: ' + error.message }), contentType);
    }
  },

        function renderMarkdown(md) {
            if (!md) return 'توضیحاتی ثبت نشده است.';
            var lines = md.split(String.fromCharCode(10));
            var result = [];
            var inCodeBlock = false;
            var codeBuffer = [];
            var tripleBacktick = String.fromCharCode(96, 96, 96);
            var singleBacktick = String.fromCharCode(96);

            function parseLinks(t) {
                var oB = t.indexOf('[');
                var cB = t.indexOf('](');
                var cP = t.indexOf(')', cB);
                if (oB !== -1 && cB > oB && cP > cB) {
                    var lText = t.substring(oB + 1, cB);
                    var lUrl = t.substring(cB + 2, cP);
                    var b = t.substring(0, oB);
                    var a = t.substring(cP + 1);
                    return b + '<a href="' + lUrl + '" target="_blank" style="color:var(--primary);">' + lText + '</a>' + parseLinks(a);
                }
                return t;
            }

            function parseBold(t) {
                var p = t.split('**');
                if (p.length > 2) {
                    var out = '';
                    for (var k = 0; k < p.length; k++) {
                        out += (k % 2 === 1) ? ('<strong>' + p[k] + '</strong>') : p[k];
                    }
                    return out;
                }
                return t;
            }

            for (var i = 0; i < lines.length; i++) {
                var line = lines[i];
                if (line.trim().startsWith(tripleBacktick)) {
                    if (inCodeBlock) {
                        result.push('<pre style="background:var(--bg); padding:10px; border-radius:6px; overflow-x:auto;"><code>' + codeBuffer.join('<br>') + '</code></pre>');
                        codeBuffer = [];
                        inCodeBlock = false;
                    } else {
                        inCodeBlock = true;
                    }
                    continue;
                }

                if (inCodeBlock) {
                    codeBuffer.push(line.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;'));
                    continue;
                }

                var safeLine = line.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
                if (safeLine.startsWith('### ')) {
                    result.push('<h4 style="margin:10px 0 5px;">' + safeLine.substring(4) + '</h4>');
                } else if (safeLine.startsWith('## ')) {
                    result.push('<h3 style="margin:12px 0 6px;">' + safeLine.substring(3) + '</h3>');
                } else if (safeLine.startsWith('# ')) {
                    result.push('<h2 style="margin:14px 0 8px;">' + safeLine.substring(2) + '</h2>');
                } else {
                    var parts = safeLine.split(singleBacktick);
                    if (parts.length > 2) {
                        var reconstructed = '';
                        for (var p = 0; p < parts.length; p++) {
                            reconstructed += (p % 2 === 1) ? ('<code>' + parts[p] + '</code>') : parts[p];
                        }
                        safeLine = reconstructed;
                    }
                    safeLine = parseBold(safeLine);
                    safeLine = parseLinks(safeLine);
                    result.push(safeLine + '<br>');
                }
            }

            if (inCodeBlock && codeBuffer.length > 0) {
                result.push('<pre style="background:var(--bg); padding:10px; border-radius:6px; overflow-x:auto;"><code>' + codeBuffer.join('<br>') + '</code></pre>');
            }

            return result.join('');
        }

        function analyzeLink(urlStr) {
            let result = { 
                original: urlStr.trim(), 
                type: 'generic', 
                subType: 'generic',
                profileLink: null, 
                rawLink: null, 
                blobLink: null, 
                repoLink: null,
                releaseWebLink: null,
                cleanWorkerLink: null, 
                gitCloneCmd: null,
                zipArchiveUrl: null,
                repoInfo: null 
            };
            try {
                const url = new URL(urlStr.trim());
                const host = url.hostname.toLowerCase();
                const parts = url.pathname.split('/').filter(Boolean);

                // ۱. گیت‌هاب استاندارد (github.com)
                if (host === 'github.com' || host === 'www.github.com') {
                    result.type = 'github';
                    if (parts.length >= 1) {
                        result.profileLink = 'https://github.com/' + parts[0];
                    }
                    if (parts.length >= 2) {
                        const repoName = parts[1].endsWith('.git') ? parts[1].slice(0, -4) : parts[1];
                        result.repoInfo = { user: parts[0], repo: repoName };
                        result.repoLink = 'https://github.com/' + parts[0] + '/' + repoName;
                        result.gitCloneCmd = 'git clone ' + WORKER_ORIGIN + '/https://github.com/' + parts[0] + '/' + repoName + '.git';
                        result.zipArchiveUrl = WORKER_ORIGIN + '/download/' + encodeURIComponent('https://github.com/' + parts[0] + '/' + repoName + '/archive/refs/heads/main.zip');
                    }
                    
                    if (parts.length >= 4 && parts[2] === 'blob') {
                        result.subType = 'blob';
                        const branch = parts[3];
                        const filePath = parts.slice(4).join('/');
                        result.blobLink = urlStr.trim();
                        result.rawLink = 'https://raw.githubusercontent.com/' + parts[0] + '/' + parts[1] + '/refs/heads/' + branch + '/' + filePath;
                        result.cleanWorkerLink = WORKER_ORIGIN + '/' + parts[0] + '/' + parts[1] + '/' + branch + '/' + filePath;
                    } else if (parts.length >= 4 && parts[2] === 'raw') {
                        result.subType = 'raw';
                        const branch = parts[3];
                        const filePath = parts.slice(4).join('/');
                        result.rawLink = urlStr.trim();
                        result.blobLink = 'https://github.com/' + parts[0] + '/' + parts[1] + '/blob/' + branch + '/' + filePath;
                        result.cleanWorkerLink = WORKER_ORIGIN + '/' + parts[0] + '/' + parts[1] + '/' + branch + '/' + filePath;
                    } else if (parts.length >= 5 && parts[2] === 'releases' && parts[3] === 'download') {

                        result.rawLink = urlStr.trim();
                        result.cleanWorkerLink = WORKER_ORIGIN + '/download/' + encodeURIComponent(urlStr.trim());
                        result.releaseWebLink = 'https://github.com/' + parts[0] + '/' + parts[1] + '/releases/tag/' + parts[4];
                    } else if (parts.length >= 4 && (parts[2] === 'commit' || parts[2] === 'commits')) {
                        result.subType = 'commit';
                        result.rawLink = urlStr.trim();

                    } else if (parts.length === 2 || (parts.length >= 3 && parts[2] === 'releases')) {
                        result.subType = 'repo';
                    } else if (parts.length === 1) {
                        result.subType = 'profile';
                    } else {
                        result.subType = 'repo';

                } 
                // ۲. گیت‌هاب پیجز (*.github.io)

                        result.blobLink = 'https://github.com/' + parts[0] + '/' + parts[1] + '/blob/' + branch + '/' + filePath;

                        result.cleanWorkerLink = WORKER_ORIGIN + '/' + parts[0] + '/' + parts[1] + '/' + branch + '/' + filePath;
                    }
                } 
                // ۴. دامنه‌های مدیا و محتوای کاربر گیت‌هاب (*.githubusercontent.com)
                else if (host.endsWith('.githubusercontent.com')) {
                    if (host === 'gist.githubusercontent.com') {
                        result.type = 'gist';
                        result.subType = 'gist_raw';
                        if (parts.length >= 1) result.profileLink = 'https://github.com/' + parts[0];
                        if (parts.length >= 2) {
                            result.blobLink = 'https://gist.github.com/' + parts[0] + '/' + parts[1];
                        }
                    } else {
                        result.type = 'github_media';
                        result.subType = 'media';
                    }
                    result.rawLink = urlStr.trim();
                    result.cleanWorkerLink = WORKER_ORIGIN + '/' + host + url.pathname;
                }
                // ۵. گیت‌هاب گیست (Gist)
                else if (host === 'gist.github.com') {
                    result.type = 'gist';
                    result.subType = 'gist_web';
                    if (parts.length >= 1) result.profileLink = 'https://github.com/' + parts[0];
                    result.rawLink = urlStr.trim();
                    result.cleanWorkerLink = WORKER_ORIGIN + '/' + host + url.pathname;
                }
                // ۶. گیت‌لب (GitLab)
                else if (host === 'gitlab.com') {
                    result.type = 'gitlab';
                    const dashIdx = parts.indexOf('-');
                    if (dashIdx > 0) {
                        result.repoLink = 'https://gitlab.com/' + parts.slice(0, dashIdx).join('/');
                    } else if (parts.length >= 2) {
                        result.repoLink = 'https://gitlab.com/' + parts[0] + '/' + parts[1];
                    }
                    if (urlStr.includes('/-/blob/')) {
                        result.subType = 'gitlab_blob';
                        result.rawLink = urlStr.trim().replace('/-/blob/', '/-/raw/');
                        result.blobLink = urlStr.trim();
                    } else if (urlStr.includes('/-/raw/')) {
                        result.subType = 'gitlab_raw';
                        result.rawLink = urlStr.trim();
                        result.blobLink = urlStr.trim().replace('/-/raw/', '/-/blob/');
                    } else {
                        result.subType = 'gitlab_repo';
                        result.rawLink = urlStr.trim();
                    }
                    let cleanGitLabPath = url.pathname;
                    if (cleanGitLabPath.includes('/-/blob/')) {
                        cleanGitLabPath = cleanGitLabPath.replace('/-/blob/', '/-/raw/');
                    }
                    result.cleanWorkerLink = WORKER_ORIGIN + '/gitlab.com' + cleanGitLabPath;
                }
                // ۷. هاگینگ فیس (Hugging Face)
                else if (host === 'huggingface.co') {
                    result.type = 'huggingface';
                    const markerIdx = parts.findIndex(p => p === 'blob' || p === 'resolve');
                    if (markerIdx > 0) {
                        result.repoLink = 'https://huggingface.co/' + parts.slice(0, markerIdx).join('/');
                    } else if (parts.length >= 1) {
                        result.repoLink = 'https://huggingface.co/' + parts.slice(0, Math.min(parts.length, 2)).join('/');
                    }
                    if (urlStr.includes('/blob/')) {
                        result.subType = 'hf_blob';
                        result.rawLink = urlStr.trim().replace('/blob/', '/resolve/');
                        result.blobLink = urlStr.trim();
                    } else if (urlStr.includes('/resolve/')) {
                        result.subType = 'hf_resolve';
                        result.rawLink = urlStr.trim();
                        result.blobLink = urlStr.trim().replace('/resolve/', '/blob/');
                    } else {
                        result.subType = 'hf_repo';
                        result.rawLink = urlStr.trim();
                    }
                    let cleanHfPath = url.pathname;
                    if (cleanHfPath.includes('/blob/')) {
                        cleanHfPath = cleanHfPath.replace('/blob/', '/resolve/');
                    }
                    result.cleanWorkerLink = WORKER_ORIGIN + '/huggingface.co' + cleanHfPath;
                }
                // ۸. سایر دامنه‌های اینترنتی
                else {
                    result.type = 'generic';
                    result.subType = 'generic';
                    let cleanPath = urlStr.trim();
                    if (cleanPath.startsWith('https://')) cleanPath = cleanPath.substring(8);
                    else if (cleanPath.startsWith('http://')) cleanPath = cleanPath.substring(7);
                    result.cleanWorkerLink = WORKER_ORIGIN + '/' + cleanPath;
                }
            } catch (e) { }
            return result;
        }

        function getSmartMirrorUrl(conv, targetUrl, isClashMode) {
            let url = conv.cleanWorkerLink || (WORKER_ORIGIN + '/download/' + encodeURIComponent(targetUrl));
            if (isClashMode) {
                url += (url.includes('?') ? '&' : '?') + 'clash=1';
            }
            return url;
        }

        function renderReleaseSection(releaseData, clientInfo, repoInfo) {
            if (!releaseData) return '';
            if (releaseData._apiError) {
                return '<div class="rel-sec"><p style="color:var(--warn);">⚠️ ' + releaseData._apiError + '</p></div>';
            }
            if (!releaseData.tag_name) return '';

            let assetsHtml = '';
            if (releaseData.assets && releaseData.assets.length > 0) {
                const primaryAssets = releaseData.assets.filter(asset => !asset.name.endsWith('.sha256'));
                const checksumAssets = releaseData.assets.filter(asset => asset.name.endsWith('.sha256'));
                const checksumMap = new Map(checksumAssets.map(asset => [asset.name.replace('.sha256', ''), asset]));

                const bestAsset = findBestAsset(primaryAssets, clientInfo);
                const sortedAssets = sortAssets(primaryAssets, clientInfo, bestAsset);
                
                let currentOs = '';
                sortedAssets.forEach(asset => {
                    const os = getAssetOs(asset);
                    if (os !== currentOs) {
                        currentOs = os;
                        assetsHtml += '<li class="os-separator" data-os="' + os + '">سیستم عامل: ' + os.toUpperCase() + '</li>';
                    }

                    const isPreferred = bestAsset && (asset.id === bestAsset.id);
                    const size = (asset.size / (1024 * 1024)).toFixed(2) + ' MB';
                    const proxyUrl = WORKER_ORIGIN + '/download/' + encodeURIComponent(asset.browser_download_url);
                    const checksumAsset = checksumMap.get(asset.name);
                    checksumMap.delete(asset.name);

                    assetsHtml += '<li class="asset-item ' + (isPreferred ? 'preferred' : '') + '" data-name="' + asset.name.toLowerCase() + '" data-os="' + os + '">' +
                        '<div class="asset-info">' +
                            '<span class="asset-name">' + asset.name + '</span>' +
                            '<span class="asset-meta">' +
                                '<span class="pref-badge">پیشنهاد سیستم شما</span>' +
                                '<span>💾 ' + size + '</span>' +
                                '<span>⬇️ ' + asset.download_count.toLocaleString() + '</span>' +
                            '</span>' +
                        '</div>' +
                        '<div class="output-field" style="margin-top:5px; width:100%;">' +
                            '<input type="text" value="' + proxyUrl + '" readonly>' +
                            '<button class="act-btn copy-btn" data-url="' + proxyUrl + '">کپی پروکسی</button>' +
                            '<a href="' + asset.browser_download_url + '" target="_blank" class="act-btn dl-direct-btn">دانلود مستقیم</a>' +
                            '<button class="act-btn force-dl-btn dl-proxy-btn" data-url="' + proxyUrl + '">دانلود با پروکسی</button>' +
                            (checksumAsset ? '<button class="act-btn force-dl-btn checksum-btn" data-url="' + WORKER_ORIGIN + '/download/' + encodeURIComponent(checksumAsset.browser_download_url) + '">دانلود SHA256</button>' : '') +
                        '</div>' +
                    '</li>';
                });

                if (checksumMap.size > 0) {
                    assetsHtml += '<li class="os-separator" data-os="checksum">فایل‌های Checksum باقی‌مانده</li>';
                    checksumMap.forEach(asset => {
                        const size = (asset.size / 1024).toFixed(2) + ' KB';
                        const proxyUrl = WORKER_ORIGIN + '/download/' + encodeURIComponent(asset.browser_download_url);
                        assetsHtml += '<li class="asset-item" data-name="' + asset.name.toLowerCase() + '" data-os="checksum">' +
                            '<div class="asset-info"><span class="asset-name">' + asset.name + '</span><span class="asset-meta">💾 ' + size + '</span></div>' +
                            '<div class="output-field" style="width:100%;"><input type="text" value="' + proxyUrl + '" readonly><button class="act-btn copy-btn" data-url="' + proxyUrl + '">کپی</button><button class="act-btn force-dl-btn dl-proxy-btn" data-url="' + proxyUrl + '">دانلود</button></div>' +
                        '</li>';
                    });
                }
            } else {
                assetsHtml = '<li>فایل کامپایل‌شده‌ای در این ریلیز وجود ندارد.</li>';
            }

            const pubDate = new Date(releaseData.published_at).toLocaleDateString('fa-IR');
            return '<div class="rel-sec">' +
                '<div class="rel-header">' +
                    '<div><span class="rel-tag">نسخه: ' + releaseData.tag_name + '</span> <strong>' + (releaseData.name || '') + '</strong></div>' +
                    '<div style="font-size:12px; color:#888;">تاریخ انتشار: ' + pubDate + '</div>' +
                '</div>' +
                '<div class="rel-body markdown-body">' + renderMarkdown(releaseData.body) + '</div>' +
                '<div class="asset-search-box"><input type="text" class="asset-search-input" placeholder="🔍 فیلتر و جستجوی آنی در فایل‌ها (مثلاً: windows, arm64, apk, zip)..."></div>' +
                '<ul class="asset-list">' + assetsHtml + '</ul>' +
            '</div>';
        }

        function setupAssetSearch(card) {
            const searchInput = card.querySelector('.asset-search-input');
            if (!searchInput) return;

            searchInput.addEventListener('input', () => {
                const query = searchInput.value.trim().toLowerCase();
                const items = card.querySelectorAll('.asset-item');
                const seps = card.querySelectorAll('.os-separator');

                items.forEach(item => {
                    const name = item.getAttribute('data-name') || '';
                    if (!query || name.includes(query)) {
                        item.classList.remove('hidden');
                    } else {
                        item.classList.add('hidden');
                    }
                });

                seps.forEach(sep => {
                    const os = sep.getAttribute('data-os');
                    const visibleInGroup = card.querySelectorAll('.asset-item[data-os="' + os + '"]:not(.hidden)').length > 0;
                    if (visibleInGroup) {
                        sep.classList.remove('hidden');
                    } else {
                        sep.classList.add('hidden');
                    }
                });
            });
        }

        function getAssetOs(asset) {
            const name = asset.name.toLowerCase();
            for (const [os, data] of Object.entries(OS_Matchers)) {
                if (data.exts && data.exts.some(ext => name.endsWith(ext))) return os;
                if (data.names && data.names.some(n => name.includes(n))) return os;
            }
            return 'other';
        }

        function sortAssets(assets, clientInfo, bestAsset) {
            const osOrder = [clientInfo.os, 'android', 'windows', 'linux', 'macos', 'ios', 'bsd', 'other'];
            return [...assets].sort((a, b) => {
                if (bestAsset && a.id === bestAsset.id) return -1;
                if (bestAsset && b.id === bestAsset.id) return 1;
                const osA = getAssetOs(a);
                const osB = getAssetOs(b);
                const indexA = osOrder.indexOf(osA);
                const indexB = osOrder.indexOf(osB);
                if (indexA !== indexB) {
                    return (indexA === -1 ? 99 : indexA) - (indexB === -1 ? 99 : indexB);
                }
                return b.download_count - a.download_count;
            });
        }

        function findBestAsset(assets, clientInfo) {
            if (!clientInfo.os || clientInfo.os === 'other') return null;
            let bestAsset = null;
            let highestScore = -1;
            assets.forEach(asset => {
                const score = calculateAssetScore(asset, clientInfo);
                if (score > highestScore) {
                    highestScore = score;
                    bestAsset = asset;
                }
            });
            return highestScore > 0 ? bestAsset : null;
        }

        function calculateAssetScore(asset, clientInfo) {
            const name = asset.name.toLowerCase();
            let score = 0;
            const assetOs = getAssetOs(asset);

            if (clientInfo.os && assetOs === clientInfo.os) {
                score += 100;
                if (clientInfo.arch && clientInfo.arch !== 'auto') {
                    const archPatterns = Arch_Matchers[clientInfo.arch] || [];
                    if (archPatterns.some(p => name.includes(p))) {
                        score += 50;
                    }
                }
                const unwanted = ['.sha256', '.asc', '.dSYM.zip', '.plist', '.mobileprovision'];
                if (unwanted.some(ext => name.endsWith(ext))) {
                    score -= 80;
                }
            }
            return score;
        }

        async function forceDownloadFile(url, btn) {
            const originalText = btn.textContent;
            btn.textContent = 'دریافت...';
            btn.disabled = true;
            try {
                const a = document.createElement('a');
                a.href = url;
                a.download = '';
                document.body.appendChild(a);
                a.click();
                document.body.removeChild(a);
            } catch (e) {
                alert('خطا در دانلود. لطفاً از دکمه باز کردن استفاده کنید.');
            } finally {
                setTimeout(() => {
                    btn.textContent = originalText;
                    btn.disabled = false;
                }, 1500);
            }
        }

        function setupActionButtons(card) {
            card.querySelectorAll('.copy-btn').forEach(btn => {
                btn.addEventListener('click', () => {
                    const text = btn.getAttribute('data-url');
                    navigator.clipboard.writeText(text).then(() => {
                        const orig = btn.textContent;
                        btn.textContent = 'کپی شد!';
                        btn.classList.add('copied');
                        setTimeout(() => {
                            btn.textContent = orig;
                            btn.classList.remove('copied');
                        }, 2000);
                    });
                });
            });

            card.querySelectorAll('.force-dl-btn').forEach(btn => {
                btn.addEventListener('click', () => {
                    forceDownloadFile(btn.getAttribute('data-url'), btn);
                });
            });
        }

        function getClientInfo() {
            let os = DOM.osSelect.value;
            let arch = DOM.archSelect.value;
            if (os === 'auto') {
                const ua = navigator.userAgent.toLowerCase();
                if (ua.includes('android')) os = 'android';
                else if (ua.includes('win')) os = 'windows';
                else if (ua.includes('linux')) os = 'linux';
                else if (ua.includes('mac')) os = 'macos';
                else if (ua.includes('iphone') || ua.includes('ipad')) os = 'ios';
                else os = 'other';
            }
            return { os, arch };
        }

        async function processLinks() {
            const links = DOM.linksInput.value.split(String.fromCharCode(10)).map(l => l.trim()).filter(Boolean);
            DOM.results.innerHTML = '';
            generatedProxyUrls = [];
            DOM.copyAllBtn.style.display = 'none';

            if (links.length === 0) {
                alert('لطفاً حداقل یک لینک معتبر وارد کنید.');
                return;
            }

            DOM.convertBtn.disabled = true;
            DOM.convertBtn.textContent = 'در حال پردازش...';
            const clientInfo = getClientInfo();
            const isClashMode = DOM.clashModeCheckbox.checked;

            for (const link of links) {
                const converted = analyzeLink(link);
                let releaseData = null;

                if (converted.repoInfo) {
                    try {
                        const token = DOM.ghTokenInput.value.trim();
                        const reqHeaders = {};
                        if (token) reqHeaders['X-GitHub-Token'] = token;

                        const res = await fetch('/api/releases/' + converted.repoInfo.user + '/' + converted.repoInfo.repo + '?beta=' + DOM.betaCheckbox.checked, {
                            headers: reqHeaders
                        });
                        releaseData = await res.json();
                    } catch (e) {
                        console.error('Fetch Execution Error:', e);
                    }
                }
                displayResult(link, converted, releaseData, clientInfo, isClashMode);
            }

            DOM.convertBtn.disabled = false;
            DOM.convertBtn.textContent = 'پردازش لینک‌ها';

            if (generatedProxyUrls.length > 1) {
                DOM.copyAllBtn.style.display = 'block';
            }
        }

        DOM.copyAllBtn.addEventListener('click', () => {
            if (generatedProxyUrls.length > 0) {
                navigator.clipboard.writeText(generatedProxyUrls.join(String.fromCharCode(10))).then(() => {
                    showToast('تمامی ' + generatedProxyUrls.length + ' لینک پروکسی در کلیپ‌بورد کپی شدند!');
                });
            }
        });

        DOM.saveTokenBtn.addEventListener('click', () => {
            const val = DOM.ghTokenInput.value.trim();
            if (val) {
                localStorage.setItem('user_gh_token', val);
                showToast('توکن گیت‌هاب در مرورگر ذخیره شد.');
            }
        });

        DOM.delTokenBtn.addEventListener('click', () => {
            localStorage.removeItem('user_gh_token');
            DOM.ghTokenInput.value = '';
            showToast('توکن حذف شد.');
        });

        DOM.convertBtn.addEventListener('click', processLinks);
        DOM.themeToggle.addEventListener('click', toggleTheme);
        DOM.osSelect.addEventListener('change', () => {
            localStorage.setItem('user_os', DOM.osSelect.value);
            updateArchOptions();
        });

        DOM.scrollTopBtn.addEventListener('click', () => window.scrollTo({ top: 0, behavior: 'smooth' }));
        DOM.scrollBottomBtn.addEventListener('click', () => window.scrollTo({ top: document.body.scrollHeight, behavior: 'smooth' }));

        initializeUI();
    </script>
</body>
</html>`;
    return new Response(html, { headers: { 'Content-Type': 'text/html;charset=UTF-8' } });
  }
