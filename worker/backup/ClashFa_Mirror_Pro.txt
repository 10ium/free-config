export default {
  async fetch(request, env, ctx) {
    const url = new URL(request.url);

    // API برای دریافت ریلیزهای گیت‌هاب
    if (url.pathname.startsWith('/api/releases/')) {
      return this.handleApiRequest(url, request, env);
    }
    
    // API برای بررسی وضعیت و حجم فایل (متد HEAD)
    if (url.pathname === '/api/inspect') {
      return this.handleInspection(url);
    }

    // پروکسی دانلود قدرتمند
    if (url.pathname.startsWith('/download/')) {
      return this.handleDownloadProxy(request, ctx);
    }

    // رابط کاربری
    if (url.pathname === '/') {
      return this.serveHtmlUi();
    }

    return new Response('Not Found', { status: 404 });
  },

  async handleDownloadProxy(request, ctx) {
    const url = new URL(request.url);
    let targetUrlStr = url.pathname.substring('/download/'.length);
    
    if (!targetUrlStr) return new Response('Invalid URL', { status: 400 });
    
    if (url.search) targetUrlStr += url.search;
    if (!targetUrlStr.startsWith('http')) targetUrlStr = 'https://' + targetUrlStr;

    const isClashMode = url.searchParams.get('clash') === '1';
    
    const targetUrlObj = new URL(decodeURIComponent(targetUrlStr));
    targetUrlObj.searchParams.delete('clash');
    const normalizedTarget = targetUrlObj.toString();

    const cache = caches.default;
    const cacheKey = new Request(normalizedTarget, request);
    
    // کش بومی کلودفلر
    if (!isClashMode && request.method === 'GET') {
      let cachedRes = await cache.match(cacheKey);
      if (cachedRes) {
        let newHeaders = new Headers(cachedRes.headers);
        newHeaders.set('X-Worker-Cache', 'HIT');
        return new Response(cachedRes.body, { status: cachedRes.status, headers: newHeaders });
      }
    }

    try {
      const upstreamReq = new Request(normalizedTarget, {
        method: request.method,
        headers: {
          'User-Agent': 'Mozilla/5.0 (MirrorPRO; Enterprise Gateway)',
          'Range': request.headers.get('Range') || '' 
        },
        redirect: 'follow'
      });

      const upstreamRes = await fetch(upstreamReq);

      if (!upstreamRes.ok && upstreamRes.status !== 304 && upstreamRes.status !== 206) {
        return upstreamRes;
      }

      let resHeaders = new Headers(upstreamRes.headers);
      resHeaders.set('Access-Control-Allow-Origin', '*');
      resHeaders.set('X-Worker-Cache', 'MISS');

      // پردازش هوشمند کلش برای تغییر لینک‌های پروایدر
      if (isClashMode && (normalizedTarget.includes('.yaml') || normalizedTarget.includes('.yml') || normalizedTarget.includes('.txt'))) {
        const textContent = await upstreamRes.text();
        const workerOrigin = url.origin;
        
        let inProviderBlock = false;
        let healthCheckIndent = -1;
        const lines = textContent.split('\n');
        
        const processedLines = lines.map(line => {
          const indentMatch = line.match(/^(\s*)/);
          const indent = indentMatch ? indentMatch[1].length : 0;
          const trimmedLine = line.trim();
          
          // تشخیص کلیدهای اصلی (Root Level)
          const isRoot = indent === 0 && trimmedLine.length > 0 && !trimmedLine.startsWith('#');
          
          if (isRoot) {
            // اگر وارد بلاک‌های پروایدر شدیم
            if (/^(rule-providers|proxy-providers)\s*:/i.test(line)) {
              inProviderBlock = true;
            } else {
              inProviderBlock = false;
            }
            healthCheckIndent = -1;
            return line;
          }
          
          if (inProviderBlock) {
            // بررسی ورود به بلاک health-check
            if (/^\s+health-check\s*:/i.test(line)) {
              healthCheckIndent = indent;
            } else if (healthCheckIndent !== -1 && indent <= healthCheckIndent && trimmedLine.length > 0 && !trimmedLine.startsWith('#')) {
              // خروج از بلاک health-check
              healthCheckIndent = -1;
            }
            
            // جایگزینی لینک در صورتی که داخل پروایدر باشیم اما در health-check نباشیم
            if (healthCheckIndent === -1) {
              return line.replace(/(https?:\/\/[^\s"'\|]+)/g, (match) => {
                // نادیده گرفتن لینک‌های قبلی و دامنه‌های مربوط به health check
                if (match.includes(workerOrigin) || 
                    match.includes('gstatic.com') || 
                    match.includes('cloudflare.com') || 
                    match.includes('v2fly.org') || 
                    match.includes('1.1.1.1')) {
                  return match;
                }
                return workerOrigin + '/download/' + encodeURIComponent(match);
              });
            }
          }
          return line;
        });

        const processedText = processedLines.join('\n');

        resHeaders.set('Content-Type', 'text/plain; charset=utf-8');
        return new Response(processedText, { status: 200, headers: resHeaders });
      }

      // کش یک ساله در کلودفلر
      if (upstreamRes.status === 200) {
        resHeaders.set('Cache-Control', 'public, max-age=31536000');
      }

      const response = new Response(upstreamRes.body, {
        status: upstreamRes.status,
        statusText: upstreamRes.statusText,
        headers: resHeaders
      });

      if (upstreamRes.status === 200 && !isClashMode && request.method === 'GET') {
        ctx.waitUntil(cache.put(cacheKey, response.clone()));
      }

      return response;
    } catch (e) {
      return new Response('Error proxying download: ' + e.message, { status: 500 });
    }
  },

  async handleInspection(url) {
    const target = url.searchParams.get("url");
    if (!target) return new Response(JSON.stringify({ success: false }), { status: 400 });
    try {
      const res = await fetch(decodeURIComponent(target), { method: "HEAD", headers: { "User-Agent": "MirrorPRO" } });
      const size = res.headers.get("content-length") || 0;
      return new Response(JSON.stringify({ success: true, size: size }), { headers: { "Content-Type": "application/json" } });
    } catch (e) {
      return new Response(JSON.stringify({ success: false, error: e.message }), { status: 500 });
    }
  },
  
  async handleApiRequest(url, request, env) {
    const parts = url.pathname.split('/');
    const contentType = { headers: { 'Content-Type': 'application/json' } };
    if (parts.length < 5) return new Response(JSON.stringify({ _apiError: 'مسیر درخواست نامعتبر است.' }), contentType); 
    
    const user = parts[3];
    const repo = parts[4];
    const includePrerelease = url.searchParams.get('beta') === 'true';
    const apiUrl = includePrerelease ? 'https://api.github.com/repos/' + user + '/' + repo + '/releases?per_page=1' : 'https://api.github.com/repos/' + user + '/' + repo + '/releases/latest';
    
    const headers = { 'User-Agent': 'Cloudflare-Worker-GitHub-Tool', 'Accept': 'application/vnd.github.v3+json' };
    
    const clientToken = request.headers.get('X-GitHub-Token');
    if (clientToken) {
      headers.Authorization = 'token ' + clientToken;
    } else if (env && env.GITHUB_TOKEN) {
      headers.Authorization = 'token ' + env.GITHUB_TOKEN;
    }
    
    try {
      const response = await fetch(apiUrl, { headers });
      
      if (response.status === 403 || response.status === 429) {
          return new Response(JSON.stringify({ _apiError: 'محدودیت درخواست گیت‌هاب (Rate Limit) پر شده است. لطفاً توکن خود را در تنظیمات وارد کنید.' }), contentType);
      }
      if (!response.ok) {
          return new Response(JSON.stringify({ _apiError: 'نسخه ریلیز برای این مخزن یافت نشد.' }), contentType);
      }
      
      let data = await response.json();
      if (Array.isArray(data)) data = data[0]; 
      if (!data) return new Response(JSON.stringify({ _apiError: 'هیچ فایلی در ریلیز یافت نشد.' }), contentType); 
      
      return new Response(JSON.stringify(data), contentType);
    } catch (error) { 
      return new Response(JSON.stringify({ _apiError: 'خطای سرور: ' + error.message }), contentType); 
    }
  },

        function analyzeLink(urlStr) {
            let result = { original: urlStr, isGithub: false, profileLink: null, rawLink: null, blobLink: null, repoInfo: null };
            try {
                const url = new URL(urlStr);
                const host = url.hostname.toLowerCase();
                const parts = url.pathname.split('/').filter(Boolean);

                if (host === 'github.com' || host === 'www.github.com') {
                    result.isGithub = true;
                    if (parts.length >= 1) result.profileLink = 'https://github.com/' + parts[0];
                    if (parts.length >= 2) result.repoInfo = { user: parts[0], repo: parts[1] };
                    
                    if (parts.length >= 4 && parts[2] === 'blob') {
                        const branch = parts[3];
                        const filePath = parts.slice(4).join('/');
                        result.blobLink = urlStr;
                        result.rawLink = 'https://raw.githubusercontent.com/' + parts[0] + '/' + parts[1] + '/refs/heads/' + branch + '/' + filePath;
                    } else if (parts.length >= 4 && parts[2] === 'raw') {
                        result.rawLink = urlStr;
                        result.blobLink = 'https://github.com/' + parts[0] + '/' + parts[1] + '/blob/' + parts[3] + '/' + parts.slice(4).join('/');
                    }
                } else if (host === 'raw.githubusercontent.com') {
                    result.isGithub = true;
                    if (parts.length >= 1) result.profileLink = 'https://github.com/' + parts[0];
                    if (parts.length >= 2) result.repoInfo = { user: parts[0], repo: parts[1] };
                    
                    result.rawLink = urlStr;
                    if (parts.length >= 5 && parts[2] === 'refs' && parts[3] === 'heads') {
                        const branch = parts[4];
                        const filePath = parts.slice(5).join('/');
                        result.blobLink = 'https://github.com/' + parts[0] + '/' + parts[1] + '/blob/' + branch + '/' + filePath;
                    } else if (parts.length >= 3) {
                        const branch = parts[2];
                        const filePath = parts.slice(3).join('/');
                        result.blobLink = 'https://github.com/' + parts[0] + '/' + parts[1] + '/blob/' + branch + '/' + filePath;
                    }
                } else if (host.endsWith('.github.io')) {
                    result.isGithub = true;
                    const user = host.replace('.github.io', '');
                    result.profileLink = 'https://github.com/' + user;
                    result.repoInfo = { user: user, repo: parts.length > 0 ? parts[0] : user + '.github.io' };
                }
            } catch (e) { }
            return result;
        }

        function getAssetOs(asset) {
            const name = asset.name.toLowerCase();
            for (const os in OS_Matchers) {
                const matcher = OS_Matchers[os];
                if (matcher.names.some(n => name.includes(n)) || (matcher.exts.length > 0 && matcher.exts.some(e => name.endsWith(e)))) {
                    return os;
                }
            }
            return 'other';
        }

        function sortAssets(assets, clientInfo, bestAsset) {
            const defaultOsPriority = ['android', 'windows', 'macos', 'linux', 'ios', 'bsd', 'other'];
            return [...assets].sort((a, b) => {
                if (bestAsset) {
                    if (a.id === bestAsset.id) return -1;
                    if (b.id === bestAsset.id) return 1;
                }
                const osA = getAssetOs(a);
                const osB = getAssetOs(b);
                const aIsMatch = osA === clientInfo.os;
                const bIsMatch = osB === clientInfo.os;
                if (aIsMatch && !bIsMatch) return -1;
                if (!aIsMatch && bIsMatch) return 1;
                const priorityA = defaultOsPriority.indexOf(osA);
                const priorityB = defaultOsPriority.indexOf(osB);
                if (priorityA !== priorityB) return priorityA - priorityB;
                return a.name.localeCompare(b.name);
            });
        }
        
        function findBestAsset(assets, clientInfo) {
            const scores = assets.map(asset => ({ asset, score: calculateAssetScore(asset, clientInfo) }));
            if (scores.length === 0) return null;
            const best = scores.reduce((a, b) => a.score > b.score ? a : b);
            return best.score > 0 ? best.asset : null;
        }
        
        function calculateAssetScore(asset, clientInfo) {
            const name = asset.name.toLowerCase();
            const unwanted = ['.sha256', '.asc', '.dSYM.zip', '.plist', '.mobileprovision'];
            if (unwanted.some(term => name.endsWith(term))) return -1;
            let score = 0;
            const archMatchers = {
                x64: ['x64', 'amd64', 'x86_64'], arm64: ['arm64', 'aarch64'], arm64e: ['arm64e'],
                'arm64-v8a': ['arm64-v8a'], 'armeabi-v7a': ['armeabi-v7a', 'armv7'],
                x86: ['x86', '386', 'i686'], universal: ['universal', 'universal2'],
            };
            const assetOs = getAssetOs(asset);
            const osMatches = assetOs === clientInfo.os;
            if (osMatches) score += 10;
            if (osMatches && clientInfo.os === 'android' && name.includes('universal')) score += 50;
            const selectedArchKeywords = archMatchers[clientInfo.arch];
            const archMatches = selectedArchKeywords && selectedArchKeywords.some(keyword => name.includes(keyword));
            if (archMatches) score += 5;
            if (osMatches && archMatches) score += 2;
            return score;
        }

        async function forceDownloadFile(url, btn) {
            const originalText = btn.textContent;
            btn.textContent = 'در حال دریافت...';
            btn.disabled = true;
            try {
                // دریافت فایل به صورت خام
                const response = await fetch(url);
                if (!response.ok) throw new Error('Network error');
                const blob = await response.blob();
                const blobUrl = URL.createObjectURL(blob);
                const link = document.createElement('a');
                link.href = blobUrl;
                
                let filename = 'download';
                try {
                    // استخراج نام فایل از آدرس
                    const cleanUrl = decodeURIComponent(url.split('/download/')[1] || url).split('?')[0];
                    filename = cleanUrl.substring(cleanUrl.lastIndexOf('/') + 1) || 'download';
                } catch(e){}

                // اگر فایل اکستنشن (فرمت) نداشت، آن را متن در نظر می‌گیریم تا به صورت فایل ذخیره شود
                if (filename.indexOf('.') === -1) {
                    filename += '.txt';
                }

                link.download = filename;
                document.body.appendChild(link);
                link.click();
                document.body.removeChild(link);
                URL.revokeObjectURL(blobUrl);
            } catch (error) {
                console.error('Download error:', error);
                // فال‌بک: در صورت خطای CORS (فیلتر بودن مستقیم)، لینک را در تب جدید باز می‌کند
                window.open(url, '_blank');
            } finally {
                btn.textContent = originalText;
                btn.disabled = false;
            }
        }

        function setupActionButtons(card) {
            card.addEventListener('click', function(event) {
                const target = event.target;
                if (target.classList.contains('copy-btn')) {
                    const textToCopy = target.dataset.url || target.parentElement.querySelector('input').value;
                    if (textToCopy) {
                        navigator.clipboard.writeText(textToCopy);
                        const originalText = target.textContent;
                        target.textContent = '✔️ کپی شد';
                        target.classList.add('copied');
                        setTimeout(() => {
                            target.textContent = originalText;
                            target.classList.remove('copied');
                        }, 2000);
                    }
                } else if (target.classList.contains('force-dl-btn')) {
                    const url = target.dataset.url;
                    if (url) forceDownloadFile(url, target);
                }
            });
        }
        
        function getClientInfo() {
            const selectedOs = DOM.osSelect.value;
