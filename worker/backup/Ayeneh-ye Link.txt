export default {
  async fetch(request) {
    const url = new URL(request.url);
    const path = url.pathname;

    if (path === "/favicon.ico") return serveFavicon();
    if (path === "/") return serveLandingPage();

    const cleanPath = path.slice(1);
    const firstPart = cleanPath.split('/')[0];

    if (firstPart.includes('.')) return handleProxyRequest(request);
    
    return handleGitHubProxyRequest(request);
  }
};

  return new Response(faviconSvg, { headers: { 'Content-Type': 'image/svg+xml' } });
}

async function processAndMirrorYamlContent(yamlText, requestUrl) {
    const workerOrigin = new URL(requestUrl).origin;

    const generateCleanWorkerUrl = (sourceUrl) => {
        if (!sourceUrl || typeof sourceUrl !== 'string') return '';
        const trimmedUrl = sourceUrl.trim();
        try {
            const url = new URL(trimmedUrl);
            const pathParts = url.pathname.split('/').filter(p => p);
            if (url.hostname === 'github.com' && pathParts.length >= 4 && pathParts[2] === 'blob') {
                const [owner, repo, , ...rest] = pathParts;
                return `${workerOrigin}/${owner}/${repo}/${rest.join('/')}`;
            }
            if (url.hostname === 'raw.githubusercontent.com' && pathParts.length >= 3) {
                const [owner, repo, ...rest] = pathParts;
                let restOfPath = rest.join('/');
                restOfPath = restOfPath.replace(/^refs\/heads\//, '');
                return `${workerOrigin}/${owner}/${repo}/${restOfPath}`;
            }
            const proxyPath = trimmedUrl.replace(/^https?:\/\//, '');
            return `${workerOrigin}/${proxyPath}`;
        } catch (e) {
            const proxyPath = trimmedUrl.replace(/^https?:\/\//, '');
            return `${workerOrigin}/${proxyPath}`;
        }
    };
    
    const mirrorableExtensions = ['.yaml', '.yml', '.txt', '.json'];
    const newLines = [];
    let processNextLineAsUrl = false;

    for (const line of yamlText.split('\n')) {
        const trimmedLine = line.trim();

        const checkAndReplaceUrl = (originalUrl) => {
            const isMirrorableExtension = mirrorableExtensions.some(ext => originalUrl.endsWith(ext));
            const isSubconverterClashLink = originalUrl.includes('target=clash');
            const shouldMirror = isMirrorableExtension || isSubconverterClashLink;

            if (shouldMirror) {
                const workerUrl = generateCleanWorkerUrl(originalUrl);
                return line.replace(originalUrl, workerUrl);
            }
            return null;
        };
        
        if (processNextLineAsUrl) {
            processNextLineAsUrl = false;
            const urlMatch = trimmedLine.match(/^(https?:[/]{2}[^"'\s]+)/);
            if (urlMatch) {
                const replacedLine = checkAndReplaceUrl(urlMatch[0]);
                if (replacedLine) {
                    newLines.push(replacedLine);
                    continue;
                }
            }
            newLines.push(line);
            continue;
        }

        const singleLineUrlRegex = /(:\s*["']?)(https?:[/]{2}[^"'\s]+)(["']?)/;
        const singleLineMatch = trimmedLine.match(singleLineUrlRegex);
        if (singleLineMatch) {
            const replacedLine = checkAndReplaceUrl(singleLineMatch[2]);
            if (replacedLine) {
                newLines.push(replacedLine);
                continue;
            }
        }
        
        if (trimmedLine.match(/^url:\s*(?:>|\|)-?\s*$/)) {
            processNextLineAsUrl = true;
        }
        
        newLines.push(line);
    }

    return newLines.join('\n');
}

async function handleProxyRequest(request) {
  const url = new URL(request.url);
  let targetUrlString = url.pathname.slice(1) + url.search;
  if (!targetUrlString.startsWith('http')) targetUrlString = 'https://' + targetUrlString;
  
  try {
    const originUrl = new URL(targetUrlString);
    originUrl.searchParams.delete('process');
    
    const response = await fetch(originUrl.toString());
    if (!response.ok) return new Response(`Error from origin server: ${response.status}`, { status: response.status });
    
    const isYaml = url.pathname.endsWith('.yaml') || url.pathname.endsWith('.yml');
    const shouldProcess = url.searchParams.get('process') === 'clash';
    
    if (isYaml && shouldProcess) {
      const originalYaml = await response.text();
      const processedYaml = await processAndMirrorYamlContent(originalYaml, request.url);
      const newHeaders = new Headers(response.headers);
      newHeaders.set('Access-Control-Allow-Origin', '*');
      newHeaders.set('Content-Type', 'text/yaml; charset=utf-8');
      return new Response(processedYaml, { status: 200, headers: newHeaders });
    }
    
    const headers = new Headers(response.headers);
    headers.set('Access-Control-Allow-Origin', '*');
    return new Response(response.body, { status: response.status, headers: headers });
  } catch (error) {
    return new Response(`Worker error: ${error.message}`, { status: 500 });
  }
}

async function handleGitHubProxyRequest(request) {
  const url = new URL(request.url);
  const cleanPath = url.pathname.replace(/^\/+/, "");
  const parts = cleanPath.split("/");
  if (parts.length < 3) return new Response("Invalid GitHub path.", { status: 400 });

  const [owner, repo, ...restOfPathParts] = parts;
  const restOfPath = restOfPathParts.join("/");
  const rawUrl = `https://raw.githubusercontent.com/${owner}/${repo}/${restOfPath}`;

  try {
    const response = await fetch(rawUrl);
    if (!response.ok) return new Response(`Error fetching from GitHub: ${response.status}`, { status: response.status });

    const isYaml = url.pathname.endsWith('.yaml') || url.pathname.endsWith('.yml');
    const shouldProcess = url.searchParams.get('process') === 'clash';

    if (isYaml && shouldProcess) {
        const originalYaml = await response.text();
        const processedYaml = await processAndMirrorYamlContent(originalYaml, request.url);
        const newHeaders = new Headers(response.headers);
        newHeaders.set('Access-Control-Allow-Origin', '*');
        newHeaders.set('Content-Type', 'text/yaml; charset=utf-8');
        return new Response(processedYaml, { status: 200, headers: newHeaders });
    }
    
    const headers = new Headers(response.headers);
    headers.set('Access-Control-Allow-Origin', '*');
    return new Response(response.body, { status: response.status, headers: headers });
  } catch (error) {
    return new Response(`Worker error: ${error.message}`, { status: 500 });
  }
}
