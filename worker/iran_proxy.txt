export default {
  async fetch(request, env, ctx) {
    if (request.method === 'OPTIONS') {
      return new Response(null, {
        headers: {
          'Access-Control-Allow-Origin': '*',
          'Access-Control-Allow-Methods': 'GET, POST, OPTIONS',
          'Access-Control-Allow-Headers': '*'
        }
      });
    }

    if (request.method !== 'GET' && request.method !== 'POST') {
      return new Response('Method Not Allowed', { status: 405 });
    }

    const url = new URL(request.url);
    const { pathname, searchParams } = url;
    const protocolFilter = (searchParams.get('protocol') || '').toLowerCase().trim();
    const isApiRoute = pathname.startsWith('/api/') || pathname.includes('/tested');

    if (pathname === '/favicon.ico') {
      return new Response('<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 100 100"><text y=".9em" font-size="90">⚡</text></svg>', {
        headers: { 'Content-Type': 'image/svg+xml', 'Cache-Control': 'public, max-age=86400' }
      });
    }

    // 1. بررسی حافظه کش کلودفلر برای پاسخ‌دهی برق‌آسا (فقط برای مسیرهای عمومی و غیرتست)
    const cache = typeof caches !== 'undefined' ? caches.default : null;
    const cacheKey = new Request(url.toString(), request);
    if (cache && !searchParams.has('nocache') && !isApiRoute) {
      const cachedResponse = await cache.match(cacheKey);
      if (cachedResponse) {
        return cachedResponse;
      }
    }

    try {
      // 2. واکشی و پالایش داده‌ها
      const { uniqueProxies, report } = await fetchAndProcessProxies();

      const corsHeaders = {
        'Access-Control-Allow-Origin': '*',
        'Access-Control-Allow-Methods': 'GET, POST, OPTIONS',
        'Cache-Control': isApiRoute ? 'no-cache, no-store' : `public, max-age=${CACHE_TTL_SECONDS}, s-maxage=${CACHE_TTL_SECONDS}`
      };

      // مسیر API: تست دسته‌ای پروکسی‌ها با سوکت TCP و اعتبارسنجی loc=IR
      if (pathname === '/api/test-batch') {
        const offset = Math.max(0, parseInt(searchParams.get('offset') || '0', 10));
        const limit = Math.min(Math.max(1, parseInt(searchParams.get('limit') || '10', 10)), 20);
        const batch = uniqueProxies.slice(offset, offset + limit);

        const testPromises = batch.map(async (proxyStr) => {
          const match = proxyStr.match(/^(socks5|socks4|https|http):\/\/([^:]+):(\d+)$/);
          if (!match) return { proxy: proxyStr, ok: false, isIran: false, error: 'Invalid format' };
          const [, proto, ip, port] = match;
          const res = await testProxy(proto, ip, parseInt(port, 10), 2200);
          return { proxy: proxyStr, proto, ip, port, ...res };
        });

        const results = await Promise.all(testPromises);
        const aliveThisBatch = results.filter(r => r.ok && r.isIran).map(r => r.proxy);

        // ادغام با لیست ذخیره‌شده در کش کلودفلر
        let existingTested = await getStoredTestedProxies(cache);
        const merged = [...new Set([...existingTested, ...aliveThisBatch])];
        if (aliveThisBatch.length > 0) {
          await saveStoredTestedProxies(cache, ctx, merged);
        }

        const nextOffset = offset + batch.length < uniqueProxies.length ? offset + batch.length : null;

        return new Response(JSON.stringify({
          success: true,
          offset,
          batchSize: batch.length,
          totalAvailable: uniqueProxies.length,
          nextOffset,
          batchAliveCount: aliveThisBatch.length,
          totalTestedAliveSoFar: merged.length,
          aliveProxies: aliveThisBatch,
          results
        }), {
          headers: { ...corsHeaders, 'Content-Type': 'application/json; charset=utf-8' }
        });
      }

      // مسیر API: ریست و پاکسازی حافظه تست
      if (pathname === '/api/reset-tested') {
        await resetStoredTestedProxies(cache, ctx);
        return new Response(JSON.stringify({
          success: true,
          message: 'حافظه پروکسی‌های تست‌شده با موفقیت پاکسازی و ریست شد.'
        }), {
          headers: { ...corsHeaders, 'Content-Type': 'application/json; charset=utf-8' }
        });
      }

      // مسیر API: دریافت لیست تست‌شده‌ها به صورت JSON
      if (pathname === '/api/tested-list') {
        const tested = await getStoredTestedProxies(cache);
        return new Response(JSON.stringify({ success: true, count: tested.length, proxies: tested }), {
          headers: { ...corsHeaders, 'Content-Type': 'application/json; charset=utf-8' }
        });
      }

      // خروجی اختصاصی ساب‌اسکریپشن ایکس‌ری برای پروکسی‌های تست‌شده سالم
      if (pathname === '/xray/tested' || pathname === '/sub/tested') {
        const tested = await getStoredTestedProxies(cache);
        let output = '';
        if (tested && tested.length > 0) {
          output = tested.join('\n');
        } else {
          output = `# هنوز تست از طریق داشبورد وب انجام نشده است.\n# لطفاً داشبورد را باز کرده و دکمه تست خودکار را بزنید تا فقط پروکسی‌های تاییدشده قرار گیرند.\n` + uniqueProxies.slice(0, 20).join('\n');
        }

        return new Response(output, {
          headers: {
            ...corsHeaders,
            'Content-Type': 'text/plain; charset=utf-8'
          }
        });
      }

      // خروجی کلش برای پروکسی‌های تست‌شده سالم
      if (pathname === '/clash/tested') {
        const tested = await getStoredTestedProxies(cache);
        const listToFormat = (tested && tested.length > 0) ? tested : uniqueProxies.slice(0, 20);
        return new Response(formatForClash(listToFormat), {
          headers: {
            ...corsHeaders,
            'Content-Type': 'text/yaml; charset=utf-8'
          }
        });
      }

      // اعمال فیلتر بر اساس پارامتر آدرس (Query Params)
      let filteredProxies = uniqueProxies;
      if (protocolFilter) {
        filteredProxies = uniqueProxies.filter(p => {
          const proto = (p.split('://')[0] || '').toLowerCase();
          return proto === protocolFilter;
        });
      }

      let response;

      switch (pathname) {
        case '/xray':
        case '/sub':
          response = new Response(filteredProxies.join('\n'), {
            headers: {
              ...corsHeaders,
              'Content-Type': 'text/plain; charset=utf-8'
            }
          });
          break;

        case '/clash':
        case '/clash.yaml':
          response = new Response(formatForClash(filteredProxies), {
            headers: {
              ...corsHeaders,
              'Content-Type': 'text/yaml; charset=utf-8'
            }
          });
          break;

        case '/json':
          response = new Response(JSON.stringify({
            success: true,
            total: filteredProxies.length,
            protocol_filter: protocolFilter || 'all',
            summary: report.خلاصه_گزارش,
            proxies: filteredProxies.map(p => {
              const [proto, rest] = p.split('://');
              const [ip, port] = (rest || '').split(':');
              return { protocol: proto, ip, port: parseInt(port, 10), raw: p };
            })
          }, null, 2), {
            headers: {
              ...corsHeaders,
              'Content-Type': 'application/json; charset=utf-8'
            }
          });
          break;

        case '/':
          const baseUrl = `${url.protocol}//${url.host}`;
          const colo = request.cf ? (request.cf.colo || 'EDGE') : 'LOCAL';
          const visitorCountry = request.cf ? (request.cf.country || 'IR') : 'IR';
          const html = render2055Dashboard(report, uniqueProxies, baseUrl, colo, visitorCountry);

          response = new Response(html, {
            headers: {
              ...corsHeaders,
              'Content-Type': 'text/html; charset=utf-8'
            }
          });
          break;

        default:
          return new Response('404 Not Found', { status: 404 });
      }

      // ذخیره پاسخ در کش کلودفلر (فقط برای مسیرهای غیر API)
      if (cache && !isApiRoute && ctx && typeof ctx.waitUntil === 'function') {
        ctx.waitUntil(cache.put(cacheKey, response.clone()));
      }

      return response;
    } catch (err) {
      console.error('Worker internal error:', err);
      return new Response('Internal Server Error: ' + err.message, { status: 500 });
    }
  }
};

async function getStoredTestedProxies(cache) {
  if (!cache) return [];
  try {
    const key = new Request('https://iran-proxy-collector.internal/tested_proxies_list');
    const resp = await cache.match(key);
    if (resp) {
      return await resp.json();
    }
  } catch (_) {}
  return [];
}

async function saveStoredTestedProxies(cache, ctx, proxies) {
  if (!cache) return;
  try {
    const key = new Request('https://iran-proxy-collector.internal/tested_proxies_list');
    const resp = new Response(JSON.stringify(proxies), {
      headers: {
        'Content-Type': 'application/json',
        'Cache-Control': 'public, max-age=86400, s-maxage=86400'
      }
    });
    if (ctx && typeof ctx.waitUntil === 'function') {
      ctx.waitUntil(cache.put(key, resp));
    } else {
      await cache.put(key, resp);
    }
  } catch (_) {}
}

async function resetStoredTestedProxies(cache, ctx) {
  if (!cache) return;
  try {
    const key = new Request('https://iran-proxy-collector.internal/tested_proxies_list');
    if (ctx && typeof ctx.waitUntil === 'function') {
      ctx.waitUntil(cache.delete(key));
    } else {
      await cache.delete(key);
    }
  } catch (_) {}
}

async function testProxy(proto, ip, port, timeoutMs = 3000) {
  let socket = null;
  let timeoutId = null;

  const timeoutPromise = new Promise((_, reject) => {
    timeoutId = setTimeout(() => reject(new Error('TIMEOUT')), timeoutMs);
  });

  const testPromise = (async () => {
    try {
      socket = connect(`${ip}:${port}`);
      const writer = socket.writable.getWriter();
      const reader = socket.readable.getReader();

      const CF_HOST = 'icanhazip.com';
      const httpReq = `GET / HTTP/1.1\r\nHost: ${CF_HOST}\r\nUser-Agent: curl/7.88.1\r\nConnection: close\r\n\r\n`;

      if (proto.startsWith('socks5')) {
        // Step 1: Handshake
        await writer.write(new Uint8Array([0x05, 0x01, 0x00]));
        const { value: hResp } = await reader.read();
        if (!hResp || hResp[0] !== 0x05 || hResp[1] !== 0x00) {
          throw new Error('SOCKS5 auth failed');
        }

        // Step 2: Connect to 104.18.27.120:80 (icanhazip Cloudflare anycast IP)
        await writer.write(new Uint8Array([0x05, 0x01, 0x00, 0x01, 104, 18, 27, 120, 0, 80]));
        const { value: cResp } = await reader.read();
        if (!cResp || cResp[1] !== 0x00) {
          throw new Error('SOCKS5 connect failed');
        }
        await writer.write(new TextEncoder().encode(httpReq));

      } else if (proto.startsWith('socks4')) {
        // SOCKS4 connect to 104.18.27.120:80
        await writer.write(new Uint8Array([0x04, 0x01, 0, 80, 104, 18, 27, 120, 0]));
        const { value: cResp } = await reader.read();
        if (!cResp || cResp[1] !== 0x5a) {
          throw new Error('SOCKS4 connect failed');
        }
        await writer.write(new TextEncoder().encode(httpReq));

      } else if (proto === 'https') {
        // CONNECT icanhazip.com:80
        await writer.write(new TextEncoder().encode(`CONNECT ${CF_HOST}:80 HTTP/1.1\r\nHost: ${CF_HOST}:80\r\n\r
`));
        const { value: cResp } = await reader.read();
        const cStr = cResp ? new TextDecoder().decode(cResp) : '';
        if (cStr.includes('200')) {
          await writer.write(new TextEncoder().encode(httpReq));
        } else {
          // Fallback to plain GET proxy
          await writer.write(new TextEncoder().encode(`GET http://${CF_HOST}/ HTTP/1.1\r\nHost: ${CF_HOST}\r\nUser-Agent: curl/7.88.1\r\nConnection: close\r\n\r
`));
        }
      } else {
        // HTTP forward proxy: try plain GET
        const reqStr = `GET http://${CF_HOST}/ HTTP/1.1\r\nHost: ${CF_HOST}\r\nUser-Agent: curl/7.88.1\r\nConnection: close\r\n\r
`;
        await writer.write(new TextEncoder().encode(reqStr));
      }

      const { value: bodyBytes } = await reader.read();
      if (!bodyBytes) throw new Error('Empty response');
      const text = new TextDecoder().decode(bodyBytes);

      try { writer.releaseLock(); } catch (_) {}
      try { reader.releaseLock(); } catch (_) {}
      try { socket.close(); } catch (_) {}

      // Split header and body
      const parts = text.split('\r\n\r\n');
      const header = parts[0] || '';
      const body = parts[1] || '';

      // Match exit IPv4 in body or text
      const ipMatch = body.match(/\b(?:\d{1,3}\.){3}\d{1,3}\b/) || text.match(/\b(?:\d{1,3}\.){3}\d{1,3}\b/);
      const exitIp = ipMatch ? ipMatch[0] : null;

      const is200 = header.includes('200') || header.includes('OK') || text.includes('200') || (exitIp !== null);
      const isIran = exitIp !== null;

      return {
        ok: is200 && isIran,
        isIran,
        exitIp,
        hasTrace: isIran
      };
    } catch (err) {
      if (socket) {
        try { socket.close(); } catch (_) {}
      }
      return { ok: false, isIran: false, error: err.message };
    }
  })();

  try {
    const res = await Promise.race([testPromise, timeoutPromise]);
    clearTimeout(timeoutId);
    return res;
  } catch (err) {
    clearTimeout(timeoutId);
    if (socket) {
      try { socket.close(); } catch (_) {}
    }
    return { ok: false, isIran: false, error: err.message };
  }
}

function getFriendlySourceName(sourceUrl) {
  try {
    const url = new URL(sourceUrl);
    if (url.hostname === 'raw.githubusercontent.com') {
      const parts = url.pathname.split('/').filter(p => p);
      if (parts.length >= 3) {
        return `${parts[0]}/${parts[1]}`;
      }
    }
    if (url.hostname.includes('proxyscrape.com')) {
      const proto = url.searchParams.get('protocol') || 'all';
      return `ProxyScrape (${proto.toUpperCase()})`;
    }
    if (url.hostname.includes('geonode.com')) {
      return `GeoNode API (Live)`;
    }
    if (url.hostname.includes('proxyspace.pro')) {
      return 'ProxySpace (Web)';
    }
    if (url.hostname.includes('proxycheck.app')) {
      return 'ProxyCheck (Web)';
    }
    if (url.hostname.includes('freeproxyupdate.com')) {
      return 'FreeProxyUpdate (Web)';
    }
    if (url.hostname.includes('freeproxy.world')) {
      return 'FreeProxyWorld (Web)';
    }
    if (url.hostname.includes('proxy-spider.com')) {
      return 'ProxySpider (Web)';
    }
    const parts = url.pathname.split('/').filter(p => p);
    return `${url.hostname}/${parts[parts.length - 1] || ''}`;
  } catch (e) {
    return sourceUrl;
  }
}

function isValidIP(ip) {
  if (!ip || typeof ip !== 'string') return false;
  const parts = ip.split('.');
  if (parts.length !== 4) return false;
  for (const part of parts) {
    if (!/^\d+$/.test(part)) return false;
    const n = parseInt(part, 10);
    if (n < 0 || n > 255) return false;
    if (part.length > 1 && part.startsWith('0')) return false;
  }
  // رد کردن آی‌پی‌های محلی و رزرو شده
  if (parts[0] === '127' || parts[0] === '10' || parts[0] === '0') return false;
  if (parts[0] === '192' && parts[1] === '168') return false;
  if (parts[0] === '172' && parseInt(parts[1], 10) >= 16 && parseInt(parts[1], 10) <= 31) return false;
  // رد کردن آی‌پی‌های ماکت و داکیومنتی (RFC 5737)
  if (parts[0] === '192' && parts[1] === '0' && parts[2] === '2') return false;
  if (parts[0] === '198' && parts[1] === '51' && parts[2] === '100') return false;
  if (parts[0] === '203' && parts[1] === '0' && parts[2] === '113') return false;
  return true;
}

function isValidPort(port) {
  const p = parseInt(port, 10);
  return !isNaN(p) && p >= 1 && p <= 65535;
}

function cleanProxyString(proto, ip, port) {
  if (!isValidIP(ip) || !isValidPort(port)) return null;
  return `${proto.toLowerCase()}://${ip}:${port}`;
}

async function fetchAndProcessProxies() {
  const sourceReports = [];
  const promises = PROXY_SOURCES.map(source =>
    fetch(source.url, { headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64)' } })
      .then(async response => {
        if (!response.ok) throw new Error(`Status ${response.status}`);
        if (source.type === 'json_vakhov') {
          const text = await response.text();
          if (!text || text.trim() === '') return [];
          const corrected = '[' + text.replace(/}\s*{/g, '},{') + ']';
          try { return JSON.parse(corrected); }
          catch (e) { throw new Error('Invalid JSON stream'); }
        }
        if (source.type === 'ndjson_fate0' || source.type.startsWith('text') || source.type.startsWith('html_')) {
          return response.text();
        }
        return response.json();
      })
      .then(data => {
        const { proxies, stats } = parseProxies(data, source);
        const friendlyName = getFriendlySourceName(source.url);
        sourceReports.push({
          source: friendlyName,
          status: 'online',
          count: proxies.length,
          protocols: stats.protocols,
          error: null
        });
        return proxies;
      })
      .catch(error => {
        const friendlyName = getFriendlySourceName(source.url);
        sourceReports.push({
          source: friendlyName,
          status: 'offline',
          count: 0,
          protocols: {},
          error: error.message
        });
        return [];
      })
  );

  const proxyGroups = await Promise.all(promises);
  const allProxies = proxyGroups.flat();
  const totalFetched = allProxies.length;
  const uniqueProxies = [...new Set(allProxies)];
  const duplicatesRemoved = totalFetched - uniqueProxies.length;

  const protocolCounts = {};
  for (const p of uniqueProxies) {
    const proto = (p.split('://')[0] || 'other').toLowerCase();
    protocolCounts[proto] = (protocolCounts[proto] || 0) + 1;
  }

  const report = {
    خلاصه_گزارش: {
      'تعداد کل': totalFetched,
      'تکراری حذف شده': duplicatesRemoved,
      'پروکسی‌های یکتا': uniqueProxies.length,
      'پروتکل‌ها': protocolCounts
    },
    جزئیات_منابع: sourceReports,
  };

  return { uniqueProxies, report };
}

function parseProxies(data, source) {
  let proxies = [];
  const protocolStats = {};
  const addProxy = (proto, ip, port) => {
    const clean = cleanProxyString(proto, ip, port);
    if (clean) {
      proxies.push(clean);
      const pr = proto.toLowerCase();
      protocolStats[pr] = (protocolStats[pr] || 0) + 1;
    }
  };

  if (source.type.startsWith('json_') && source.type !== 'json_geonode' && !Array.isArray(data)) {
    return { proxies: [], stats: { protocols: {} } };
  }

  switch (source.type) {
    case 'text':
      (typeof data === 'string' ? data : '').split('\n').forEach(line => {
        line = line.trim();
        if (!line) return;
        const match = line.match(/^([a-zA-Z0-9]+):\/\/([^:]+):(\d+)/);
        if (match) {
          addProxy(match[1], match[2], match[3]);
        } else if (line.includes(':')) {
          const [ip, port] = line.split(':');
          addProxy('http', ip, port);
        }
      });
      break;

    case 'text_simple':
      (typeof data === 'string' ? data : '').split('\n').forEach(line => {
        line = line.trim();
        if (line && line.includes(':')) {
          const [ip, port] = line.split(':');
          addProxy(source.protocol, ip, port);
        }
      });
      break;

    case 'text_clarketm':
      (typeof data === 'string' ? data : '').split('\n').forEach(line => {
        const m = line.trim().match(/^([0-9.]+):(\d+)\s+IR/);
        if (m) {
          addProxy('http', m[1], m[2]);
        }
      });
      break;

    case 'ndjson_fate0':
      (typeof data === 'string' ? data : '').split('\n').forEach(line => {
        line = line.trim();
        if (!line) return;
        try {
          const item = JSON.parse(line);
          if (item && item.country === 'IR') {
            addProxy(item.type || 'http', item.host, item.port);
          }
        } catch (e) {}
      });
      break;

    case 'json_geonode':
      if (data && Array.isArray(data.data)) {
        data.data.forEach(item => {
          const proto = (item.protocols && item.protocols.length > 0 ? item.protocols[0] : 'http').toLowerCase();
          addProxy(proto, item.ip, item.port);
        });
      }
      break;

    case 'json_sunny':
      if (Array.isArray(data)) {
        data.forEach(p => {
          if (p.country && p.country.startsWith('Iran')) {
            addProxy(source.protocol, p.ip, p.port);
          }
        });
      }
      break;

    case 'json_themiralay':
      if (Array.isArray(data)) {
        data.forEach(p => {
          if (p.geolocation && p.geolocation.countryCode === 'IR') {
            addProxy('http', p.ip, p.port);
          }
        });
      }
      break;

    case 'json_vakhov':
      if (Array.isArray(data)) {
        data.forEach(p => {
          if (p.country_code === 'IR') {
            if (p.socks5 === '1') addProxy('socks5', p.ip, p.port);
            else if (p.socks4 === '1') addProxy('socks4', p.ip, p.port);
            else if (p.ssl === '1') addProxy('https', p.ip, p.port);
            else if (p.http === '1') addProxy('http', p.ip, p.port);
          }
        });
      }
      break;

    case 'html_proxyspace':
      for (const m of (typeof data === 'string' ? data : '').matchAll(/\b(\d{1,3}(?:\.\d{1,3}){3}):(\d{2,5})\b/g)) {
        addProxy('http', m[1], m[2]);
      }
      break;

    case 'html_proxycheck':
      for (const m of (typeof data === 'string' ? data : '').matchAll(/(\d{1,3}(?:\.\d{1,3}){3})[\s\S]*?(\d{2,5})[\s\S]*?(http|https|socks5|socks4)/gi)) {
        addProxy(m[3], m[1], m[2]);
      }
      break;

    case 'html_freeproxyupdate':
      for (const m of (typeof data === 'string' ? data : '').matchAll(/<tr>\s*<td>\s*(\d{1,3}(?:\.\d{1,3}){3})\s*<\/td>\s*<td>\s*(\d{1,5})\s*<\/td>[\s\S]*?<td>\s*<a[^>]*>\s*(https|http|socks4|socks5)/gi)) {
        addProxy(m[3], m[1], m[2]);
      }
      break;

    case 'html_freeproxyworld':
      for (const m of (typeof data === 'string' ? data : '').matchAll(/<td[^>]*>\s*(\d{1,3}(?:\.\d{1,3}){3})\s*<\/td>[\s\S]*?port=(\d{1,5})[\s\S]*?type=([a-zA-Z0-9]+)/gi)) {
        addProxy(m[3], m[1], m[2]);
      }
      break;

    case 'html_proxyspider':
      for (const m of (typeof data === 'string' ? data : '').matchAll(/title="(\d{1,3}(?:\.\d{1,3}){3})"[\s\S]*?title="(\d{1,5})"/g)) {
        addProxy('http', m[1], m[2]);
      }
      break;
  }

  return { proxies, stats: { protocols: protocolStats } };
}

function formatForClash(proxies) {
  let clash = 'proxies:\n';
  for (const proxyStr of proxies) {
    const match = proxyStr.match(/^(socks5|socks4|https|http):\/\/([^:]+):(\d+)$/);
    if (!match) continue;

    let [, protocol, server, port] = match;
    let type = protocol;
    let extra = '';

    if (protocol === 'https') {
      type = 'http';
      extra += '    tls: true\n    skip-cert-verify: true\n';
    } else if (protocol === 'socks5') {
      extra += '    udp: true\n';
    }

    if (!['http', 'socks5'].includes(type)) continue;

    const name = `IR - ${protocol.toUpperCase()} - ${server}:${port}`;
    clash += `  - name: "${name}"\n`;
    clash += `    type: ${type}\n`;
    clash += `    server: ${server}\n`;
    clash += `    port: ${port}\n`;
    clash += extra;
  }
  return clash;
}

  <script>
    const ALL_PROXIES = ${jsonProxies};
    let currentFilter = 'all';
    let searchQuery = '';

    function updateTerminal() {
      let filtered = ALL_PROXIES.filter(p => {
        const proto = (p.split('://')[0] || '').toLowerCase();
        const matchesProto = (currentFilter === 'all') || (proto === currentFilter);
        const matchesSearch = !searchQuery || p.toLowerCase().includes(searchQuery.toLowerCase());
        return matchesProto && matchesSearch;
      });

      document.getElementById('terminalBox').value = filtered.join('\\n');
      document.getElementById('filteredCount').textContent = 'نمایش: ' + filtered.length + ' از ' + ALL_PROXIES.length;
    }

    function setProtocolFilter(proto) {
      currentFilter = proto;
      document.querySelectorAll('.tab-btn').forEach(btn => {
        btn.classList.toggle('active', btn.getAttribute('data-proto') === proto);
      });
      updateTerminal();
    }

    function handleSearch() {
      searchQuery = document.getElementById('searchInput').value.trim();
      updateTerminal();
    }

    function showToast(msg) {
      const toast = document.getElementById('toast');
      toast.textContent = msg || '✅ متن با موفقیت کپی شد!';
      toast.classList.add('show');
      setTimeout(() => toast.classList.remove('show'), 2200);
    }

    function copyText(text) {
      navigator.clipboard.writeText(text).then(() => showToast('✅ لینک کپی شد!'));
    }

    function copyVisibleProxies() {
      const val = document.getElementById('terminalBox').value;
      if (!val) return;
      navigator.clipboard.writeText(val).then(() => showToast('✅ لیست پروکسی‌های فیلترشده کپی شد!'));
    }

    function downloadTXT() {
      const val = document.getElementById('terminalBox').value;
      if (!val) return;
      const blob = new Blob([val], { type: 'text/plain;charset=utf-8' });
      const a = document.createElement('a');
      a.href = URL.createObjectURL(blob);
      a.download = 'iran_proxies.txt';
      a.click();
    }

    function downloadJSON() {
      const val = document.getElementById('terminalBox').value;
      if (!val) return;
      const list = val.split('\\n').filter(Boolean);
      const blob = new Blob([JSON.stringify(list, null, 2)], { type: 'application/json;charset=utf-8' });
      const a = document.createElement('a');
      a.href = URL.createObjectURL(blob);
      a.download = 'iran_proxies.json';
      a.click();
    }

    // منطق سیستم تست زنده و عیب‌یابی دسته‌ای
    let testOffset = 0;
    let isAutoTesting = false;
    let totalTestedCount = 0;
    let aliveTestedProxies = [];
    const BATCH_SIZE = 10;

    async function loadInitialTestedList() {
      try {
        const resp = await fetch('/api/tested-list');
        if (resp.ok) {
          const data = await resp.json();
          if (data && data.success && Array.isArray(data.proxies) && data.proxies.length > 0) {
            aliveTestedProxies = data.proxies;
            updateTestedUI();
            document.getElementById('progressText').textContent = 'حافظه قبلی لود شد (' + aliveTestedProxies.length + ' پروکسی سالم)';
          }
        }
      } catch (_) {}
    }

    function updateTestedUI() {
      const box = document.getElementById('testedTerminalBox');
      box.value = aliveTestedProxies.join('\\n');
      document.getElementById('statAliveCount').textContent = aliveTestedProxies.length;
      document.getElementById('statTestedCount').textContent = totalTestedCount;
      document.getElementById('testedListCount').textContent = 'لیست پروکسی‌های تاییدشده سالم (' + aliveTestedProxies.length + ' مورد)';

      const percent = ALL_PROXIES.length > 0 ? Math.min(100, Math.round((totalTestedCount / ALL_PROXIES.length) * 100)) : 0;
      document.getElementById('progressBarFill').style.width = percent + '%';
      document.getElementById('progressPercent').textContent = percent + '٪';

      const rate = totalTestedCount > 0 ? Math.round((aliveTestedProxies.length / totalTestedCount) * 100) : 0;
      document.getElementById('statSuccessRate').textContent = rate + '٪';
    }

    async function runSingleBatchTest() {
      if (testOffset >= ALL_PROXIES.length) {
        showToast('ℹ️ تمام پروکسی‌های مخزن بررسی شده‌اند!');
        document.getElementById('progressText').textContent = 'پایان تست: تمام پروکسی‌ها بررسی شدند.';
        return false;
      }

      const btnBatch = document.getElementById('btnSingleBatch');
      btnBatch.disabled = true;
      btnBatch.textContent = 'در حال تست...';

      const currentBatchStart = testOffset;
      document.getElementById('progressText').textContent = 'در حال تست دسته ' + (Math.floor(testOffset / BATCH_SIZE) + 1) + ' (پروکسی‌های ' + (currentBatchStart + 1) + ' تا ' + Math.min(testOffset + BATCH_SIZE, ALL_PROXIES.length) + ')...';

      try {
        const resp = await fetch('/api/test-batch?offset=' + testOffset + '&limit=' + BATCH_SIZE);
        const data = await resp.json();

        if (data && data.success) {
          totalTestedCount += data.batchSize;
          if (Array.isArray(data.aliveProxies) && data.aliveProxies.length > 0) {
            data.aliveProxies.forEach(p => {
              if (!aliveTestedProxies.includes(p)) {
                aliveTestedProxies.push(p);
              }
            });
          }

          if (data.nextOffset !== null && data.nextOffset !== undefined) {
            testOffset = data.nextOffset;
          } else {
            testOffset = ALL_PROXIES.length;
          }

          updateTestedUI();
          document.getElementById('progressText').textContent = 'دسته تکمیل شد. یافته جدید: ' + data.batchAliveCount + ' مورد سالم.';
          btnBatch.disabled = false;
          btnBatch.textContent = 'تست دستی یک دسته (۱۰ تایی)';
          return testOffset < ALL_PROXIES.length;
        } else {
          throw new Error('خطا در پاسخ سرور');
        }
      } catch (err) {
        document.getElementById('progressText').textContent = 'خطا در اجرای تست: ' + err.message;
        btnBatch.disabled = false;
        btnBatch.textContent = 'تست دستی یک دسته (۱۰ تایی)';
        return false;
      }
    }

    async function toggleAutoTest() {
      const btn = document.getElementById('btnAutoTest');
      if (isAutoTesting) {
        isAutoTesting = false;
        btn.classList.remove('running');
        btn.textContent = '⚡ ادامه تست خودکار هوشمند (پیوسته)';
        document.getElementById('progressText').textContent = 'تست خودکار متوقف شد.';
        return;
      }

      isAutoTesting = true;
      btn.classList.add('running');
      btn.textContent = '⏸️ توقف تست خودکار';

      while (isAutoTesting && testOffset < ALL_PROXIES.length) {
        const hasMore = await runSingleBatchTest();
        if (!hasMore || !isAutoTesting) break;
        // وقفه ۳۰۰ میلی‌ثانیه‌ای بین دسته‌ها
        await new Promise(r => setTimeout(r, 300));
      }

      isAutoTesting = false;
      btn.classList.remove('running');
      if (testOffset >= ALL_PROXIES.length) {
        btn.textContent = '✅ تست کامل به پایان رسید';
        document.getElementById('progressText').textContent = '🎉 تمام پروکسی‌ها با موفقیت آزمایش شدند.';
        showToast('🎉 فرایند تست ۱۰۰٪ با موفقیت پایان یافت!');
      } else {
        btn.textContent = '⚡ ادامه تست خودکار هوشمند (پیوسته)';
      }
    }

    async function resetTestState() {
      if (!confirm('آیا از ریست کردن حافظه پروکسی‌های تست‌شده مطمئن هستید؟')) return;
      try {
        await fetch('/api/reset-tested');
      } catch (_) {}
      isAutoTesting = false;
      testOffset = 0;
      totalTestedCount = 0;
      aliveTestedProxies = [];
      const btn = document.getElementById('btnAutoTest');
      btn.classList.remove('running');
      btn.textContent = '⚡ شروع تست خودکار هوشمند (پیوسته)';
      updateTestedUI();
      document.getElementById('testedTerminalBox').value = '';
      document.getElementById('progressText').textContent = 'حافظه پروکسی‌های تست‌شده ریست شد.';
      showToast('🗑️ حافظه پروکسی‌های تست‌شده با موفقیت پاک شد!');
    }
