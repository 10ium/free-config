export default {
  async fetch(request) {
    const url = new URL(request.url);
    const path = url.pathname;

    // Route 1: Favicon request
    if (path === "/favicon.ico") {
      return serveFavicon();
    }

    // Route 2: Landing page request
    if (path === "/") {
      return serveLandingPage();
    }

    // From here, we handle proxy requests.
    const cleanPath = path.slice(1); // e.g., "github.com/user/..." or "user/repo/..."
    const firstPart = cleanPath.split('/')[0];

    // Route 3: Generic Proxy (if path looks like a domain, e.g., "github.com/...")
    // This preserves the old behavior for full URLs.
    if (firstPart.includes('.')) {
      return handleProxyRequest(url);
    }
    
    // Route 4: New GitHub-specific Proxy (if path is like "user/repo/branch/...")
    // This is the new, cleaner URL format.
    return handleGitHubProxyRequest(path);
  }
};

/**
 * An SVG favicon representing a link is created and served.
 * (This function is unchanged)
 */
function serveFavicon() {
  const faviconSvg = `
  <svg xmlns="http://www.w3.org/2000/svg" width="32" height="32" viewBox="0 0 24 24" fill="none" stroke="#007bff" stroke-width="2" stroke-linecap="round" stroke-linejoin="round">
    <path d="M10 13a5 5 0 0 0 7.54.54l3-3a5 5 0 0 0-7.07-7.07l-1.72 1.72"></path>
    <path d="M14 11a5 5 0 0 0-7.54-.54l-3 3a5 5 0 0 0 7.07 7.07l1.72-1.72"></path>
  </svg>
  `.trim();
  return new Response(faviconSvg, {
    headers: { 'Content-Type': 'image/svg+xml' },
  });
}

/**
 * Handles incoming requests as a generic proxy.
 * (This function is unchanged)
 * @param {URL} url - The URL object from the incoming request.
 */
async function handleProxyRequest(url) {
  let targetUrlString = url.pathname.slice(1);
  if (!targetUrlString.startsWith('http')) {
    targetUrlString = 'https://' + targetUrlString;
  }
  if (targetUrlString.includes('github.com') && !targetUrlString.includes('raw.githubusercontent.com')) {
    targetUrlString = targetUrlString.replace('github.com', 'raw.githubusercontent.com').replace('/blob/', '/');
  }
  try {
    const response = await fetch(targetUrlString);
    if (!response.ok) {
      if (response.status === 404) {
        return new Response('محتوا در آدرس مقصد پیدا نشد. لطفاً لینک را بررسی کنید. (خطای 404)', {
          status: 404,
          headers: { 'Content-Type': 'text/plain; charset=utf-8' },
        });
      }
      return new Response(`سرور مقصد با خطا پاسخ داد: ${response.status} ${response.statusText}`, {
        status: response.status,
        headers: { 'Content-Type': 'text/plain; charset=utf-8' },
      });
    }
    const headers = new Headers(response.headers);
    headers.set('Access-Control-Allow-Origin', '*');
    headers.set('Access-Control-Allow-Methods', 'GET, HEAD, POST, OPTIONS');
    return new Response(response.body, {
      status: response.status,
      headers: headers,
    });
  } catch (error) {
    return new Response(`خطا در اتصال به سرور مقصد: ${error.message}`, {
      status: 500,
      headers: { 'Content-Type': 'text/plain; charset=utf-8' },
    });
  }
}

/**
 * Handles requests for GitHub raw content using the clean URL format /owner/repo/branch/file.
 * (This function is unchanged)
 * @param {string} path - The pathname from the incoming request.
 */
async function handleGitHubProxyRequest(path) {
  const cleanPath = path.replace(/^\/+/, ""); // remove leading slash
  const parts = cleanPath.split("/");

  if (parts.length < 3) { // It should be at least owner/repo/branch
    return new Response(
      "Invalid GitHub path. Usage: /owner/repo/branch/path/to/file.ext",
      { 
        status: 400,
        headers: { 'Content-Type': 'text/plain; charset=utf-8' },
      }
    );
  }

  const [owner, repo, ...restOfPathParts] = parts;
  const restOfPath = restOfPathParts.join("/");
  const rawUrl = `https://raw.githubusercontent.com/${owner}/${repo}/${restOfPath}`;

  try {
    const response = await fetch(rawUrl);
    if (!response.ok) {
      return new Response(`Error fetching from GitHub: ${response.status} ${response.statusText}`, {
        status: response.status,
        headers: { 'Content-Type': 'text/plain; charset=utf-8' },
      });
    }
    const headers = new Headers(response.headers);
    headers.set('Access-Control-Allow-Origin', '*');
    headers.set('Access-Control-Allow-Methods', 'GET, HEAD, POST, OPTIONS');
    return new Response(response.body, {
      status: response.status,
      headers: headers,
    });
  } catch (error) {
    return new Response(`Network error while fetching from GitHub: ${error.message}`, {
      status: 500,
      headers: { 'Content-Type': 'text/plain; charset=utf-8' },
    });
  }
}
